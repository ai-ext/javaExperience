const readD12 = ()=>{const r=document.querySelector('[data-d12-learning]'),img=r.querySelector('[data-d12-image]'),fr=r.querySelector('[data-d12-image-frame]'),c=r.querySelector('[data-d12-code]');return {topic:r.dataset.topic,stage:r.dataset.stage,title:r.querySelector('[data-d12-title]').textContent,lead:r.querySelector('[data-d12-lead]').textContent,code:c.textContent,codeKind:r.querySelector('[data-d12-code-kind]').textContent,evidenceKind:r.querySelector('[data-d12-evidence-kind]').textContent,values:[...r.querySelectorAll('[data-d12-state] dt')].map(t=>[t.textContent,t.nextElementSibling.textContent]),image:{src:img.currentSrc,natural:[img.naturalWidth,img.naturalHeight],rect:img.getBoundingClientRect().toJSON()},frame:fr.getBoundingClientRect().toJSON(),zoom:r.querySelector('[data-d12-zoom]').getAttribute('aria-pressed'),codeFont:getComputedStyle(c).fontSize,valueFonts:[...r.querySelectorAll('[data-d12-state] dd')].map(e=>getComputedStyle(e).fontSize),overflow:document.documentElement.scrollWidth>innerWidth,workspaceDisplay:getComputedStyle(r.querySelector('.p09-workspace')).display,observationColumns:getComputedStyle(r.querySelector('.p09-observation')).gridTemplateColumns,url:location.href};};

const replayD12 = async(target,read,topic,stage)=>{await target.playwright.locator('[data-d12-topic]').selectOption(topic);await target.playwright.locator(`[data-d12-stage="${stage}"]`).click();await target.getAXState({emit:false});if(await target.playwright.locator('[data-d12-zoom]').getAttribute('aria-pressed')==='true')await target.playwright.locator('[data-d12-zoom]').click();let row=await target.playwright.evaluate(read);await target.playwright.locator('[data-d12-copy]').click();row.clipboardPayload=await target.clipboard.readText();row.clipboardExact=row.clipboardPayload===row.code;row.copyStatus=await target.playwright.locator('[data-d12-copy-status]').textContent();let r=await target.playwright.evaluate(read);for(let i=0;i<3&&(r.frame.top<100||r.frame.bottom>930);i++){const d=r.frame.top-170;await target.scroll([1540,580],d>0?'down':'up',Math.abs(d)/941);await target.getAXState({emit:false});r=await target.playwright.evaluate(read);}row.baseObservation=r;await fs24.writeFile(`/tmp/d12-browser/${stage}-pc-dark.png`,await target.screenshot({}));await target.playwright.locator('[data-d12-zoom]').click();r=await target.playwright.evaluate(read);for(let i=0;i<3&&(r.frame.top<100||r.frame.bottom>930);i++){const d=r.frame.top-170;await target.scroll([1540,580],d>0?'down':'up',Math.abs(d)/941);await target.getAXState({emit:false});r=await target.playwright.evaluate(read);}row.zoomObservation=r;await fs24.writeFile(`/tmp/d12-browser/${stage}-zoom-pc-dark.png`,await target.screenshot({}));return row;};

const contrastD12 = ()=>{const parse=c=>{const a=c.match(/[\d.]+/g);return a&&a.length>=3?{rgb:a.slice(0,3).map(Number),alpha:a.length>3?Number(a[3]):1}:null;};const lum=c=>c.map(v=>v/255).map(v=>v<=.04045?v/12.92:((v+.055)/1.055)**2.4).reduce((n,v,i)=>n+v*[.2126,.7152,.0722][i],0);let rows=[],skipped=0;for(const el of document.querySelectorAll('body *')){if(['SCRIPT','STYLE','NOSCRIPT','TEMPLATE'].includes(el.tagName)||el.closest('svg,canvas')||!el.getClientRects().length||!Array.from(el.childNodes).some(n=>n.nodeType===3&&n.textContent.trim())||el.closest('button:disabled,[aria-disabled="true"]'))continue;let skip=false,bg=null;for(let p=el;p;p=p.parentElement){const s=getComputedStyle(p);if(p.hidden||s.display==='none'||s.visibility==='hidden'||Number(s.opacity)<.99||s.backgroundImage!=='none'){skip=true;break;}const c=parse(s.backgroundColor);if(c&&c.alpha>0&&c.alpha<.999){skip=true;break;}if(!bg&&c?.alpha>=.999)bg=c.rgb;}const s=getComputedStyle(el),fg=parse(s.color);if(skip||!bg||!fg||fg.alpha<.999){skipped++;continue;}const a=lum(fg.rgb),b=lum(bg),ratio=(Math.max(a,b)+.05)/(Math.min(a,b)+.05),size=Number(s.fontSize.replace('px','')),weight=Number(s.fontWeight)||400,target=size>=24||size>=18.666&&weight>=700?3:4.5;rows.push({tag:el.tagName,text:el.textContent.trim().slice(0,70),fg:fg.rgb,bg,size,weight,ratio,target,pass:ratio>=target});}return {topic:document.querySelector('[data-d12-topic]').value,stage:document.querySelector('[data-d12-stage][aria-current="step"]').dataset.d12Stage,viewport:[innerWidth,innerHeight],theme:document.documentElement.dataset.theme,samples:rows.length,skipped,failures:rows.filter(x=>!x.pass),rows};};
