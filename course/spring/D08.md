```java
package com.example.board.config;

import org.apache.hc.client5.http.config.ConnectionConfig;
import org.apache.hc.client5.http.config.RequestConfig;
import org.apache.hc.client5.http.impl.classic.CloseableHttpClient;
import org.apache.hc.client5.http.impl.classic.HttpClients;
import org.apache.hc.client5.http.impl.io.PoolingHttpClientConnectionManager;
import org.apache.hc.client5.http.impl.io.PoolingHttpClientConnectionManagerBuilder;
import org.apache.hc.core5.util.Timeout;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.client.HttpComponentsClientHttpRequestFactory;
import org.springframework.web.client.RestClient;

import com.example.board.payment.PgClient;

@Configuration(proxyBeanMethods = false)
public class PgClientConfig {

    @Bean
    PgClient pgClient(BoardProperties properties) {
        return new PgClient(pgRestClient(properties.pg()));
    }

    /** 연결·응답·풀 대기 타임아웃을 따로 건다. 테스트가 풀 크기를 바꿔 만들 수 있게 정적 메서드로 둔다. */
    public static RestClient pgRestClient(BoardProperties.Pg pg) {
        PoolingHttpClientConnectionManager pool = PoolingHttpClientConnectionManagerBuilder.create()
                .setDefaultConnectionConfig(ConnectionConfig.custom()
                        .setConnectTimeout(Timeout.of(pg.connectTimeout()))
                        .setSocketTimeout(Timeout.of(pg.responseTimeout()))
                        .build())
                .setMaxConnTotal(pg.maxConnections())
                .setMaxConnPerRoute(pg.maxConnections())
                .build();
        CloseableHttpClient httpClient = HttpClients.custom()
                .setConnectionManager(pool)
                .setDefaultRequestConfig(RequestConfig.custom()
                        .setConnectionRequestTimeout(Timeout.of(pg.connectionRequestTimeout()))
                        .setResponseTimeout(Timeout.of(pg.responseTimeout()))
                        .build())
                // 재시도는 우리 정책 한 곳에서만 한다. 클라이언트 기본 재시도가 겹치면 횟수를 셀 수 없다.
                .disableAutomaticRetries()
                .build();
        return RestClient.builder()
                .baseUrl(pg.baseUrl())
                .requestFactory(new HttpComponentsClientHttpRequestFactory(httpClient))
                .build();
    }
}
```

```java
package com.example.board.payment;

import java.net.ConnectException;
import java.net.NoRouteToHostException;
import java.net.UnknownHostException;

import org.apache.hc.client5.http.ConnectTimeoutException;
import org.apache.hc.core5.http.ConnectionRequestTimeoutException;
import org.springframework.web.client.HttpClientErrorException;
import org.springframework.web.client.HttpServerErrorException;
import org.springframework.web.client.ResourceAccessException;
import org.springframework.web.client.RestClient;

/** 외부 PG 호출. DB 트랜잭션 안에서 부르지 않는다(spec-v2 §8). */
public class PgClient {

    private final RestClient restClient;

    public PgClient(RestClient restClient) {
        this.restClient = restClient;
    }

    public PgApproval approve(String orderId, int amount, String idempotencyKey) {
        try {
            return restClient.post()
                    .uri("/pg/v1/payments")
                    .header("Idempotency-Key", idempotencyKey)
                    .body(new ApproveRequest(orderId, amount))
                    .retrieve()
                    .body(PgApproval.class);
        } catch (RuntimeException e) {
            throw translate(e);
        }
    }

    private record ApproveRequest(String orderId, int amount) {
    }

    static PgException translate(RuntimeException e) {
        if (e instanceof HttpServerErrorException) {
            return new PgException(PgException.Kind.SERVER_ERROR, "PG 서버 오류", e);
        }
        if (e instanceof HttpClientErrorException) {
            return new PgException(PgException.Kind.DECLINED, "PG 승인 거절", e);
        }
        if (e instanceof ResourceAccessException) {
            Throwable cause = e.getCause();
            if (cause instanceof ConnectionRequestTimeoutException) {
                return new PgException(PgException.Kind.POOL_TIMEOUT, "PG 커넥션 풀 대기 시간 초과", e);
            }
            if (cause instanceof ConnectException || cause instanceof ConnectTimeoutException
                    || cause instanceof NoRouteToHostException || cause instanceof UnknownHostException) {
                return new PgException(PgException.Kind.UNREACHABLE, "PG 연결 실패", e);
            }
            // 응답 타임아웃과 응답 도중 끊김은 모두 결과 불명이다
            return new PgException(PgException.Kind.TIMEOUT, "PG 응답 없음", e);
        }
        throw e;
    }
}
```

```java
package com.example.board.payment;

/** PG 호출 실패를 "요청이 PG에 닿았는가"로 나눈다. 재시도와 상태 결정이 이 구분에 달려 있다. */
public class PgException extends RuntimeException {

    public enum Kind {
        /** 연결을 맺지 못했다. PG는 요청을 받지 않았다. */
        UNREACHABLE,
        /** 우리 커넥션 풀에서 빈 커넥션을 얻지 못했다. 요청을 보내지 않았다. */
        POOL_TIMEOUT,
        /** 요청은 보냈지만 응답을 받지 못했다. PG는 승인했을 수 있다(결과 불명). */
        TIMEOUT,
        /** PG가 5xx로 답했다. */
        SERVER_ERROR,
        /** PG가 4xx로 거절했다. 같은 요청은 같은 실패를 낸다. */
        DECLINED
    }

    private final Kind kind;

    public PgException(Kind kind, String message, Throwable cause) {
        super(message, cause);
        this.kind = kind;
    }

    public Kind kind() {
        return kind;
    }
}
```

```java
package com.example.board;

import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.ArrayList;
import java.util.List;

import org.testcontainers.containers.GenericContainer;
import org.testcontainers.containers.Network;
import org.testcontainers.containers.wait.strategy.Wait;
import org.testcontainers.toxiproxy.ToxiproxyContainer;
import org.testcontainers.utility.MountableFile;

import eu.rekawek.toxiproxy.Proxy;
import eu.rekawek.toxiproxy.ToxiproxyClient;
import eu.rekawek.toxiproxy.model.Toxic;
import tools.jackson.databind.JsonNode;
import tools.jackson.databind.json.JsonMapper;

/**
 * PG 모의(WireMock)와 그 앞단 Toxiproxy. JVM 안에서 한 번만 띄워 모든 테스트가 같이 쓴다.
 * 서버는 Toxiproxy를 거쳐 WireMock에 닿으므로 지연·연결 끊김을 주입할 수 있다.
 */
public final class PgContainers {

    private static final int PROXY_PORT = 8666;
    private static final Network NETWORK = Network.newNetwork();
    private static final GenericContainer<?> WIREMOCK = new GenericContainer<>("wiremock/wiremock:3.13.1")
            .withNetwork(NETWORK)
            .withNetworkAliases("wiremock")
            .withExposedPorts(8080)
            .withCopyFileToContainer(MountableFile.forClasspathResource("wiremock"), "/home/wiremock")
            .waitingFor(Wait.forHttp("/__admin/mappings").forStatusCode(200));
    private static final ToxiproxyContainer TOXIPROXY = new ToxiproxyContainer("ghcr.io/shopify/toxiproxy:2.12.0")
            .withNetwork(NETWORK);
    private static final HttpClient HTTP = HttpClient.newHttpClient();
    private static final JsonMapper JSON = JsonMapper.builder().build();

    private static Proxy proxy;

    private PgContainers() {
    }

    public static synchronized void start() {
        if (proxy != null) {
            return;
        }
        WIREMOCK.start();
        TOXIPROXY.start();
        try {
            proxy = new ToxiproxyClient(TOXIPROXY.getHost(), TOXIPROXY.getControlPort())
                    .createProxy("pg", "0.0.0.0:" + PROXY_PORT, "wiremock:8080");
        } catch (IOException e) {
            throw new IllegalStateException(e);
        }
    }

    /** 서버가 쓰는 PG 주소(Toxiproxy 경유). */
    public static String baseUrl() {
        start();
        return "http://" + TOXIPROXY.getHost() + ":" + TOXIPROXY.getMappedPort(PROXY_PORT);
    }

    public static Proxy proxy() {
        start();
        return proxy;
    }

    /** 테스트마다 지연 주입·임시 스텁·요청 기록을 지우고 파일 스텁만 남긴다. */
    public static void reset() {
        start();
        try {
            for (Toxic toxic : proxy.toxics().getAll()) {
                toxic.remove();
            }
            proxy.enable();
        } catch (IOException e) {
            throw new IllegalStateException(e);
        }
        admin("POST", "/__admin/reset", "");
    }

    /** WireMock 스텁 JSON을 등록한다. 파일 스텁보다 먼저 맞도록 priority를 낮게 준다. */
    public static void stub(String mappingJson) {
        admin("POST", "/__admin/mappings", mappingJson);
    }

    /** POST /pg/v1/payments 로 받은 요청의 Idempotency-Key 목록(받은 순서). */
    public static List<String> approveIdempotencyKeys() {
        JsonNode found = JSON.readTree(admin("POST", "/__admin/requests/find",
                "{\"method\":\"POST\",\"urlPath\":\"/pg/v1/payments\"}"));
        List<String> keys = new ArrayList<>();
        for (JsonNode r : found.get("requests")) {
            keys.add(r.get("headers").get("Idempotency-Key").asString());
        }
        return keys.reversed();
    }

    public static int approveRequestCount() {
        return approveIdempotencyKeys().size();
    }

    private static String admin(String method, String path, String body) {
        start();
        URI uri = URI.create("http://" + WIREMOCK.getHost() + ":" + WIREMOCK.getMappedPort(8080) + path);
        HttpRequest request = HttpRequest.newBuilder(uri)
                .header("Content-Type", "application/json")
                .method(method, HttpRequest.BodyPublishers.ofString(body))
                .build();
        try {
            HttpResponse<String> response = HTTP.send(request, HttpResponse.BodyHandlers.ofString());
            if (response.statusCode() >= 300) {
                throw new IllegalStateException(path + " -> " + response.statusCode() + " " + response.body());
            }
            return response.body();
        } catch (IOException e) {
            throw new IllegalStateException(e);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IllegalStateException(e);
        }
    }
}
```

```java
    @Column(name = "responded_at")
    private LocalDateTime respondedAt;
```

```java
package com.example.board.payment;

import java.time.LocalDateTime;
import java.time.ZoneId;

import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.support.TransactionTemplate;

import com.example.board.common.BusinessException;
import com.example.board.common.ErrorCode;
import com.example.board.member.MemberRepository;
import com.example.board.point.PointService;

/**
 * 충전은 세 단계다: REQUESTED 저장(트랜잭션 1) → PG 승인(트랜잭션 밖) → 결과 반영(트랜잭션 2).
 * 이 클래스에 @Transactional을 두지 않는다. 두면 PG를 기다리는 동안 DB 커넥션을 쥐어 Hikari 풀이 바닥난다.
 */
@Service
public class ChargeService {

    private static final ZoneId KST = ZoneId.of("Asia/Seoul");
    private static final String REF_TYPE = "PG_PAYMENT";

    private final PgPaymentRepository paymentRepository;
    private final MemberRepository memberRepository;
    private final PointService pointService;
    private final PgClient pgClient;
    private final TransactionTemplate tx;

    public ChargeService(PgPaymentRepository paymentRepository, MemberRepository memberRepository,
            PointService pointService, PgClient pgClient, TransactionTemplate tx) {
        this.paymentRepository = paymentRepository;
        this.memberRepository = memberRepository;
        this.pointService = pointService;
        this.pgClient = pgClient;
        this.tx = tx;
    }

    public ChargeResponse charge(Long memberId, String idempotencyKey, int amount) {
        Started started = start(memberId, idempotencyKey, amount);
        PgPayment payment = started.payment();
        if (!started.fresh()) {
            return replay(payment);
        }

        PgApproval approval;
        try {
            approval = pgClient.approve(payment.getOrderId(), amount, idempotencyKey);
        } catch (PgException e) {
            throw fail(payment, e);
        }
        return apply(payment, approval);
    }

    /** 트랜잭션 1: 호출 전에 REQUESTED를 커밋해 둔다. 같은 키가 이미 있으면 그 기록을 돌려준다. */
    private Started start(Long memberId, String idempotencyKey, int amount) {
        try {
            return tx.execute(s -> paymentRepository.findByIdempotencyKey(idempotencyKey)
                    .map(existing -> new Started(sameRequest(existing, memberId, amount), false))
                    .orElseGet(() -> new Started(paymentRepository.saveAndFlush(PgPayment.request(
                            memberRepository.getReferenceById(memberId), amount, idempotencyKey, now())), true)));
        } catch (DataIntegrityViolationException duplicateKey) {
            // UNIQUE(idempotency_key) 위반 = 같은 키의 요청이 먼저 커밋됐다
            return tx.execute(s -> paymentRepository.findByIdempotencyKey(idempotencyKey)
                    .map(existing -> new Started(sameRequest(existing, memberId, amount), false))
                    .orElseThrow(() -> duplicateKey));
        }
    }

    private PgPayment sameRequest(PgPayment existing, Long memberId, int amount) {
        if (!existing.getMember().getId().equals(memberId) || existing.getAmount() != amount) {
            throw new BusinessException(ErrorCode.IDEMPOTENCY_MISMATCH, "같은 Idempotency-Key에 다른 요청 본문입니다.");
        }
        return existing;
    }

    /** 같은 키의 재요청은 PG를 다시 부르지 않는다. 결과 불명 건은 콜백·조회로만 확정한다. */
    private ChargeResponse replay(PgPayment payment) {
        return switch (payment.getStatus()) {
            case DONE -> response(payment);
            case FAILED -> throw declined(payment);
            case REQUESTED, UNKNOWN -> throw timeout(payment);
        };
    }

    /** 트랜잭션 2: 승인 결과를 반영한다. */
    private ChargeResponse apply(PgPayment payment, PgApproval approval) {
        return switch (approval.status()) {
            case DONE -> tx.execute(s -> {
                settleDone(payment, approval.paymentKey());
                return response(reload(payment));
            });
            case FAILED -> {
                tx.executeWithoutResult(s -> paymentRepository.markFailed(payment.getId(), approval.paymentKey(), now()));
                throw declined(payment);
            }
            case PENDING -> tx.execute(s -> {
                paymentRepository.markUnknown(payment.getId(), now());
                return response(reload(payment));
            });
        };
    }

    private BusinessException fail(PgPayment payment, PgException e) {
        return switch (e.kind()) {
            // 요청을 보냈지만 답을 못 받았다. 실패로 단정하지 않는다.
            case TIMEOUT -> {
                markUnknown(payment);
                yield timeout(payment);
            }
            case SERVER_ERROR -> {
                markUnknown(payment);
                yield declined(payment);
            }
            // PG에 닿지 않았거나 PG가 명시적으로 거절했다
            case UNREACHABLE, POOL_TIMEOUT, DECLINED -> {
                tx.executeWithoutResult(s -> paymentRepository.markFailed(payment.getId(), null, now()));
                yield declined(payment);
            }
        };
    }

    /** 영향 행 수 1일 때만 적립한다. 승인 응답·콜백·조회 중 먼저 온 하나만 1을 받는다. */
    void settleDone(PgPayment payment, String paymentKey) {
        if (paymentRepository.markDone(payment.getId(), paymentKey, now()) == 1) {
            pointService.charge(payment.getMember().getId(), payment.getAmount(), REF_TYPE, payment.getId());
        }
    }

    private void markUnknown(PgPayment payment) {
        tx.executeWithoutResult(s -> paymentRepository.markUnknown(payment.getId(), now()));
    }

    private PgPayment reload(PgPayment payment) {
        return paymentRepository.findById(payment.getId()).orElseThrow();
    }

    private ChargeResponse response(PgPayment payment) {
        Long memberId = payment.getMember().getId();
        return new ChargeResponse(payment.getOrderId(), payment.getAmount(), payment.getStatus(),
                pointService.getBalance(memberId));
    }

    private static BusinessException timeout(PgPayment payment) {
        return new BusinessException(ErrorCode.PG_TIMEOUT,
                "PG 응답을 받지 못했습니다. 결과를 확인 중입니다. orderId=" + payment.getOrderId());
    }

    private static BusinessException declined(PgPayment payment) {
        return new BusinessException(ErrorCode.PG_DECLINED, "PG 승인에 실패했습니다. orderId=" + payment.getOrderId());
    }

    private static LocalDateTime now() {
        return LocalDateTime.now(KST);
    }

    private record Started(PgPayment payment, boolean fresh) {
    }
}
```

```java
package com.example.board.payment;

import java.util.concurrent.ThreadLocalRandom;
import java.util.function.Supplier;

/**
 * 연결 실패와 5xx만 다시 보낸다. 최대 3회(첫 시도 포함), 백오프 200ms·400ms + 지터 0~100ms.
 * 응답 타임아웃은 PG가 이미 승인했을 수 있으므로 재시도하지 않고 결과 불명으로 넘긴다.
 * 4xx 거절은 같은 요청이 같은 실패를 내므로 재시도하지 않는다.
 */
public class PgRetryPolicy {

    private static final int DEFAULT_MAX_ATTEMPTS = 3;
    private static final long[] BACKOFF_MILLIS = {200, 400};
    private static final int MAX_JITTER_MILLIS = 100;

    private final int maxAttempts;

    public PgRetryPolicy() {
        this(DEFAULT_MAX_ATTEMPTS);
    }

    public PgRetryPolicy(int maxAttempts) {
        if (maxAttempts < 1 || maxAttempts > BACKOFF_MILLIS.length + 1) {
            throw new IllegalArgumentException("maxAttempts: " + maxAttempts);
        }
        this.maxAttempts = maxAttempts;
    }

    public <T> T execute(Supplier<T> call) {
        for (int attempt = 1;; attempt++) {
            try {
                return call.get();
            } catch (PgException e) {
                if (!retryable(e) || attempt >= maxAttempts) {
                    throw e;
                }
                sleep(BACKOFF_MILLIS[attempt - 1] + ThreadLocalRandom.current().nextInt(MAX_JITTER_MILLIS + 1), e);
            }
        }
    }

    static boolean retryable(PgException e) {
        return e.kind() == PgException.Kind.UNREACHABLE || e.kind() == PgException.Kind.SERVER_ERROR;
    }

    private static void sleep(long millis, PgException last) {
        try {
            Thread.sleep(millis);
        } catch (InterruptedException interrupted) {
            Thread.currentThread().interrupt();
            throw last;
        }
    }
}
```

```java
package com.example.board.payment;

import java.nio.charset.StandardCharsets;
import java.security.GeneralSecurityException;
import java.security.MessageDigest;
import java.util.HexFormat;

import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;

/** X-PG-Signature = 본문 원문 바이트의 HMAC-SHA256, 16진 소문자(spec-v2 §7). */
public class PgSignatureVerifier {

    private static final String ALGORITHM = "HmacSHA256";

    private final SecretKeySpec key;

    public PgSignatureVerifier(String secret) {
        this.key = new SecretKeySpec(secret.getBytes(StandardCharsets.UTF_8), ALGORITHM);
    }

    /** 반드시 파싱 전 원문으로 부른다. 파싱 후 다시 직렬화하면 공백·필드 순서가 바뀌어 바이트가 달라진다. */
    public boolean verify(byte[] rawBody, String signatureHex) {
        if (signatureHex == null || signatureHex.isEmpty()) {
            return false;
        }
        byte[] expected = HexFormat.of().formatHex(hmac(rawBody)).getBytes(StandardCharsets.US_ASCII);
        // equals는 첫 불일치에서 멈춰 응답 시간으로 서명을 한 글자씩 추측할 수 있다. 상수 시간 비교를 쓴다.
        return MessageDigest.isEqual(expected, signatureHex.getBytes(StandardCharsets.US_ASCII));
    }

    private byte[] hmac(byte[] body) {
        try {
            Mac mac = Mac.getInstance(ALGORITHM);
            mac.init(key);
            return mac.doFinal(body);
        } catch (GeneralSecurityException e) {
            throw new IllegalStateException(e);
        }
    }
}
```

### TC-D08-1

**질문** PG 클라이언트에 연결 1s, 응답 3s, 커넥션 풀 대기 2s를 따로 두었습니다. 셋은 각각 무엇을 막나요?

<details class="card-answer" data-card="TC-D08-1">
<summary>답에 들어가야 할 것</summary>

- 연결 타임아웃은 TCP 연결을 맺는 시간, 응답 타임아웃은 요청 후 응답을 기다리는 시간, 풀 대기 타임아웃은 클라이언트 풀에서 빈 커넥션을 얻는 시간이다
- 셋 중 하나라도 비어 있으면 그 구간에서 요청 스레드가 오래(클라이언트에 따라 무한히) 묶일 수 있다
- 값은 명세 운영 값을 쓰고, WireMock·Toxiproxy로 지연을 주입해 각 타임아웃이 어떤 예외로 나는지 직접 확인한다

**근거** spec-v2 §8

</details>

### TC-D08-2

**질문** 포인트 충전 흐름을 트랜잭션 몇 개로 나눴나요? 왜 그렇게 나눴나요?

<details class="card-answer" data-card="TC-D08-2">
<summary>답에 들어가야 할 것</summary>

- `pg_payment`를 `REQUESTED`로 저장하는 짧은 트랜잭션 → 트랜잭션 밖에서 PG 호출 → 결과로 상태를 바꾸고 포인트를 적립하는 짧은 트랜잭션, 세 단계다
- 요청 기록을 먼저 커밋해 두어야 PG 호출 중 서버가 죽어도 무엇을 보냈는지 남는다
- PG를 기다리는 동안 DB 커넥션을 쥐지 않으므로, PG가 느려도 Hikari 풀(10)이 고갈되지 않는다

**근거** spec-v2 §8

</details>

### TC-D08-3

**질문** PG 응답이 3초를 넘겨 타임아웃이 났습니다. 사용자에게 "충전 실패"라고 보여 주면 왜 안 되나요?

<details class="card-answer" data-card="TC-D08-3">
<summary>답에 들어가야 할 것</summary>

- 타임아웃은 실패가 아니라 결과 불명이다. PG는 승인을 끝냈을 수 있다
- `pg_payment`를 `UNKNOWN`으로 남기고 504 `/errors/pg-timeout`을 돌려준다
- 상태는 `orderId`가 담긴 PG 콜백이나 `GET /pg/v1/payments?orderId=` 조회(재처리 워커)로 확정한다. 같은 `Idempotency-Key`로 승인을 다시 요청해도 PG는 처음 결과를 돌려준다

**근거** spec-v2 §5-5, spec-v2 §7

</details>

### TC-D08-4

**질문** PG 호출이 실패하면 무조건 세 번 재시도하게 했다가 리뷰에서 지적을 받았습니다. 무엇을 재시도하고 무엇을 하지 않나요?

<details class="card-answer" data-card="TC-D08-4">
<summary>답에 들어가야 할 것</summary>

- 연결 실패·5xx는 같은 멱등 키로 재시도하고, 응답 타임아웃은 PG가 처리했을 수 있어 재시도하지 않고 조회·콜백으로 결과를 확인한다. 400·401 같은 4xx는 같은 요청이 같은 실패를 내므로 재시도하지 않는다
- 결제 같은 POST는 같은 멱등 키로 재시도할 때만 이중 승인이 생기지 않는다
- 재시도 간격은 지수 백오프와 지터로 늘리고, 전체 시간이 상위 요청의 시간 예산을 넘지 않게 횟수를 제한한다

</details>

### TC-D08-5

**질문** PG 콜백 경로는 인증 없이 공개했습니다. 그러면 아무나 "결제 완료"를 보내 포인트를 받을 수 있지 않나요?

<details class="card-answer" data-card="TC-D08-5">
<summary>답에 들어가야 할 것</summary>

- 공개 경로라도 서버가 본문 원문 바이트를 같은 비밀키로 HMAC-SHA256해 `X-PG-Signature`와 상수 시간으로 비교하고, 다르면 401을 낸다
- 서명은 파싱 전 원문으로 계산한다. JSON을 파싱해 다시 직렬화하면 공백이나 필드 순서가 바뀌어 바이트가 달라질 수 있다
- 같은 `paymentKey` 콜백이 여러 번 와도 한 번만 적립하고, `amount`가 저장된 충전 금액과 다르면 거부한다

**근거** spec-v2 §5-5, spec-v2 §7

</details>

```text
    · 값은 명세 운영 값을 쓰고, WireMock·Toxiproxy로 지연을 주입해 각 타임아웃이 어떤 예외로 나는지 직접 확인한다
- [TC-D08-2] 포인트 충전 흐름을 트랜잭션 몇 개로 나눴나요? 왜 그렇게 나눴나요?
    · pg_payment를 REQUESTED로 저장하는 짧은 트랜잭션 → 트랜잭션 밖에서 PG 호출 → 결과로 상태를 바꾸고 포인트를 적립하는 짧은 트랜잭션, 세 단계다
    · 요청 기록을 먼저 커밋해 두어야 PG 호출 중 서버가 죽어도 무엇을 보냈는지 남는다
    · PG를 기다리는 동안 DB 커넥션을 쥐지 않으므로, PG가 느려도 Hikari 풀(10)이 고갈되지 않는다
- [TC-D08-3] PG 응답이 3초를 넘겨 타임아웃이 났습니다. 사용자에게 "충전 실패"라고 보여 주면 왜 안 되나요?
    · 타임아웃은 실패가 아니라 결과 불명이다. PG는 승인을 끝냈을 수 있다
    · pg_payment를 UNKNOWN으로 남기고 504 /errors/pg-timeout을 돌려준다
    · 상태는 orderId가 담긴 PG 콜백이나 GET /pg/v1/payments?orderId= 조회(재처리 워커)로 확정한다. 같은 Idempotency-Key로 승인을 다시 요청해도 PG는 처음 결과를 돌려준다
- [TC-D08-4] PG 호출이 실패하면 무조건 세 번 재시도하게 했다가 리뷰에서 지적을 받았습니다. 무엇을 재시도하고 무엇을 하지 않나요?
    · 연결 실패·5xx는 같은 멱등 키로 재시도하고, 응답 타임아웃은 PG가 처리했을 수 있어 재시도하지 않고 조회·콜백으로 결과를 확인한다. 400·401 같은 4xx는 같은 요청이 같은 실패를 내므로 재시도하지 않는다
    · 결제 같은 POST는 같은 멱등 키로 재시도할 때만 이중 승인이 생기지 않는다
    · 재시도 간격은 지수 백오프와 지터로 늘리고, 전체 시간이 상위 요청의 시간 예산을 넘지 않게 횟수를 제한한다
- [TC-D08-5] PG 콜백 경로는 인증 없이 공개했습니다. 그러면 아무나 "결제 완료"를 보내 포인트를 받을 수 있지 않나요?
    · 공개 경로라도 서버가 본문 원문 바이트를 같은 비밀키로 HMAC-SHA256해 X-PG-Signature와 상수 시간으로 비교하고, 다르면 401을 낸다
    · 서명은 파싱 전 원문으로 계산한다. JSON을 파싱해 다시 직렬화하면 공백이나 필드 순서가 바뀌어 바이트가 달라질 수 있다
    · 같은 paymentKey 콜백이 여러 번 와도 한 번만 적립하고, amount가 저장된 충전 금액과 다르면 거부한다

리뷰 초점
- PG 호출이 `@Transactional` 범위 밖에서 일어나는지(트랜잭션 로그로 확인)
- PG 비밀키가 코드·yml에 없고 `BOARD_PG_SECRET` 환경 변수로 들어오는지
- 결과 불명(`UNKNOWN`) 건을 실패로 단정해 포인트를 지급·환불하지 않는지

아래 형식으로만 답해라.
① 채점표 — | 항목 | 기대 | 실제 | 결과 |  (결과: 통과 / 실패 / 확인 불가)
② 잘한 점
③ 부족한 점과 이유 — 실패한 C 항목부터
④ 내일 전에 고칠 것 — 우선순위순, 3개 이내
⑤ 말하기 카드 — | 카드 | 빠진 줄 | 결과 |
마지막 줄: `결과: 필수 N/M 통과 · 말하기 카드 K/L 통과` (필수 = C 항목, R은 참고)
```
