package com.example.board.common;

import java.io.IOException;
import java.util.UUID;
import java.util.regex.Pattern;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import org.slf4j.MDC;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
public class RequestIdFilter extends OncePerRequestFilter {

    public static final String HEADER = "X-Request-Id";
    public static final String MDC_KEY = "requestId";

    // 클라이언트가 보낸 값이 로그 줄을 깨지 않도록 안전한 문자만 받는다
    private static final Pattern SAFE = Pattern.compile("^[A-Za-z0-9_-]{1,64}$");

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        String id = request.getHeader(HEADER);
        if (id == null || !SAFE.matcher(id).matches()) {
            id = UUID.randomUUID().toString();
        }
        MDC.put(MDC_KEY, id);
        response.setHeader(HEADER, id);
        try {
            chain.doFilter(request, response);
        } finally {
            // 톰캣은 스레드를 재사용하므로 지우지 않으면 다음 요청 로그에 이전 ID가 남는다
            MDC.remove(MDC_KEY);
        }
    }
}

#TC-D01-4

질문 동시 요청 세 건의 로그가 섞여서 한 사용자 요청만 골라낼 수 없었습니다. 요청 ID 필터를 어떻게 만들었나요?

  • 서블릿 필터가 요청 시작에서 요청 ID를 MDC.put하고, 로그 패턴에 그 키를 넣어 모든 줄에 찍는다

  • MDC는 스레드 단위라 @Async나 별도 스레드풀로 넘어가면 전파되지 않으므로 TaskDecorator로 복사한다

#AI 채점 프롬프트

오늘 순서를 다 만들었으면 서버를 띄운 채로 아래 프롬프트를 복사해 Codex 또는 Claude Code에 붙여 넣는다. 에이전트가 완료 조건·함정·말하기 카드로 채점표를 준다. 코드는 고치지 않는다.

[javaExperience 일자 채점] D01. 프로젝트 뼈대
레포 경로: 내 프로젝트 루트(이 프롬프트를 붙여 넣은 에이전트를 연 폴더)
서버 주소: http://localhost:8080

너는 Java·Spring 온보딩 과정 v2로 백엔드를 배우는 사람의 리뷰어다. 학습자는 빈 프로젝트에서 명세서 v2의 서버를 하루씩 만들고 있고, 오늘은 D01다.
지킬 것: 코드는 절대 수정하지 마라. DB는 조회만 한다(SELECT·SHOW·EXPLAIN). 서버를 대신 띄우거나 끄지 마라. 정답 코드를 통째로 주지 말고 방향과 힌트로 알려 줘라.

1. 서버 주소의 `GET /actuator/health`로 응답 가능 여부를 확인해라. 꺼져 있으면 `./gradlew bootRun`으로 띄우라고만 알려 주고 멈춰라.
2. 아래 "채점 기준"이 통과/실패의 유일한 기준이다. 기준에 없는 것으로 실패 처리하지 말고, 다른 날의 기준은 보지 마라.
3. [C] 항목: HTTP 요청은 서버 주소로 직접 보내 상태 코드·응답 헤더·본문을 기대값과 비교한다. 테스트 이름이 적힌 항목은 레포 루트에서 `./gradlew test --tests '<테스트 클래스 이름>'`으로 실행해 통과 여부를 본다(테스트 코드도 고치지 마라). 로그 항목은 기동 로그에서 찾거나 학습자에게 해당 줄을 붙여 달라고 한다. DB 확인이 필요하면 `docker compose exec -T mysql mysql -uboard -pboard board -e "<SELECT 문>"`으로 조회만 한다.
4. [R] 항목: 관련 코드를 찾아 읽고 그 함정을 피했는지 판단해라. 지적할 때는 `파일:줄`을 붙여라.
5. 말하기 카드: 학습자에게 카드 질문마다 답을 말하거나 적어 달라고 해라. "답에 들어가야 할 것"(· 표시) 세 줄이 모두 나오면 통과다. 빠진 줄은 정답 문장을 먼저 말하지 말고 되묻는 질문으로 끌어내라.
6. 리뷰 초점은 채점표에 넣지 말고 ②·③에 적어라.

채점 기준
① start.spring.io로 프로젝트 만들고 헬스 체크 열기 (Java 21, Boot 4.1, 의존성은 Web·Actuator·Validation·Lombok만)
  - [C1-1] ./gradlew bootRun으로 기동
  - [C1-2] GET /actuator/health → 200 {"status":"UP"}
  - [R1] 함정을 피했는지 코드로 확인: 헬스 체크(liveness)에 DB 확인을 넣으면 DB가 잠깐 끊길 때 모든 인스턴스가 재시작돼 장애가 커진다.
② Colima에서 MySQL 8.4를 띄우고 JPA 연결하기 (build.gradle에 JPA·MySQL 의존성 추가, 스키마는 src/main/resources/schema.sql로 관리하고 Flyway는 쓰지 않는다)
  - [C2-1] docker compose up -d 뒤 기동 로그에 HikariPool-1 - Start completed
  - [C2-2] GET /actuator/health → 200
  - [C2-3] application.yml에 ---로 나눈 test 프로파일 문서가 있다
  - [R2] 함정을 피했는지 코드로 확인: MySQL 같은 외부 DB에서는 spring.sql.init.mode=always를 켜지 않으면 schema.sql이 실행되지 않아 테이블 없이 기동된다.
③ 프로파일을 local·prod로 나누고 비밀값은 자리표시자로만 두기
  - [C3-1] --spring.profiles.active=local 기동 로그에 The following 1 profile is active: "local"
  - [C3-2] SERVER_PORT=8081로 실행하면 yml의 server.port보다 환경 변수가 이긴다: GET http://localhost:8081/actuator/health → 200
  - [C3-3] prod 프로파일은 DB_PASSWORD 환경 변수 없이 기동하면 실패한다
  - [C3-4] GET /actuator/env → 404 (HTTP로는 health만 노출)
  - [R3] 함정을 피했는지 코드로 확인: DB 비밀번호를 yml에 적어 커밋하면 파일에서 지워도 git 이력에 남으므로 yml에는 ${DB_PASSWORD} 자리표시자만 둔다.
④ logback 패턴에 요청 ID를 넣고 MDC 요청 ID 필터 만들기
  - [C4-1] GET /actuator/health → 200, 응답 헤더에 X-Request-Id
  - [C4-2] 같은 요청의 로그 줄마다 같은 요청 ID가 찍힌다
  - [C4-3] 동시에 보낸 3건의 로그를 요청 ID로 나눠 볼 수 있다
  - [R4] 함정을 피했는지 코드로 확인: MDC를 finally에서 지우지 않으면 톰캣이 재사용하는 스레드의 다음 요청 로그에 이전 요청 ID가 남는다.

말하기 카드
- [TC-D01-1] application.yml의 server.port를 바꿨는데 서버가 예전 포트로 뜹니다. 어디부터 확인하나요?
    · 커맨드라인 인자와 OS 환경 변수는 application.yml보다 우선하므로, 같은 키가 그쪽에 있으면 yml 수정은 반영되지 않는다
    · 활성 프로파일의 설정(application-local.yml 또는 on-profile 문서)이 기본 문서의 같은 키를 덮어쓴다
    · 값은 키 단위로 덮어써지므로 추측하지 않고, 로컬에서 연 /actuator/env의 출처 목록이나 기동 로그로 어느 출처가 이겼는지 확인한다
- [TC-D01-2] prod 서버가 로컬 DB 주소로 떠 있었습니다. 원인으로 무엇을 의심하나요?
    · 활성 프로파일을 지정하지 않으면 default 프로파일로 기동해 prod 문서가 적용되지 않는다
    · 프로파일 전용 문서(on-profile 문서, application-prod.yml) 안에서는 spring.profiles.active를 지정할 수 없으므로, 활성화는 실행 인자나 환경 변수로 한다
    · 기동 로그에 찍히는 활성 프로파일 줄로 어떤 프로파일이 켜졌는지 먼저 확인한다
- [TC-D01-3] DB 비밀번호를 application.yml에 적어 커밋했습니다. 무엇이 문제고 어떻게 바꾸나요?
    · 파일에서 지워도 git 이력에 평문이 남으므로, 이미 커밋된 비밀번호는 노출된 것으로 보고 교체한다
    · Spring Boot에는 내장 프로퍼티 암호화가 없어서 yml에는 ${DB_PASSWORD} 같은 자리표시자만 둔다
    · 실제 값은 환경 변수(또는 configtree:로 마운트한 파일)로 주입한다. PG 비밀키도 같은 방식이다
- [TC-D01-4] 동시 요청 세 건의 로그가 섞여서 한 사용자 요청만 골라낼 수 없었습니다. 요청 ID 필터를 어떻게 만들었나요?
    · 서블릿 필터가 요청 시작에서 요청 ID를 MDC.put하고, 로그 패턴에 그 키를 넣어 모든 줄에 찍는다
    · finally에서 MDC를 지운다. 톰캣 스레드는 재사용되므로 지우지 않으면 다음 요청 로그에 이전 ID가 남을 수 있다
    · MDC는 스레드 단위라 @Async나 별도 스레드풀로 넘어가면 전파되지 않으므로 TaskDecorator로 복사한다
- [TC-D01-5] 헬스 체크(liveness)에 DB 연결 확인을 넣자는 제안을 받았습니다. 어떻게 답하나요?
    · liveness에 DB 같은 외부 시스템을 넣으면 DB가 잠깐 끊겼을 때 모든 인스턴스가 재시작돼 장애가 커진다
    · 외부 의존 상태는 readiness에서 다룰지 판단한다. 넣으면 그 시스템 장애 때 인스턴스가 트래픽에서 빠진다
    · Boot는 기본으로 HTTP에 health만 노출하고, 명세도 /actuator/health 외의 actuator는 공개하지 않는다

리뷰 초점
- `application.yml`·`application-prod.yml`과 git 이력에 실제 비밀번호 문자열이 없는지
- 패키지가 `common`·`config`처럼 역할별 도메인형으로 시작하는지(`controller`·`service` 계층형 패키지로 시작하지 않았는지)
