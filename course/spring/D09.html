package com.example.board.post;

import java.time.Duration;
import java.time.ZoneId;
import java.util.List;
import java.util.concurrent.locks.ReentrantLock;

import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Service;

import com.example.board.common.CursorPage;

import tools.jackson.core.type.TypeReference;
import tools.jackson.databind.json.JsonMapper;

/** 좋아요 내림차순 상위 20개를 Redis에 60초 캐시한다. DB 조회는 트랜잭션 없이 하고, 트랜잭션 안에서 Redis를 부르지 않는다. */
@Service
public class PopularPostService {

    static final String KEY = "popular:posts";
    static final Duration TTL = Duration.ofSeconds(60);
    static final int LIMIT = 20;
    private static final ZoneId KST = ZoneId.of("Asia/Seoul");
    private static final TypeReference<List<PopularPost>> LIST = new TypeReference<>() {
    };

    private final PostQueryMapper mapper;
    private final StringRedisTemplate redis;
    private final JsonMapper json;

    // 스탬피드 방지(single-flight). 캐시가 빈 순간 서버 한 대 안에서는 한 요청만 DB를 읽고 나머지는 그 결과를 기다린다.
    // Redis SET NX 락 대신 JVM 락을 쓴 이유: DB 조회가 많아도 서버 대수만큼이라 충분하고, Redis 장애 때 락까지 함께 잃지 않는다.
    private final ReentrantLock loadLock = new ReentrantLock();

    public PopularPostService(PostQueryMapper mapper, StringRedisTemplate redis, JsonMapper json) {
        this.mapper = mapper;
        this.redis = redis;
        this.json = json;
    }

    public List<PopularPost> getPopular() {
        List<PopularPost> cached = readCache();
        if (cached != null) {
            return cached;
        }
        loadLock.lock();
        try {
            // 락을 기다리는 동안 앞 요청이 채웠을 수 있다. 다시 보지 않으면 줄 선 요청이 모두 DB를 읽는다.
            cached = readCache();
            if (cached != null) {
                return cached;
            }
            List<PopularPost> loaded = mapper.findPopular(LIMIT).stream().map(PopularPost::from).toList();
            redis.opsForValue().set(KEY, json.writeValueAsString(loaded), TTL);
            return loaded;
        } finally {
            loadLock.unlock();
        }
    }

    /** 상위 20개 고정 목록이라 한 쪽뿐이다. size만큼 앞에서 자르고 nextCursor는 항상 null이다. */
    public CursorPage<PopularPost> page(String cursor, int size) {
        // 커서 형식만 검사한다(서버가 준 적 없는 값은 400). 다음 쪽이 없으니 위치로는 쓰지 않는다.
        CursorPage.decode(cursor);
        List<PopularPost> all = getPopular();
        return new CursorPage<>(all.subList(0, Math.min(size, all.size())), null);
    }

    private List<PopularPost> readCache() {
        String value = redis.opsForValue().get(KEY);
        if (value == null) {
            return null;
        }
        // JSON에서 되읽은 시각은 UTC로 맞춰지므로 DB에서 바로 읽은 응답과 같은 KST 오프셋으로 되돌린다
        return json.readValue(value, LIST).stream()
                .map(p -> new PopularPost(p.id(), p.authorId(), p.authorNickname(), p.title(), p.tags(),
                        p.likeCount(), p.createdAt().atZoneSameInstant(KST).toOffsetDateTime()))
                .toList();
    }
}

package com.example.board;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;

import org.apache.ibatis.executor.Executor;
import org.apache.ibatis.mapping.MappedStatement;
import org.apache.ibatis.plugin.Interceptor;
import org.apache.ibatis.plugin.Intercepts;
import org.apache.ibatis.plugin.Invocation;
import org.apache.ibatis.plugin.Signature;
import org.apache.ibatis.session.ResultHandler;
import org.apache.ibatis.session.RowBounds;
import org.apache.ibatis.session.SqlSessionFactory;

/**
 * MyBatis 조회 횟수를 매퍼 메서드별로 센다. Hibernate 통계에는 MyBatis 쿼리가 잡히지 않아서 따로 둔다.
 * 빈으로 등록하면 컨텍스트 캐시 키가 바뀌어 MySQL이 하나 더 뜨므로, 이미 뜬 SqlSessionFactory에 붙인다.
 */
@Intercepts(@Signature(type = Executor.class, method = "query",
        args = {MappedStatement.class, Object.class, RowBounds.class, ResultHandler.class}))
public final class MyBatisQueryCounter implements Interceptor {

    private final Map<String, AtomicInteger> counts = new ConcurrentHashMap<>();

    /** 컨텍스트마다 한 번만 붙인다. 새로 여는 SqlSession부터 적용된다. */
    public static synchronized MyBatisQueryCounter attach(SqlSessionFactory factory) {
        for (Interceptor i : factory.getConfiguration().getInterceptors()) {
            if (i instanceof MyBatisQueryCounter counter) {
                return counter;
            }
        }
        MyBatisQueryCounter counter = new MyBatisQueryCounter();
        factory.getConfiguration().addInterceptor(counter);
        return counter;
    }

    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        MappedStatement ms = (MappedStatement) invocation.getArgs()[0];
        counts.computeIfAbsent(ms.getId(), k -> new AtomicInteger()).incrementAndGet();
        return invocation.proceed();
    }

    /** 매퍼 메서드 이름(예: findPopular)으로 끝나는 statement의 실행 횟수. */
    public int count(String methodName) {
        return counts.entrySet().stream().filter(e -> e.getKey().endsWith("." + methodName))
                .mapToInt(e -> e.getValue().get()).sum();
    }

    public void clear() {
        counts.clear();
    }
}

 * Redis 호출을 감싸 실패를 호출자에게 올리지 않고 폴백 값으로 바꾼다(경고 로그만 남긴다).
 * 한 번 실패하면 잠시 Redis를 부르지 않는다. 지연 장애 때 명령마다 타임아웃(500ms)을 기다리면
 * 글 작성처럼 Redis를 여러 번 부르는 요청이 1초를 넘기고, 끊김 장애는 REJECT_COMMANDS로 이미 바로 실패한다.
 */
@Component
public class RedisGuard {

    private static final Logger log = LoggerFactory.getLogger(RedisGuard.class);
    static final Duration SKIP_AFTER_FAILURE = Duration.ofSeconds(5);

    // System.nanoTime 기준. 이 시각 전에는 Redis를 건너뛴다.
    private volatile long skipUntil = System.nanoTime();

    public <T> T call(String operation, Supplier<T> action, Supplier<T> fallback) {
        if (skipUntil - System.nanoTime() > 0) {
            return fallback.get();
        }
        try {
            return action.get();
        } catch (DataAccessException e) {
            skipUntil = System.nanoTime() + SKIP_AFTER_FAILURE.toNanos();
            log.warn("Redis {} 실패, {}초 동안 Redis 없이 처리한다: {}", operation, SKIP_AFTER_FAILURE.toSeconds(),
                    e.toString());
            return fallback.get();
        }
    }

    public void run(String operation, Runnable action) {
        call(operation, () -> {
            action.run();
            return null;
        }, () -> null);
    }

    /** 건너뛰기를 끝낸다. 장애 주입 테스트가 다음 테스트에 상태를 남기지 않게 할 때 쓴다. */
    public void reset() {
        skipUntil = System.nanoTime();
    }
}

import java.util.List;

import org.springframework.data.redis.core.StringRedisTemplate;

import org.springframework.data.redis.core.script.RedisScript;
import org.springframework.stereotype.Component;

import com.example.board.common.BusinessException;
import com.example.board.common.ErrorCode;
import com.example.board.config.RedisGuard;

/**
 * 로그인 실패 횟수 제한(계정당 10분 5회, spec-v2 §8). 카운터는 Redis 키 login:fail:{email}에 둔다.
 * 서버 메모리에 두면 인스턴스마다 따로 세어 서버 수만큼 더 시도할 수 있다.
 * Redis 장애면 제한 없이 통과시킨다(fail-open). 막으면 Redis 장애가 모든 회원의 로그인 장애가 된다.
 */
@Component
public class LoginAttemptLimiter {

    static final int MAX_FAILURES = 5;
    static final Duration WINDOW = Duration.ofMinutes(10);

    // 첫 실패에만 만료를 건다(첫 실패부터 10분). INCR과 EXPIRE 사이에 죽어 만료 없는 키가 남지 않게 한 번에 실행한다.
    private static final RedisScript<Long> INCREMENT = new DefaultRedisScript<>("""
            local n = redis.call('INCR', KEYS[1])
            if n == 1 then redis.call('EXPIRE', KEYS[1], ARGV[1]) end
            return n
            """, Long.class);

    private final StringRedisTemplate redis;
    private final RedisGuard guard;

    public LoginAttemptLimiter(StringRedisTemplate redis, RedisGuard guard) {
        this.redis = redis;
        this.guard = guard;
    }

    /** 이미 5번 실패했으면 비밀번호를 확인하지 않고 429. */
    public void check(String email) {
        String count = guard.call("로그인 실패 횟수 조회", () -> redis.opsForValue().get(key(email)), () -> null);
        if (count != null && Integer.parseInt(count) >= MAX_FAILURES) {
            throw new BusinessException(ErrorCode.TOO_MANY_ATTEMPTS, "로그인 시도 횟수를 넘었습니다. 잠시 뒤 다시 시도하세요.");
        }
    }

    public void recordFailure(String email) {
        guard.run("로그인 실패 기록",
                () -> redis.execute(INCREMENT, List.of(key(email)), String.valueOf(WINDOW.toSeconds())));
    }

    public void reset(String email) {
        guard.run("로그인 실패 초기화", () -> redis.delete(key(email)));
    }

    // DB 비교(utf8mb4_0900_ai_ci)는 대소문자를 가리지 않는다. 키도 소문자로 맞춰 대소문자만 바꾼 우회를 막는다.
    private static String key(String email) {
        return "login:fail:" + email.toLowerCase(Locale.ROOT);
    }
}

  - [C3-3] 트랜잭션 경계: DB 변경과 캐시 무효화가 만나는 시점과 그 이유를 적는다. 실패 시 DB rollback 뒤 캐시를 유지하는지 로그·조회 결과로 설명한다.
  - [R3] 함정을 피했는지 코드로 확인: 커밋 전에 캐시를 지우면 그 사이 다른 요청이 옛 값을 다시 적재해 TTL 동안 남는다.
④ 선착순 쿠폰 잔여 수량을 Redis 원자적 감소로 거르기 (기준 데이터는 DB)
  - [C4-1] CouponIssueConcurrencyTest 통과: 수량 10장에 100명 동시 요청 → 정확히 10장, DB 발급량 = 10

말하기 카드
- [TC-D09-1] 인기글 캐시(TTL 60초)가 만료되는 순간마다 DB 쿼리가 몰립니다. 왜 그렇고 어떻게 막나요?
    · 키가 만료된 순간 들어온 요청이 모두 캐시 미스를 보고 동시에 DB를 조회한다(스탬피드)
    · 한 요청만 다시 계산하고 나머지는 그 결과를 기다리게 한다(single-flight). 방법은 인스턴스 안의 JVM 락(ReentrantLock, @Cacheable(sync = true))이나 분산 락(SET NX), 만료 전 선갱신이 있다
    · 무엇을 골랐고 왜인지 말한다. JVM 락이면 서버 대수만큼 DB 조회가 남는 한계가 있고(정답은 이 정도면 충분하고 Redis 장애 때 락까지 잃지 않아 JVM 락을 골랐다), 분산 락이면 Redis 장애 때 락도 함께 잃는다
- [TC-D09-2] Redis가 느려지거나 죽으면 인기글 조회가 어떻게 동작하도록 만들었나요?
    · Redis 명령 타임아웃을 500ms로 둬서 Redis 지연이 요청 스레드를 오래 잡지 않게 한다
    · 캐시 조회 실패는 경고 로그를 남기고 DB 조회로 폴백한다
    · 캐시 호출을 직접 감쌌다면 예외를 잡아 폴백한다(정답 RedisGuard). Spring Cache를 쓴다면 CacheErrorHandler를 등록해 조회 오류를 캐시 미스로 처리한다(기본 처리기는 예외를 그대로 던진다)
- [TC-D09-3] 선착순 쿠폰 잔여 수량을 Redis DECR로 관리합니다. Redis 값과 DB의 issued_quantity가 다르면 무엇을 믿나요?
    · 남은 수량 확인과 감소를 원자적으로 묶어(Lua 스크립트 등) 0 아래로 내려가지 않게 한다. DECR 뒤 음수면 INCR로 되돌리는 방식은 초과 요청과 DB 실패 되돌리기가 겹치면 수량이 어긋날 수 있다
    · 발급의 최종 기준은 DB(member_coupon UNIQUE, issued_quantity <= total_quantity)이고, Redis는 DB 앞에서 요청을 거르는 장치다
    · Redis에서 차감한 뒤 DB 저장이 실패하면 Redis 수량을 되돌리거나 DB 기준으로 다시 맞추는 절차가 있어야 한다
- [TC-D09-4] 관리자가 글을 숨기면 인기글 캐시를 지웁니다. 지우는 시점을 트랜잭션 커밋 뒤로 둔 이유는?
    · 커밋 전에 지우면 그 사이 들어온 조회가 아직 커밋되지 않은, 숨기기 전 값을 DB에서 읽어 캐시에 다시 넣는다
    · 그러면 TTL(60초)이 끝날 때까지 숨긴 글이 인기글에 보인다
    · 그래서 @TransactionalEventListener(phase = AFTER_COMMIT)이나 TransactionSynchronization의 afterCommit에서 지운다
- [TC-D09-5] 로그인 시도 제한(10분에 5회)을 서버 메모리가 아니라 Redis에 둔 이유와 구현은?
    · 메모리 카운터는 인스턴스마다 따로 세므로 인스턴스가 여럿이면 제한이 인스턴스 수만큼 느슨해진다
    · 계정별 키를 INCR하고 첫 증가 때 600초 만료를 건다. 두 명령 사이에 실패하면 만료 없는 키가 남아 계정이 계속 막힐 수 있으므로 Lua 스크립트로 묶는다
    · 5회를 넘으면 429 /errors/too-many-attempts로 응답한다

리뷰 초점
- Redis가 멈춰도 글쓰기·상세 같은 핵심 경로가 200·201로 동작하는지
- Redis에만 있는 데이터(랭킹·카운터)를 DB에서 다시 만들 수 있는지
