/** 좋아요 내림차순 상위 20개를 Redis에 60초 캐시한다. DB 조회는 트랜잭션 없이 하고, 트랜잭션 안에서 Redis를 부르지 않는다. */
@Service
public class PopularPostService {

    static final String KEY = "popular:posts";
    static final Duration TTL = Duration.ofSeconds(60);
    static final int LIMIT = 20;
    private static final ZoneId KST = ZoneId.of("Asia/Seoul");
    private static final TypeReference<List<PopularPost>> LIST = new TypeReference<>() {
    };

    private final PostQueryMapper mapper;
    private final StringRedisTemplate redis;
    private final JsonMapper json;

    // 스탬피드 방지(single-flight). 캐시가 빈 순간 서버 한 대 안에서는 한 요청만 DB를 읽고 나머지는 그 결과를 기다린다.
    // Redis SET NX 락 대신 JVM 락을 쓴 이유: DB 조회가 많아도 서버 대수만큼이라 충분하고, Redis 장애 때 락까지 함께 잃지 않는다.
    private final ReentrantLock loadLock = new ReentrantLock();

    public PopularPostService(PostQueryMapper mapper, StringRedisTemplate redis, JsonMapper json) {
        this.mapper = mapper;
        this.redis = redis;
        this.json = json;
    }

    public List<PopularPost> getPopular() {
        List<PopularPost> cached = readCache();
        if (cached != null) {
            return cached;
        }
        loadLock.lock();
        try {
            // 락을 기다리는 동안 앞 요청이 채웠을 수 있다. 다시 보지 않으면 줄 선 요청이 모두 DB를 읽는다.
            cached = readCache();
            if (cached != null) {
                return cached;
            }
            List<PopularPost> loaded = mapper.findPopular(LIMIT).stream().map(PopularPost::from).toList();
            redis.opsForValue().set(KEY, json.writeValueAsString(loaded), TTL);
            return loaded;
        } finally {
            loadLock.unlock();
        }
    }

    /** 상위 20개 고정 목록이라 한 쪽뿐이다. size만큼 앞에서 자르고 nextCursor는 항상 null이다. */
    public CursorPage<PopularPost> page(String cursor, int size) {
        // 커서 형식만 검사한다(서버가 준 적 없는 값은 400). 다음 쪽이 없으니 위치로는 쓰지 않는다.
        CursorPage.decode(cursor);
        List<PopularPost> all = getPopular();
        return new CursorPage<>(all.subList(0, Math.min(size, all.size())), null);
    }

    private List<PopularPost> readCache() {
        String value = redis.opsForValue().get(KEY);
        if (value == null) {
            return null;
        }
        // JSON에서 되읽은 시각은 UTC로 맞춰지므로 DB에서 바로 읽은 응답과 같은 KST 오프셋으로 되돌린다
        return json.readValue(value, LIST).stream()
                .map(p -> new PopularPost(p.id(), p.authorId(), p.authorNickname(), p.title(), p.tags(),
                        p.likeCount(), p.createdAt().atZoneSameInstant(KST).toOffsetDateTime()))
                .toList();
    }
}

package com.example.board;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;

import org.apache.ibatis.executor.Executor;
import org.apache.ibatis.mapping.MappedStatement;
import org.apache.ibatis.plugin.Interceptor;
import org.apache.ibatis.plugin.Intercepts;
import org.apache.ibatis.plugin.Invocation;
import org.apache.ibatis.plugin.Signature;
import org.apache.ibatis.session.ResultHandler;
import org.apache.ibatis.session.RowBounds;
import org.apache.ibatis.session.SqlSessionFactory;

/**
 * MyBatis 조회 횟수를 매퍼 메서드별로 센다. Hibernate 통계에는 MyBatis 쿼리가 잡히지 않아서 따로 둔다.
 * 빈으로 등록하면 컨텍스트 캐시 키가 바뀌어 MySQL이 하나 더 뜨므로, 이미 뜬 SqlSessionFactory에 붙인다.
 */
@Intercepts(@Signature(type = Executor.class, method = "query",
        args = {MappedStatement.class, Object.class, RowBounds.class, ResultHandler.class}))
public final class MyBatisQueryCounter implements Interceptor {

    private final Map<String, AtomicInteger> counts = new ConcurrentHashMap<>();

    /** 컨텍스트마다 한 번만 붙인다. 새로 여는 SqlSession부터 적용된다. */
    public static synchronized MyBatisQueryCounter attach(SqlSessionFactory factory) {
        for (Interceptor i : factory.getConfiguration().getInterceptors()) {
            if (i instanceof MyBatisQueryCounter counter) {
                return counter;
            }
        }
        MyBatisQueryCounter counter = new MyBatisQueryCounter();
        factory.getConfiguration().addInterceptor(counter);
        return counter;
    }

    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        MappedStatement ms = (MappedStatement) invocation.getArgs()[0];
        counts.computeIfAbsent(ms.getId(), k -> new AtomicInteger()).incrementAndGet();
        return invocation.proceed();
    }

    /** 매퍼 메서드 이름(예: findPopular)으로 끝나는 statement의 실행 횟수. */
    public int count(String methodName) {
        return counts.entrySet().stream().filter(e -> e.getKey().endsWith("." + methodName))
                .mapToInt(e -> e.getValue().get()).sum();
    }

    public void clear() {
        counts.clear();
    }
}

    private static final Network NETWORK = Network.newNetwork();
    private static final GenericContainer<?> REDIS = new GenericContainer<>("redis:7.4")
            .withNetwork(NETWORK)
            .withNetworkAliases("redis")
            .withExposedPorts(6379);

            .withNetwork(NETWORK);

    private static Proxy proxy;

    private RedisContainers() {
    }

    public static synchronized void start() {
        if (proxy != null) {
            return;
        }
        REDIS.start();
        TOXIPROXY.start();
        toxiproxyClient = new ToxiproxyClient(TOXIPROXY.getHost(), TOXIPROXY.getControlPort());
        createProxy();
    }

    private static void createProxy() {
        try {
            proxy = toxiproxyClient.createProxy("redis", "0.0.0.0:" + PROXY_PORT, "redis:6379");
        } catch (IOException e) {
            throw new IllegalStateException(e);
        }
    }

        start();
        return REDIS.getHost();
    }

        start();
        return REDIS.getMappedPort(6379);
    }

    }

            List<Integer> result = new ArrayList<>();

            }
            return result;
        } finally {

        }

package com.example.board.config;

import java.time.Duration;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataAccessException;
import org.springframework.stereotype.Component;

/**
 * Redis 호출을 감싸 실패를 호출자에게 올리지 않고 폴백 값으로 바꾼다(경고 로그만 남긴다).
 * 한 번 실패하면 잠시 Redis를 부르지 않는다. 지연 장애 때 명령마다 타임아웃(500ms)을 기다리면
 * 글 작성처럼 Redis를 여러 번 부르는 요청이 1초를 넘기고, 끊김 장애는 REJECT_COMMANDS로 이미 바로 실패한다.
 */
@Component
public class RedisGuard {

    private static final Logger log = LoggerFactory.getLogger(RedisGuard.class);
    static final Duration SKIP_AFTER_FAILURE = Duration.ofSeconds(5);

    // System.nanoTime 기준. 이 시각 전에는 Redis를 건너뛴다.
    private volatile long skipUntil = System.nanoTime();

    public <T> T call(String operation, Supplier<T> action, Supplier<T> fallback) {
        if (skipUntil - System.nanoTime() > 0) {
            return fallback.get();
        }
        try {
            return action.get();
        } catch (DataAccessException e) {
            skipUntil = System.nanoTime() + SKIP_AFTER_FAILURE.toNanos();
            log.warn("Redis {} 실패, {}초 동안 Redis 없이 처리한다: {}", operation, SKIP_AFTER_FAILURE.toSeconds(),
                    e.toString());
            return fallback.get();
        }
    }

    public void run(String operation, Runnable action) {
        call(operation, () -> {
            action.run();
            return null;
        }, () -> null);
    }

    /** 건너뛰기를 끝낸다. 장애 주입 테스트가 다음 테스트에 상태를 남기지 않게 할 때 쓴다. */
    public void reset() {
        skipUntil = System.nanoTime();
    }
}

package com.example.board.coupon;

import java.util.List;

import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.DefaultRedisScript;
import org.springframework.data.redis.core.script.RedisScript;
import org.springframework.stereotype.Component;

import com.example.board.config.RedisGuard;

/**
 * 선착순 쿠폰 잔여 수량의 앞단 거름망(Redis 키 coupon:{couponId}:stock). 기준 데이터는 DB의 issued_quantity다.
 * 수량이 0이면 DB까지 가지 않고 거절해 몰린 요청이 DB 커넥션을 잡지 않게 한다.
 * Redis 장애면 모두 통과시켜 DB 조건부 UPDATE만으로 판단한다.
 */
@Component
public class CouponStockCounter {

    private static final long DECREASED = 1;
    private static final long MISSING = -1;

    // 읽고 확인한 뒤 줄이면 그 사이 다른 요청이 끼므로 스크립트 하나로 원자 실행한다.
    // 키가 없으면 ARGV[1](DB에서 읽은 남은 수량)로 채운다. 0 아래로는 내려가지 않아 되돌리기(INCR)가 자리를 잃지 않는다.
    private static final RedisScript<Long> DECREASE = new DefaultRedisScript<>("""
            local v = redis.call('GET', KEYS[1])
            if not v then
              if not ARGV[1] then return -1 end
              redis.call('SET', KEYS[1], ARGV[1])
              v = ARGV[1]
            end
            if tonumber(v) <= 0 then return 0 end
            redis.call('DECR', KEYS[1])
            return 1
            """, Long.class);

    // 키가 없을 때 INCR하면 1로 새로 생겨 DB와 어긋난다. 있을 때만 되돌린다.
    private static final RedisScript<Long> RESTORE = new DefaultRedisScript<>("""
            if redis.call('EXISTS', KEYS[1]) == 1 then return redis.call('INCR', KEYS[1]) end
            return 0
            """, Long.class);

    private final StringRedisTemplate redis;
    private final CouponRepository couponRepository;
    private final RedisGuard guard;

    public CouponStockCounter(StringRedisTemplate redis, CouponRepository couponRepository, RedisGuard guard) {
        this.redis = redis;
        this.couponRepository = couponRepository;
        this.guard = guard;
    }

    /** 한 장 줄였거나 Redis로 판단할 수 없으면 true(DB로 진행), 남은 수량이 0이면 false. */
    public boolean tryDecrease(Long couponId) {
        List<String> keys = List.of(key(couponId));
        Long result = guard.call("쿠폰 수량 감소", () -> redis.execute(DECREASE, keys), () -> null);
        if (result != null && result == MISSING) {
            // 첫 요청: DB의 남은 수량으로 채운다. 없는 쿠폰이면 DB 경로가 404로 답하게 둔다.
            Coupon coupon = couponRepository.findById(couponId).orElse(null);
            if (coupon == null) {
                return true;
            }
            String remaining = String.valueOf(coupon.getTotalQuantity() - coupon.getIssuedQuantity());
            result = guard.call("쿠폰 수량 감소", () -> redis.execute(DECREASE, keys, remaining), () -> null);
        }
        return result == null || result == DECREASED;
    }

    /** tryDecrease를 통과했는데 DB 발급이 실패(중복 등)했으면 한 장 되돌린다. 안 그러면 실제보다 적게 나간다. */
    public void restore(Long couponId) {
        guard.run("쿠폰 수량 되돌리기", () -> redis.execute(RESTORE, List.of(key(couponId))));
    }

    /** DB가 소진이라고 답했으면 DB를 기준으로 Redis도 0으로 맞춘다. */
    public void markExhausted(Long couponId) {
        guard.run("쿠폰 수량 맞추기", () -> redis.opsForValue().set(key(couponId), "0"));
    }

    private static String key(Long couponId) {
        return "coupon:" + couponId + ":stock";
    }
}

    private static final String REBUILD_KEY = KEY + ":rebuild";
    // 랭킹에 들어가는 적립 종류. 원장 합산(rebuild)과 증분(add)이 같은 기준을 쓴다.
    static final Set<String> SCORED_REF_TYPES = Set.of("POST", "COMMENT");

    private final StringRedisTemplate redis;
    private final PointLedgerRepository ledgerRepository;
    private final MemberRepository memberRepository;
    private final RedisGuard guard;

    public ActivityRankingService(StringRedisTemplate redis, PointLedgerRepository ledgerRepository,
            MemberRepository memberRepository, RedisGuard guard) {
        this.redis = redis;
        this.ledgerRepository = ledgerRepository;
        this.memberRepository = memberRepository;
        this.guard = guard;
    }

    public void add(Long memberId, int score) {
        guard.run("랭킹 가산", () -> redis.opsForZSet().incrementScore(KEY, memberId.toString(), score));
    }

    /** 원장 기록이 커밋된 뒤에만 올린다. 롤백된 적립이 점수로 남지 않고, 트랜잭션 안에서 Redis를 부르지 않는다. */
    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    public void onPointEarned(PointEarnedEvent event) {
        if (SCORED_REF_TYPES.contains(event.refType())) {
            add(event.memberId(), event.amount());
        }
    }

    /** 커서는 다음 쪽이 시작하는 순위 위치(0부터)다. */
    public CursorPage<ActivityRank> top(String cursor, int size) {
        Long after = CursorPage.decode(cursor);
        long offset = after == null ? 0 : after;
        // 한 건 더 읽어 다음 쪽이 있는지만 판단한다
        List<MemberScore> rows = guard.call("랭킹 조회", () -> fromRedis(offset, size + 1),
                () -> fromLedger(offset, size + 1));
        boolean hasNext = rows.size() > size;
        List<MemberScore> page = hasNext ? rows.subList(0, size) : rows;
        Map<Long, String> nicknames = memberRepository.findAllById(page.stream().map(MemberScore::memberId).toList())
                .stream().collect(Collectors.toMap(Member::getId, Member::getNickname));
        List<ActivityRank> items = new ArrayList<>();
        for (int i = 0; i < page.size(); i++) {
            MemberScore s = page.get(i);
            items.add(new ActivityRank(offset + i + 1, s.memberId(), nicknames.get(s.memberId()), s.score()));
        }
        return new CursorPage<>(items, hasNext ? CursorPage.encode(offset + size) : null);
    }

    /**
     * Redis 랭킹을 원장에서 다시 만든다(Redis 데이터 유실·불일치 복구용). 임시 키에 채운 뒤 RENAME으로 바꿔 끼워
     * 만드는 동안에도 옛 랭킹이 보인다. 다시 만든 회원 수를 돌려준다.
     */
    public int rebuild() {
        List<MemberScore> scores = ledgerRepository.sumActivityScores(SCORED_REF_TYPES, Pageable.unpaged());
        redis.delete(REBUILD_KEY);
        if (scores.isEmpty()) {
            redis.delete(KEY);
            return 0;
        }
        Set<TypedTuple<String>> tuples = scores.stream()
                .map(s -> TypedTuple.of(s.memberId().toString(), s.score().doubleValue()))
                .collect(Collectors.toSet());
        redis.opsForZSet().add(REBUILD_KEY, tuples);
        redis.rename(REBUILD_KEY, KEY);
        return scores.size();
    }

    private List<MemberScore> fromRedis(long offset, int count) {
        Set<TypedTuple<String>> tuples = redis.opsForZSet().reverseRangeWithScores(KEY, offset, offset + count - 1);
        return tuples == null ? List.of() : tuples.stream()
                .map(t -> new MemberScore(Long.valueOf(t.getValue()), t.getScore().longValue()))
                .toList();
    }

    // Redis 장애 때만 쓴다. 앞 순위까지 읽고 건너뛰므로 깊은 쪽일수록 느리다.
    private List<MemberScore> fromLedger(long offset, int count) {
        return ledgerRepository.sumActivityScores(SCORED_REF_TYPES, PageRequest.of(0, (int) (offset + count)))
                .stream().skip(offset).toList();
    }
}

/**
 * 로그인 실패 횟수 제한(계정당 10분 5회, spec-v2 §8). 카운터는 Redis 키 login:fail:{email}에 둔다.
 * 서버 메모리에 두면 인스턴스마다 따로 세어 서버 수만큼 더 시도할 수 있다.
 * Redis 장애면 제한 없이 통과시킨다(fail-open). 막으면 Redis 장애가 모든 회원의 로그인 장애가 된다.
 */
@Component
public class LoginAttemptLimiter {

    static final int MAX_FAILURES = 5;
    static final Duration WINDOW = Duration.ofMinutes(10);

    // 첫 실패에만 만료를 건다(첫 실패부터 10분). INCR과 EXPIRE 사이에 죽어 만료 없는 키가 남지 않게 한 번에 실행한다.
    private static final RedisScript<Long> INCREMENT = new DefaultRedisScript<>("""
            local n = redis.call('INCR', KEYS[1])
            if n == 1 then redis.call('EXPIRE', KEYS[1], ARGV[1]) end
            return n
            """, Long.class);

    private final StringRedisTemplate redis;
    private final RedisGuard guard;

    public LoginAttemptLimiter(StringRedisTemplate redis, RedisGuard guard) {
        this.redis = redis;
        this.guard = guard;
    }

    /** 이미 5번 실패했으면 비밀번호를 확인하지 않고 429. */
    public void check(String email) {
        String count = guard.call("로그인 실패 횟수 조회", () -> redis.opsForValue().get(key(email)), () -> null);
        if (count != null && Integer.parseInt(count) >= MAX_FAILURES) {
            throw new BusinessException(ErrorCode.TOO_MANY_ATTEMPTS, "로그인 시도 횟수를 넘었습니다. 잠시 뒤 다시 시도하세요.");
        }
    }

    public void recordFailure(String email) {
        guard.run("로그인 실패 기록",
                () -> redis.execute(INCREMENT, List.of(key(email)), String.valueOf(WINDOW.toSeconds())));
    }

    public void reset(String email) {
        guard.run("로그인 실패 초기화", () -> redis.delete(key(email)));
    }

    // DB 비교(utf8mb4_0900_ai_ci)는 대소문자를 가리지 않는다. 키도 소문자로 맞춰 대소문자만 바꾼 우회를 막는다.
    private static String key(String email) {
        return "login:fail:" + email.toLowerCase(Locale.ROOT);
    }
}

  • 캐시 조회 실패는 경고 로그를 남기고 DB 조회로 폴백한다
  • Spring Cache를 쓰면 CacheErrorHandler를 등록해 조회 오류를 캐시 미스로 처리한다. 기본 처리기는 예외를 그대로 던진다

#TC-D09-3

질문 선착순 쿠폰 잔여 수량을 Redis DECR로 관리합니다. Redis 값과 DB의 issued_quantity가 다르면 무엇을 믿나요?

답에 들어가야 할 것
  • DECR은 원자적이라 동시 요청에서도 정확히 하나씩 줄고, 결과가 0보다 작으면 소진으로 보고 INCR로 되돌린다
  • 발급의 최종 기준은 DB(member_coupon UNIQUE, issued_quantity <= total_quantity)이고, Redis는 DB 앞에서 요청을 거르는 장치다
  • Redis에서 차감한 뒤 DB 저장이 실패하면 Redis 수량을 되돌리거나 DB 기준으로 다시 맞추는 절차가 있어야 한다

#TC-D09-4

질문 관리자가 글을 숨기면 인기글 캐시를 지웁니다. 지우는 시점을 트랜잭션 커밋 뒤로 둔 이유는?

답에 들어가야 할 것
  • 커밋 전에 지우면 그 사이 들어온 조회가 아직 커밋되지 않은, 숨기기 전 값을 DB에서 읽어 캐시에 다시 넣는다
  • 그러면 TTL(60초)이 끝날 때까지 숨긴 글이 인기글에 보인다
  • 그래서 @TransactionalEventListener(phase = AFTER_COMMIT)이나 TransactionSynchronization의 afterCommit에서 지운다

- [TC-D09-1] 인기글 캐시(TTL 60초)가 만료되는 순간마다 DB 쿼리가 몰립니다. 왜 그렇고 어떻게 막나요?
    · 키가 만료된 순간 들어온 요청이 모두 캐시 미스를 보고 동시에 DB를 조회한다(스탬피드)
    · 짧은 TTL의 락 키(SET NX)를 잡은 요청 하나만 다시 계산하게 하거나, 만료 전에 미리 갱신한다
    · @Cacheable(sync = true)는 기본적으로 한 인스턴스(JVM) 안의 요청만 합치므로, 인스턴스가 여럿이면 이것만으로는 막지 못한다
- [TC-D09-2] Redis가 느려지거나 죽으면 인기글 조회가 어떻게 동작하도록 만들었나요?

    · 캐시 조회 실패는 경고 로그를 남기고 DB 조회로 폴백한다
    · Spring Cache를 쓰면 CacheErrorHandler를 등록해 조회 오류를 캐시 미스로 처리한다. 기본 처리기는 예외를 그대로 던진다

- [TC-D09-4] 관리자가 글을 숨기면 인기글 캐시를 지웁니다. 지우는 시점을 트랜잭션 커밋 뒤로 둔 이유는?
