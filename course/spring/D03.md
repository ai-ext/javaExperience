```java
package com.example.board.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
public class SecurityConfig {

    @Bean
    PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    SecurityFilterChain filterChain(HttpSecurity http, ProblemDetailSecurityHandler handler) throws Exception {
        http
                // 토큰 기반 무상태 API라 세션·CSRF·기본 로그인 화면을 쓰지 않는다
                .csrf(AbstractHttpConfigurer::disable)
                .formLogin(AbstractHttpConfigurer::disable)
                .httpBasic(AbstractHttpConfigurer::disable)
                .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .exceptionHandling(e -> e.authenticationEntryPoint(handler).accessDeniedHandler(handler))
                // 위에서부터 처음 매치된 규칙만 적용된다. 구체적인 규칙을 먼저, 마지막은 deny-by-default.
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/api/admin/**").hasRole("ADMIN")
                        .requestMatchers(HttpMethod.POST, "/api/auth/signup", "/api/auth/login").permitAll()
                        .requestMatchers(HttpMethod.GET, "/actuator/health").permitAll()
                        .requestMatchers(HttpMethod.GET, "/api/posts", "/api/posts/popular", "/api/posts/{postId}",
                                "/api/posts/{postId}/comments", "/api/rankings/activity", "/api/shop/items").permitAll()
                        .requestMatchers("/swagger-ui/**", "/v3/api-docs/**").permitAll()
                        .requestMatchers(HttpMethod.POST, "/api/pg/callbacks").permitAll()
                        .anyRequest().authenticated());
        return http.build();
    }
}

package com.example.board.config;

import java.io.IOException;
import java.util.LinkedHashMap;
import java.util.Map;

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import org.springframework.http.MediaType;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.web.AuthenticationEntryPoint;
import org.springframework.security.web.access.AccessDeniedHandler;
import org.springframework.stereotype.Component;

import com.example.board.common.ErrorCode;

import tools.jackson.databind.json.JsonMapper;

/** 필터 단계에서 막힌 401·403도 컨트롤러 에러와 같은 ProblemDetail 형식으로 내보낸다. */
@Component
public class ProblemDetailSecurityHandler implements AuthenticationEntryPoint, AccessDeniedHandler {

    private final JsonMapper jsonMapper;

    public ProblemDetailSecurityHandler(JsonMapper jsonMapper) {
        this.jsonMapper = jsonMapper;
    }

    @Override
    public void commence(HttpServletRequest request, HttpServletResponse response, AuthenticationException e)
            throws IOException {
        write(request, response, ErrorCode.UNAUTHORIZED, "인증이 필요합니다.");
    }

    @Override
    public void handle(HttpServletRequest request, HttpServletResponse response, AccessDeniedException e)
            throws IOException {
        write(request, response, ErrorCode.FORBIDDEN, "접근 권한이 없습니다.");
    }

    private void write(HttpServletRequest request, HttpServletResponse response, ErrorCode code, String detail)
            throws IOException {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("type", code.type());
        body.put("title", code.title());
        body.put("status", code.status().value());
        body.put("detail", detail);
        body.put("instance", request.getRequestURI());
        response.setStatus(code.status().value());
        response.setContentType(MediaType.APPLICATION_PROBLEM_JSON_VALUE);
        response.setCharacterEncoding("UTF-8");
        response.getWriter().write(jsonMapper.writeValueAsString(body));
    }
}
```

package com.example.board.member;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;

import com.example.board.common.BaseEntity;

import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;

@Getter
@Entity
@Table(name = "member")
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class Member extends BaseEntity {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 100)
    private String email;

    @Column(name = "password_hash", nullable = false, length = 100)
    private String passwordHash;

    @Column(nullable = false, length = 20)
    private String nickname;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 10)
    private Role role;

    // 잔액은 D07에서 원자적 UPDATE로만 바꾼다
    @Column(name = "point_balance", nullable = false)
    private long pointBalance;

    private Member(String email, String passwordHash, String nickname, Role role) {
        this.email = email;
        this.passwordHash = passwordHash;
        this.nickname = nickname;
        this.role = role;
    }

    public static Member create(String email, String passwordHash, String nickname, Role role) {
        return new Member(email, passwordHash, nickname, role);
    }
}

package com.example.board.member;

import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.example.board.common.BusinessException;
import com.example.board.common.ErrorCode;
import com.example.board.common.NicknameNormalizer;

@Service
public class AuthService {

    private final MemberRepository memberRepository;
    private final PasswordEncoder passwordEncoder;

    public AuthService(MemberRepository memberRepository, PasswordEncoder passwordEncoder) {
        this.memberRepository = memberRepository;
        this.passwordEncoder = passwordEncoder;
    }

    @Transactional
    public Long signup(SignupRequest request) {
        String nickname = NicknameNormalizer.normalize(request.nickname());
        if (memberRepository.existsByEmail(request.email())) {
            throw new BusinessException(ErrorCode.DUPLICATE, "이미 가입된 이메일입니다.");
        }
        if (memberRepository.existsByNickname(nickname)) {
            throw new BusinessException(ErrorCode.DUPLICATE, "이미 사용 중인 닉네임입니다.");
        }
        Member member = Member.create(request.email(), passwordEncoder.encode(request.password()), nickname, Role.USER);
        try {
            // 동시 가입은 UNIQUE 제약이 막는다. 즉시 flush해서 여기서 잡는다.
            return memberRepository.saveAndFlush(member).getId();
        } catch (DataIntegrityViolationException e) {
            throw new BusinessException(ErrorCode.DUPLICATE, "이미 가입된 이메일 또는 닉네임입니다.");
        }
    }
}

package com.example.board.config;

import java.nio.charset.StandardCharsets;
import java.time.Clock;
import java.util.Date;

import javax.crypto.SecretKey;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import com.example.board.common.BusinessException;
import com.example.board.common.ErrorCode;
import com.example.board.member.AuthMember;
import com.example.board.member.Role;

@Component
public class JwtProvider {

    private final SecretKey key;
    private final long expiresInSeconds;
    private final Clock clock;

    @Autowired
    public JwtProvider(BoardProperties properties) {
        this(properties, Clock.systemUTC());
    }

    // 만료 토큰 테스트를 위해 시계를 주입할 수 있게 둔다
    JwtProvider(BoardProperties properties, Clock clock) {
        this.key = Keys.hmacShaKeyFor(properties.jwt().secret().getBytes(StandardCharsets.UTF_8));
        this.expiresInSeconds = properties.jwt().expiresInSeconds();
        this.clock = clock;
    }

    public String issue(Long memberId, Role role) {
        Date now = Date.from(clock.instant());
        return Jwts.builder()
                .subject(String.valueOf(memberId))
                .claim("role", role.name())
                .issuedAt(now)
                .expiration(new Date(now.getTime() + expiresInSeconds * 1000))
                .signWith(key, Jwts.SIG.HS256)
                .compact();
    }

    public long expiresInSeconds() {
        return expiresInSeconds;
    }

    public AuthMember parse(String token) {
        try {
            Claims claims = Jwts.parser().verifyWith(key).clock(() -> Date.from(clock.instant())).build()
                    .parseSignedClaims(token).getPayload();
            return new AuthMember(Long.valueOf(claims.getSubject()), Role.valueOf(claims.get("role", String.class)));
        } catch (JwtException | IllegalArgumentException e) {
            // 서명 불일치·만료·형식 오류를 구분해 알리지 않는다
            throw new BusinessException(ErrorCode.UNAUTHORIZED, "유효하지 않은 토큰입니다.");
        }
    }
}

 * 토큰이 없거나 틀려도 여기서 응답하지 않고 인증 없이 다음 필터로 넘긴다.

 * 빈으로 등록하면 서블릿 필터로 한 번 더 걸리므로 SecurityConfig에서 직접 만든다.
 */
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private static final String PREFIX = "Bearer ";

    private final JwtProvider jwtProvider;

    public JwtAuthenticationFilter(JwtProvider jwtProvider) {
        this.jwtProvider = jwtProvider;
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        String header = request.getHeader(HttpHeaders.AUTHORIZATION);
        if (header != null && header.startsWith(PREFIX)) {
            try {
                AuthMember member = jwtProvider.parse(header.substring(PREFIX.length()));
                var authentication = new UsernamePasswordAuthenticationToken(member, null,
                        List.of(new SimpleGrantedAuthority("ROLE_" + member.role().name())));
                SecurityContextHolder.getContext().setAuthentication(authentication);
            } catch (BusinessException e) {
                SecurityContextHolder.clearContext();
            }
        }
        chain.doFilter(request, response);
    }
}

## ⑥ CORS 허용 출처 설정과 preflight를 인증보다 먼저 처리하기

**이 순서에서 만들 층** 해당 없음 · 공통 설정·지원 코드

**만들 파일** (내 프로젝트 · 경로는 구현 예시 기준)

- **새로 만들기**: `src/main/java/com/example/board/config/CorsConfig.java`
- **기존 수정**: `src/main/java/com/example/board/config/SecurityConfig.java`

**겪을 것** preflight가 인증 필터에 걸리면 curl은 되는데 브라우저에서만 요청이 막힌다.

**완료 조건**

- `OPTIONS /api/auth/login` + `Origin: http://localhost:3000` → 200, `Access-Control-Allow-Origin: http://localhost:3000`
- `Origin: http://evil.example` → 403
- 브라우저(`http://127.0.0.1:8765`)에서 `POST /api/auth/login` → 200

**명세** [`POST /api/auth/login`](../spec-v2.md#4-엔드포인트) ([5-1. 인증·회원](../spec-v2.md#5-1-인증회원))

<summary>막히면</summary>

- `CorsConfigurationSource` 빈과 `http.cors(...)`를 찾아본다. 허용 출처는 설정 값으로 받는다.
- curl로 `-X OPTIONS -H 'Origin: ...' -H 'Access-Control-Request-Method: POST'`를 보내 preflight 응답 헤더를 확인한다.

package com.example.board.config;

import java.util.List;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

@Configuration
public class CorsConfig {

    // spec-v2 §3의 허용 출처. 와일드카드(*)는 쓰지 않는다
    private static final List<String> ALLOWED_ORIGINS = List.of("http://localhost:3000", "http://127.0.0.1:8765");

    @Bean
    CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration config = new CorsConfiguration();
        config.setAllowedOrigins(ALLOWED_ORIGINS);
        config.setAllowedMethods(List.of("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"));
        config.setAllowedHeaders(List.of("Authorization", "Content-Type", "Idempotency-Key"));
        config.setMaxAge(3600L);
        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", config);
        return source;
    }
}

### TC-D03-2

**질문** 프론트(`http://localhost:3000`)에서만 로그인 요청이 막히고 curl로는 성공합니다. 왜 그런가요?

<details class="card-answer" data-card="TC-D03-2">
<summary>답에 들어가야 할 것</summary>

- 브라우저는 다른 출처로 JSON 본문이나 `Authorization` 헤더를 보낼 때 먼저 `OPTIONS` preflight를 보내고, curl은 CORS를 검사하지 않는다
- preflight에는 인증 정보가 없으므로 CORS 처리가 인증보다 먼저 돼야 한다. 그렇지 않으면 401·403으로 막히고 브라우저에는 CORS 에러로만 보인다
- `http.cors(...)`로 Security 체인에 CORS를 붙이고, 허용 출처는 명세의 두 곳만 둔다

**근거** spec-v2 §3

</details>

### TC-D03-3

**질문** 토큰 없이 `GET /api/members/me`를 부르면 401, USER 토큰으로 `GET /api/admin/reports`를 부르면 403입니다. 두 응답을 어디서 만들었나요?

<details class="card-answer" data-card="TC-D03-3">
<summary>답에 들어가야 할 것</summary>

- 401(인증 없음·만료)은 `AuthenticationEntryPoint`가, 403(인증됐지만 권한 없음)은 `AccessDeniedHandler`가 만든다
- URL 인가 단계의 401·403은 필터 체인에서 처리돼 `@RestControllerAdvice`까지 오지 않으므로, 두 핸들러에서 직접 응답을 쓴다
- 두 응답도 다른 에러와 같은 ProblemDetail 형식으로 낸다

**근거** spec-v2 §2

</details>

### TC-D03-4

**질문** 로그인 실패 때 "없는 이메일"과 "틀린 비밀번호"를 구분해 알려 주면 안 되는 이유는?

<details class="card-answer" data-card="TC-D03-4">
<summary>답에 들어가야 할 것</summary>

- 구분해 알려 주면 응답만으로 가입된 이메일을 알아낼 수 있다
- 명세대로 두 경우 모두 같은 401 `/errors/unauthorized`로 응답한다
- 비밀번호는 BCrypt 해시로 저장하고 `PasswordEncoder.matches`로 비교한다. 해시에 솔트가 들어 있어 같은 비밀번호도 매번 다른 해시가 나온다

**근거** spec-v2 §5-1

</details>

### TC-D03-5

**질문** JWT 필터는 토큰이 없을 때 401을 바로 내지 않고 다음 필터로 넘깁니다. 왜 그렇게 했나요?

<details class="card-answer" data-card="TC-D03-5">
<summary>답에 들어가야 할 것</summary>

- 공개 경로(가입·로그인·공개 GET)도 같은 필터를 지나므로, 토큰이 없다고 막으면 공개 API까지 막힌다
- 필터는 토큰이 유효할 때만 `SecurityContextHolder`에 인증을 올리고, 막을지는 인가 규칙(`authorizeHttpRequests`)이 정한다
- 토큰이 없거나 잘못된 요청이 보호 경로에 오면 진입점(`AuthenticationEntryPoint`)이 401을 낸다

</details>

- [TC-D03-1] permitAll 규칙을 관리자 규칙보다 앞에 두면 어떤 일이 생기나요?
    · 인가 규칙은 위에서부터 첫 번째로 매치된 것만 적용된다
    · 넓은 패턴이 앞에 있으면 뒤의 관리자 규칙은 평가되지 않는다
    · 테스트로 관리자 경로에 USER 토큰을 보내 403을 확인한다
- [TC-D03-2] 프론트(http://localhost:3000)에서만 로그인 요청이 막히고 curl로는 성공합니다. 왜 그런가요?
    · 브라우저는 다른 출처로 JSON 본문이나 Authorization 헤더를 보낼 때 먼저 OPTIONS preflight를 보내고, curl은 CORS를 검사하지 않는다
    · preflight에는 인증 정보가 없으므로 CORS 처리가 인증보다 먼저 돼야 한다. 그렇지 않으면 401·403으로 막히고 브라우저에는 CORS 에러로만 보인다
    · http.cors(...)로 Security 체인에 CORS를 붙이고, 허용 출처는 명세의 두 곳만 둔다
- [TC-D03-3] 토큰 없이 GET /api/members/me를 부르면 401, USER 토큰으로 GET /api/admin/reports를 부르면 403입니다. 두 응답을 어디서 만들었나요?
    · 401(인증 없음·만료)은 AuthenticationEntryPoint가, 403(인증됐지만 권한 없음)은 AccessDeniedHandler가 만든다
    · URL 인가 단계의 401·403은 필터 체인에서 처리돼 @RestControllerAdvice까지 오지 않으므로, 두 핸들러에서 직접 응답을 쓴다
    · 두 응답도 다른 에러와 같은 ProblemDetail 형식으로 낸다
- [TC-D03-4] 로그인 실패 때 "없는 이메일"과 "틀린 비밀번호"를 구분해 알려 주면 안 되는 이유는?
    · 구분해 알려 주면 응답만으로 가입된 이메일을 알아낼 수 있다
    · 명세대로 두 경우 모두 같은 401 /errors/unauthorized로 응답한다
    · 비밀번호는 BCrypt 해시로 저장하고 PasswordEncoder.matches로 비교한다. 해시에 솔트가 들어 있어 같은 비밀번호도 매번 다른 해시가 나온다
- [TC-D03-5] JWT 필터는 토큰이 없을 때 401을 바로 내지 않고 다음 필터로 넘깁니다. 왜 그렇게 했나요?
    · 공개 경로(가입·로그인·공개 GET)도 같은 필터를 지나므로, 토큰이 없다고 막으면 공개 API까지 막힌다
    · 필터는 토큰이 유효할 때만 SecurityContextHolder에 인증을 올리고, 막을지는 인가 규칙(authorizeHttpRequests)이 정한다
    · 토큰이 없거나 잘못된 요청이 보호 경로에 오면 진입점(AuthenticationEntryPoint)이 401을 낸다

리뷰 초점
- JWT 서명 키가 코드·yml에 하드코딩되지 않고 `BOARD_JWT_SECRET` 환경 변수로 들어오는지
- 로그인 실패 응답이 이메일 존재 여부를 드러내지 않는지
