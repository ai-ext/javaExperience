package com.example.board.common;

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.List;

/** 커서 방식 목록 응답. 마지막 쪽이면 nextCursor는 null이다. */
public record CursorPage<T>(List<T> items, String nextCursor) {

    /** 커서는 마지막 항목의 id를 Base64URL로 감싼 불투명 문자열이다. */
    public static String encode(long id) {
        return Base64.getUrlEncoder().withoutPadding().encodeToString(Long.toString(id).getBytes(StandardCharsets.UTF_8));
    }

    /** 첫 쪽(null·빈 값)이면 null. 서버가 준 적 없는 값은 VALIDATION 예외다. */
    public static Long decode(String cursor) {
        if (cursor == null || cursor.isEmpty()) {
            return null;
        }
        try {
            long id = Long.parseLong(new String(Base64.getUrlDecoder().decode(cursor), StandardCharsets.UTF_8));
            if (id > 0) {
                return id;

    @Transactional(readOnly = true)
    public CursorPage<PostSummary> search(PostSearchCondition condition, String cursor, int size) {
        Long afterId = CursorPage.decode(cursor);
        PostSearchCondition safe = new PostSearchCondition(escapeLike(condition.keyword()), condition.tag());
        // 한 건 더 읽어 다음 쪽이 있는지만 판단한다
        List<PostSummary> rows = mapper.search(safe, afterId, size + 1);
        boolean hasNext = rows.size() > size;
        List<PostSummary> page = hasNext ? rows.subList(0, size) : rows;
        String next = hasNext ? CursorPage.encode(page.get(page.size() - 1).getId()) : null;
        return new CursorPage<>(page, next);
    }

    /** 사용자가 넣은 %·_가 와일드카드로 해석되지 않게 한다. */
    private static String escapeLike(String keyword) {
        return keyword == null ? null : keyword.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_");
    }

#TC-D06-1

질문 정렬 키를 요청 파라미터로 받아 ORDER BY ${sort}로 넣는 코드가 있습니다. 무엇이 문제고 어떻게 바꾸나요?

답에 들어가야 할 것
  • ${}는 값을 SQL 문자열에 그대로 끼워 넣으므로 SQL 인젝션이 가능하다
  • #{}는 PreparedStatement 파라미터로 바인딩돼 안전하지만, 값으로만 들어가므로 컬럼명이나 ORDER BY 대상에는 쓸 수 없다
  • 정렬은 허용 목록(명세상 latest만)으로 받고, 서버가 고정된 SQL 조각을 고른다

#TC-D06-2

질문 글 목록(WHERE status = 'ACTIVE' ... ORDER BY id DESC LIMIT 20)이 데이터가 늘자 느려졌습니다. 인덱스를 어떻게 정하고 확인했나요?

답에 들어가야 할 것
  • 동등 조건 컬럼을 앞에, 정렬 컬럼을 뒤에 둔 (status, id) 인덱스면 인덱스가 정렬을 제공해 filesort 없이 LIMIT만큼 읽고 멈춘다

#TC-D06-3

질문 같은 트랜잭션에서 JPA로 글 제목을 고친 직후 MyBatis로 목록을 읽었더니 옛 제목이 나왔습니다. 왜 그런가요?

답에 들어가야 할 것
  • JPA의 변경은 flush 전까지 영속성 컨텍스트에만 있고 DB에는 아직 UPDATE가 나가지 않았다
  • Hibernate의 자동 flush는 JPQL 실행 전에 일어나며, MyBatis가 JDBC로 직접 보내는 SQL은 Hibernate가 모르므로 flush를 일으키지 않는다
  • MyBatis 조회 전에 명시적으로 flush하고, 두 도구가 같은 트랜잭션(같은 커넥션)에 참여하는지 실험으로 확인한다

    · 동등 조건 컬럼을 앞에, 정렬 컬럼을 뒤에 둔 (status, id) 인덱스면 인덱스가 정렬을 제공해 filesort 없이 LIMIT만큼 읽고 멈춘다
    · EXPLAIN의 type, key, rows, Extra(Using filesort 여부)를 인덱스 전후로 비교하고, EXPLAIN ANALYZE로 실제 읽은 행 수를 본다
    · 커서 페이징은 id < 마지막 id로 이어 읽어, offset처럼 앞쪽 행을 읽고 버리지 않는다
- [TC-D06-3] 같은 트랜잭션에서 JPA로 글 제목을 고친 직후 MyBatis로 목록을 읽었더니 옛 제목이 나왔습니다. 왜 그런가요?
    · JPA의 변경은 flush 전까지 영속성 컨텍스트에만 있고 DB에는 아직 UPDATE가 나가지 않았다
    · Hibernate의 자동 flush는 JPQL 실행 전에 일어나며, MyBatis가 JDBC로 직접 보내는 SQL은 Hibernate가 모르므로 flush를 일으키지 않는다
    · MyBatis 조회 전에 명시적으로 flush하고, 두 도구가 같은 트랜잭션(같은 커넥션)에 참여하는지 실험으로 확인한다
- [TC-D06-4] MyBatis 목록에서 글마다 태그를 <collection select="...">로 가져왔더니 쿼리가 글 수만큼 늘었습니다. 어떻게 고치나요?
    · 중첩 select는 부모 행마다 쿼리를 따로 실행하므로 글 20개면 태그 쿼리가 20번 나간다
    · JOIN 한 번으로 가져와 resultMap의 중첩 결과로 묶고, <id>를 선언해 같은 글의 행을 합친다
    · 다만 1:N JOIN에 LIMIT을 걸면 글이 아니라 행 기준으로 잘리므로, 글 ID를 먼저 커서로 뽑고 태그는 IN으로 따로 가져온다

리뷰 초점
- 매퍼 XML에 `${}`로 사용자 입력이 들어가는 곳이 없는지
- 쓰기는 JPA, 목록·검색 읽기는 MyBatis로 경계가 지켜지는지
