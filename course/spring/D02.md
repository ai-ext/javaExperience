```java
package com.example.board.common;

import java.net.URI;
import java.util.List;
import java.util.Map;

import jakarta.servlet.http.HttpServletRequest;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.ServletWebRequest;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

@RestControllerAdvice
public class GlobalExceptionHandler extends ResponseEntityExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(BusinessException.class)
    ProblemDetail handleBusiness(BusinessException e, HttpServletRequest request) {
        ErrorCode code = e.errorCode();
        return problem(code.status(), code.type(), code.title(), e.getMessage(), request.getRequestURI());
    }

    // IllegalArgumentException 등을 400으로 바꾸지 않는다. 서버 버그가 클라이언트 잘못으로 보이면 장애가 숨는다.
    @ExceptionHandler(Exception.class)
    ProblemDetail handleUnexpected(Exception e, HttpServletRequest request) {
        log.error("처리되지 않은 예외", e);
        return problem(HttpStatus.INTERNAL_SERVER_ERROR, "/errors/internal", "서버 오류",
                "요청을 처리하지 못했습니다.", request.getRequestURI());
    }

    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(MethodArgumentNotValidException ex,
            HttpHeaders headers, HttpStatusCode status, WebRequest request) {
        ProblemDetail pd = problem(HttpStatus.BAD_REQUEST, ErrorCode.VALIDATION.type(), ErrorCode.VALIDATION.title(),
                "요청 값이 올바르지 않습니다.", path(request));
        List<Map<String, String>> errors = ex.getBindingResult().getFieldErrors().stream()
                .map(GlobalExceptionHandler::toError)
                .toList();
        pd.setProperty("errors", errors);
        return ResponseEntity.badRequest().body(pd);
    }

    // 나머지 Spring MVC 예외(404, 405, 본문 파싱 실패 등)는 상위 클래스가 ProblemDetail로 만든다. instance와 type만 맞춘다.
    @Override
    protected ResponseEntity<Object> handleExceptionInternal(Exception ex, Object body, HttpHeaders headers,
            HttpStatusCode statusCode, WebRequest request) {
        if (body instanceof ProblemDetail pd) {
            pd.setInstance(URI.create(path(request)));
            pd.setType(typeOf(statusCode));
        }
        return super.handleExceptionInternal(ex, body, headers, statusCode, request);
    }

    private static URI typeOf(HttpStatusCode status) {
        return URI.create(switch (status.value()) {
            case 404 -> ErrorCode.NOT_FOUND.type();
            case 400 -> ErrorCode.VALIDATION.type();
            default -> "about:blank";
        });
    }

    private static Map<String, String> toError(FieldError fe) {
        return Map.of("field", fe.getField(), "reason", String.valueOf(fe.getDefaultMessage()));
    }

    private static String path(WebRequest request) {
        return ((ServletWebRequest) request).getRequest().getRequestURI();
    }

    private static ProblemDetail problem(HttpStatus status, String type, String title, String detail, String instance) {
        ProblemDetail pd = ProblemDetail.forStatusAndDetail(status, detail);
        pd.setType(URI.create(type));
        pd.setTitle(title);
        pd.setInstance(URI.create(instance));
        return pd;
    }
}
```

package com.example.board.common;

import java.time.LocalDateTime;

import jakarta.persistence.Column;
import jakarta.persistence.EntityListeners;
import jakarta.persistence.MappedSuperclass;

import org.springframework.data.annotation.CreatedDate;
import org.springframework.data.annotation.LastModifiedDate;
import org.springframework.data.jpa.domain.support.AuditingEntityListener;

import lombok.Getter;

@Getter
@MappedSuperclass
@EntityListeners(AuditingEntityListener.class)
public abstract class BaseEntity {

    @CreatedDate
    @Column(name = "created_at", nullable = false, updatable = false)
    private LocalDateTime createdAt;

    @LastModifiedDate
    @Column(name = "updated_at", nullable = false)

}

```java
package com.example.board.common;

import java.util.Collection;
import java.util.Locale;
import java.util.Set;
import java.util.stream.Collectors;

import org.springframework.stereotype.Component;

import com.example.board.config.BoardProperties;

@Component
public class ForbiddenWordChecker {

    private final boolean ignoreCase;
    // 적재 시 통째로 교체하는 불변 집합. 요청별 상태는 두지 않는다.
    private volatile Set<String> words = Set.of();

    public ForbiddenWordChecker(BoardProperties properties) {
        this.ignoreCase = properties.forbiddenWords().ignoreCase();
    }

    public void replaceWords(Collection<String> newWords) {
        this.words = newWords.stream().map(this::fold).collect(Collectors.toUnmodifiableSet());
    }

    public int size() {
        return words.size();
    }

    public boolean contains(String text) {
        if (text == null || text.isEmpty()) {
            return false;
        }
        String target = fold(text);
        return words.stream().anyMatch(target::contains);
    }

    private String fold(String s) {
        return ignoreCase ? s.toLowerCase(Locale.ROOT) : s;
    }
}
```

    // "group:code" -> name. 적재 때 통째로 교체한다.
    private volatile Map<String, String> names = Map.of();

        this.names = Map.copyOf(newNames);
    }

    public int size() {
        return names.size();
    }

    public Optional<String> name(String group, String code) {
        return Optional.ofNullable(names.get(key(group, code)));

    static String key(String group, String code) {

}

### TC-D02-1

**질문** 금칙어 목록을 DB에서 읽어 메모리에 올리는 코드를 `@PostConstruct`가 아니라 `ApplicationReadyEvent`에 둔 이유는?

<details class="card-answer" data-card="TC-D02-1">
<summary>답에 들어가야 할 것</summary>

- `@PostConstruct`는 빈 하나를 초기화하는 도중에 실행되므로, 그 시점에는 다른 빈이나 트랜잭션 프록시가 아직 준비되지 않았을 수 있다
- `@PostConstruct` 안에서 같은 클래스의 `@Transactional` 메서드를 불러도 프록시를 거치지 않아 트랜잭션이 걸리지 않는다
- `ApplicationReadyEvent`는 컨텍스트 refresh와 runner 실행이 끝난 뒤 발행되므로 기동 후 적재는 여기나 `ApplicationRunner`에 둔다

</details>

### TC-D02-2

**질문** 금칙어 검사기에 마지막 검사 결과를 담는 필드 `lastMatched`를 두자는 코드가 올라왔습니다. 리뷰에서 뭐라고 하나요?

<details class="card-answer" data-card="TC-D02-2">
<summary>답에 들어가야 할 것</summary>

- Spring 빈의 기본 스코프는 singleton이라 모든 요청 스레드가 같은 인스턴스를 공유한다
- 요청마다 바뀌는 값을 인스턴스 필드에 쓰면 동시 요청끼리 덮어써 다른 사용자의 값이 섞인다
- 요청별 값은 지역 변수·인자·반환값으로 다루고, 필드에는 `final` 의존성과 불변 설정만 둔다

</details>

**질문** 닉네임 정규화 함수는 static 유틸로, 금칙어 검사기는 빈으로 만들었습니다. 무엇을 기준으로 나눴나요?

<details class="card-answer" data-card="TC-D02-3">
<summary>답에 들어가야 할 것</summary>

- 입력만 받아 결과를 돌려주고 외부 자원이 없는 순수 함수는 static 유틸로 둔다
- DB·설정·다른 빈에 의존하거나 트랜잭션·AOP가 필요한 것은 빈으로 만들어 주입받는다
- 빈은 생성자 주입으로 테스트 대역을 넣을 수 있지만, static 호출은 대역으로 바꾸기 어렵다

</details>

### TC-D02-4

**질문** 전역 예외 처리기에서 `IllegalArgumentException`을 전부 400으로 바꾸자는 제안이 있습니다. 어떻게 답하나요?

<details class="card-answer" data-card="TC-D02-4">
<summary>답에 들어가야 할 것</summary>

- `IllegalArgumentException`은 서버 버그에서도 나므로, 일괄 400으로 바꾸면 서버 결함이 클라이언트 잘못으로 보이고 5xx 지표에서 사라진다
- 클라이언트 잘못이 명백한 경우만 도메인 예외로 정의해 4xx로 매핑하고, 예상하지 못한 예외는 500으로 둔다
- 모든 에러는 ProblemDetail(`application/problem+json`)로 내고, 클라이언트는 `type`으로 분기한다

### TC-D02-5

**질문** 여러 클래스에서 `@Value`로 읽던 금칙어 설정을 `@ConfigurationProperties` 하나로 묶었습니다. 무엇이 좋아졌나요?

<details class="card-answer" data-card="TC-D02-5">
<summary>답에 들어가야 할 것</summary>

- 관련 설정이 한 타입에 모이고, `@Validated`를 붙이면 잘못된 값이 기동 시점에 실패한다
- relaxed binding으로 kebab-case, camelCase, 환경 변수 표기를 모두 받는다
- `@Value`는 relaxed binding이 제한적이라, 키를 정규형(kebab-case)으로 쓰지 않으면 다른 표기의 값을 못 찾을 수 있다

</details>

## AI 채점 프롬프트

오늘 순서를 다 만들었으면 서버를 띄운 채로 아래 프롬프트를 복사해 Codex 또는 Claude Code에 붙여 넣는다. 에이전트가 완료 조건·함정·말하기 카드로 채점표를 준다. 코드는 고치지 않는다.

```text

레포 경로: 내 프로젝트 루트(이 프롬프트를 붙여 넣은 에이전트를 연 폴더)
서버 주소: http://localhost:8080

너는 Java·Spring 온보딩 과정 v2로 백엔드를 배우는 사람의 리뷰어다. 학습자는 빈 프로젝트에서 명세서 v2의 서버를 하루씩 만들고 있고, 오늘은 D02다.
설정 형식 채점: application.yml과 application.properties를 모두 인정한다. application-local·prod·test 등 프로파일별 파일도 두 형식을 인정한다. 파일 확장자나 YAML의 --- 구조만으로 실패 처리하지 말고 같은 키·값, 활성 프로파일, 자리표시자, 실행 결과를 확인한다. ApplicationYmlTest·ProfileConfigTest·LocalProfileSeedTest처럼 YAML 원문을 읽는 예시는 properties의 동등한 키·값·프로파일 로딩 검사로 대체할 수 있다.
지킬 것: 코드는 절대 수정하지 마라. DB는 조회만 한다(SELECT·SHOW·EXPLAIN). 서버를 대신 띄우거나 끄지 마라. 구현 예시 코드를 통째로 주지 말고 방향과 힌트로 알려 줘라.

1. 서버 주소의 `GET /actuator/health`로 응답 가능 여부를 확인해라. 꺼져 있으면 `./gradlew bootRun --args='--spring.profiles.active=local'`(또는 `SPRING_PROFILES_ACTIVE=local ./gradlew bootRun`)으로 띄우라고만 알려 주고 멈춰라. D01 ③부터 DB 설정이 `local` 프로파일에 있어 프로파일 없이는 기동에 실패한다.
설계 채점: 테이블·컬럼·키·제약·인덱스 이름은 학습자의 설계를 따른다. board-server의 이름·분할과 다르다는 이유로 실패 처리하지 마라. 설계 메모와 DB 불변식 확인은 제출 SQL·거부 결과·테스트를 읽고 판단한다. 채점자가 INSERT·UPDATE·DELETE로 재현하지 말고 증거가 없으면 확인 불가로 적어라. API 필드·상태·오류 계약은 유지한다.
2. 아래 "채점 기준"이 통과/실패의 유일한 기준이다. 기준에 없는 것으로 실패 처리하지 말고, 다른 날의 기준은 보지 마라.
3. [C] 항목: HTTP 요청은 서버 주소로 직접 보내 상태 코드·응답 헤더·본문을 기대값과 비교한다. 테스트 이름이 적힌 항목은 레포 루트에서 `./gradlew test --tests '<테스트 클래스 이름>'`으로 실행해 통과 여부를 본다(테스트 코드도 고치지 마라). 로그 항목은 기동 로그에서 찾거나 학습자에게 해당 줄을 붙여 달라고 한다. DB 확인이 필요하면 `docker compose exec -T mysql mysql -uboard -pboard board -e "<SELECT 문>"`으로 조회만 한다.
4. [R] 항목: 관련 코드를 찾아 읽고 그 함정을 피했는지 판단해라. 지적할 때는 `파일:줄`을 붙여라.
5. 말하기 카드: 학습자에게 카드 질문마다 답을 말하거나 적어 달라고 해라. "답에 들어가야 할 것"(· 표시) 세 줄이 모두 나오면 통과다. 빠진 줄은 정답 문장을 먼저 말하지 말고 되묻는 질문으로 끌어내라.
6. 리뷰 초점은 채점표에 넣지 말고 ②·③에 적어라.

말하기 카드
- [TC-D02-1] 금칙어 목록을 DB에서 읽어 메모리에 올리는 코드를 @PostConstruct가 아니라 ApplicationReadyEvent에 둔 이유는?
    · @PostConstruct는 빈 하나를 초기화하는 도중에 실행되므로, 그 시점에는 다른 빈이나 트랜잭션 프록시가 아직 준비되지 않았을 수 있다
    · @PostConstruct 안에서 같은 클래스의 @Transactional 메서드를 불러도 프록시를 거치지 않아 트랜잭션이 걸리지 않는다
    · ApplicationReadyEvent는 컨텍스트 refresh와 runner 실행이 끝난 뒤 발행되므로 기동 후 적재는 여기나 ApplicationRunner에 둔다
- [TC-D02-2] 금칙어 검사기에 마지막 검사 결과를 담는 필드 lastMatched를 두자는 코드가 올라왔습니다. 리뷰에서 뭐라고 하나요?
    · Spring 빈의 기본 스코프는 singleton이라 모든 요청 스레드가 같은 인스턴스를 공유한다
    · 요청마다 바뀌는 값을 인스턴스 필드에 쓰면 동시 요청끼리 덮어써 다른 사용자의 값이 섞인다
    · 요청별 값은 지역 변수·인자·반환값으로 다루고, 필드에는 final 의존성과 불변 설정만 둔다
- [TC-D02-3] 닉네임 정규화 함수는 static 유틸로, 금칙어 검사기는 빈으로 만들었습니다. 무엇을 기준으로 나눴나요?
    · 입력만 받아 결과를 돌려주고 외부 자원이 없는 순수 함수는 static 유틸로 둔다
    · DB·설정·다른 빈에 의존하거나 트랜잭션·AOP가 필요한 것은 빈으로 만들어 주입받는다
    · 빈은 생성자 주입으로 테스트 대역을 넣을 수 있지만, static 호출은 대역으로 바꾸기 어렵다
- [TC-D02-4] 전역 예외 처리기에서 IllegalArgumentException을 전부 400으로 바꾸자는 제안이 있습니다. 어떻게 답하나요?
    · IllegalArgumentException은 서버 버그에서도 나므로, 일괄 400으로 바꾸면 서버 결함이 클라이언트 잘못으로 보이고 5xx 지표에서 사라진다
    · 클라이언트 잘못이 명백한 경우만 도메인 예외로 정의해 4xx로 매핑하고, 예상하지 못한 예외는 500으로 둔다
    · 모든 에러는 ProblemDetail(application/problem+json)로 내고, 클라이언트는 type으로 분기한다
- [TC-D02-5] 여러 클래스에서 @Value로 읽던 금칙어 설정을 @ConfigurationProperties 하나로 묶었습니다. 무엇이 좋아졌나요?
    · 관련 설정이 한 타입에 모이고, @Validated를 붙이면 잘못된 값이 기동 시점에 실패한다
    · relaxed binding으로 kebab-case, camelCase, 환경 변수 표기를 모두 받는다
    · @Value는 relaxed binding이 제한적이라, 키를 정규형(kebab-case)으로 쓰지 않으면 다른 표기의 값을 못 찾을 수 있다

리뷰 초점
- 모든 오류 응답이 같은 ProblemDetail 형식(`type`·`title`·`status`·`detail`·`instance`)으로 나가는지
- 서버 버그(NPE 등)가 4xx로 바뀌어 숨지 않고 500으로 남는지

아래 형식으로만 답해라.
① 채점표 — | 항목 | 기대 | 실제 | 결과 |  (결과: 통과 / 실패 / 확인 불가)
② 잘한 점
③ 부족한 점과 이유 — 실패한 C 항목부터
④ 내일 전에 고칠 것 — 우선순위순, 3개 이내
⑤ 말하기 카드 — | 카드 | 빠진 줄 | 결과 |
마지막 줄: `결과: 필수 N/M 통과 · 말하기 카드 K/L 통과` (필수 = C 항목, R은 참고)
```
