#6-3. 내 북마크 목록

#상황

"저장한 글 목록을 보여 주세요. 폴더별로도요. 그런데 저장해 둔 글이 지워졌으면 목록에서 말없이 사라지면 안 돼요 — 사용자는 '내가 저장한 게 왜 없어졌지?' 해요. '볼 수 없는 글'이라고 표시해 주세요."

4-3(좋아요한 글)과 비슷한 커서 목록이지만 규칙 하나가 반대다. 좋아요 목록은 볼 수 없는 글을 뺐고, 북마크 목록은 남기고 표시한다. 사용자가 직접 모아 둔 것이기 때문이다.

#화면 시나리오

BookmarkService.java, LikeService.java, PostAccessPolicy.java, BlockPolicy.java. 위 서버 처리 순서가 어디에서 검증·조회·상태 변경으로 이어지는지 따라간다. 변경된 메서드: myBookmarks, viewableBy, blockRelatedIds. 트랜잭션 경계는 아래 코드의 @Transactional 위치에서 확인한다.

JPA / Repository — DB 조건

BookmarkRepository.java, FollowRepository.java, MemberBlockRepository.java. 쿼리·파생 메서드의 조건을 화면 시나리오의 DB 단계 및 직접 확인 SQL과 대조한다. 직접 작성한 JPQL·SQL과 아래 파일의 @Query를 비교한다.

SELECT b.id, b.post_id, b.folder_id, b.memo, b.created_at
FROM bookmark b
WHERE b.member_id = :memberId
  AND (b.created_at < :cursorAt OR (b.created_at = :cursorAt AND b.id < :cursorId))
ORDER BY b.created_at DESC, b.id DESC
LIMIT :sizePlusOne;

첫 페이지는 커서 조건을 생략한다. JPA findMyBookmarks는 글·작성자를 함께 fetch하고, ALL·DEFAULT·ID 폴더 조건을 적용한다. size+1번째 행은 다음 페이지 존재 여부를 판단하는 데만 쓴다.

예외 처리 — 실패 응답

ApiExceptionHandler.java, InvalidCursorException.java. 예외가 어떤 HTTP 상태와 오류 본문으로 바뀌는지 실패 계약과 대조한다. 변경된 메서드: handleInvalidCursor.

@Modifying
@Query(&quot;UPDATE Bookmark b SET b.folder = null WHERE b.folder.id = :folderId&quot;)

// 6-3 내 북마크 — 북마크한 시각 최신순 키셋 페이징. 글·작성자를 함께 읽는다(항목마다 따로 조회하지 않음).
// 4-3과 달리 &quot;볼 수 있는 글&quot; 조건을 여기 넣지 않는다 — 볼 수 없게 된 글도 남기고 항목별로 표시하기 때문이다.
// folderMode: ALL(전체) / DEFAULT(기본 폴더 = folder_id IS NULL) / ID(그 폴더)
@Query(&quot;&quot;&quot;
        SELECT b FROM Bookmark b
        JOIN FETCH b.post p
        JOIN FETCH p.author
        LEFT JOIN b.folder f
        WHERE b.member.id = :memberId
          AND (:folderMode = 'ALL'
               OR (:folderMode = 'DEFAULT' AND f.id IS NULL)
               OR (:folderMode = 'ID' AND f.id = :folderId))
          AND (:cursorAt IS NULL
               OR b.createdAt &lt; :cursorAt
               OR (b.createdAt = :cursorAt AND b.id &lt; :cursorId))
        ORDER BY b.createdAt DESC, b.id DESC
        &quot;&quot;&quot;)

                               @Param(&quot;folderMode&quot;) String folderMode,
                               @Param(&quot;folderId&quot;) Long folderId,
                               @Param(&quot;cursorAt&quot;) LocalDateTime cursorAt,
                               @Param(&quot;cursorId&quot;) Long cursorId,
                               Pageable pageable);
// UNIQUE(member_id,post_id)를 이용해 동시 반복 PUT도 한 행으로 합친다.
@Modifying
@Query(value = &quot;&quot;&quot;
        INSERT INTO bookmark(member_id, post_id, folder_id, memo, created_at, updated_at)
        VALUES (:memberId, :postId, :folderId, :memo, NOW(), NOW())
        ON DUPLICATE KEY UPDATE id = id

int insertIfAbsent(@Param(&quot;memberId&quot;) Long memberId, @Param(&quot;postId&quot;) Long postId,
                   @Param(&quot;folderId&quot;) Long folderId, @Param(&quot;memo&quot;) String memo);

// REPEATABLE_READ의 이전 스냅샷이 아닌 현재 행을 읽는다.
@Query(value = &quot;SELECT * FROM bookmark WHERE member_id = :memberId AND post_id = :postId FOR UPDATE&quot;,

                                         @Param(&quot;postId&quot;) Long postId);

@Modifying

// 동시에 두 번 와서 둘 다 &quot;없음&quot;을 보더라도 UNIQUE(member_id, post_id)가 최후의 방어선이다.
@Transactional
public BookmarkResponse bookmark(Long postId, AuthenticatedMember loginMember, BookmarkRequest request) {
    Post post = findBookmarkablePost(postId, loginMember);
    BookmarkFolder folder = findMyFolder(request.folderId(), loginMember.id());
    bookmarkRepository.upsert(loginMember.id(), postId,
            folder == null ? null : folder.getId(), request.memo());
    Bookmark bookmark = bookmarkRepository.findCurrentForUpdate(loginMember.id(), postId)
            .orElseThrow(() -&gt; new IllegalStateException(&quot;북마크 저장 결과를 찾을 수 없습니다.&quot;));

    return toResponse(bookmark);
}

import java.time.LocalDateTime;

// 키셋(커서) 페이징의 "마지막으로 받은 항목" — (시각, id)를 Base64url로 감싼다.
// 클라이언트는 열어 보지 않고 그대로 돌려보낸다. 4-3(좋아요한 글)과 6-3(북마크)이 같이 쓴다.
public record Cursor(LocalDateTime at, Long id) {


public String encode() {
    String raw = at + &quot;|&quot; + id;
    return Base64.getUrlEncoder().withoutPadding().encodeToString(raw.getBytes(StandardCharsets.UTF_8));
}

// null이면 첫 페이지(null 반환). 형식이 깨졌으면 400
public static Cursor decode(String value) {
    if (value == null) return null;
    if (value.isBlank()) throw new InvalidCursorException();
    try {
        String raw = new String(Base64.getUrlDecoder().decode(value), StandardCharsets.UTF_8);
        String[] parts = raw.split(&quot;\\|&quot;, -1);
        if (parts.length != 2) throw new IllegalArgumentException();
        LocalDateTime at = LocalDateTime.parse(parts[0]);
        long id = Long.parseLong(parts[1]);
        if (id &lt; 1) throw new IllegalArgumentException();
        return new Cursor(at, id);
    } catch (IllegalArgumentException | ArrayIndexOutOfBoundsException | DateTimeParseException e) {
        throw new InvalidCursorException();
    }
}

private static final String PUBLISHED = &quot;PUBLISHED&quot;;
private static final String DRAFT = &quot;DRAFT&quot;;
private static final String HIDDEN = &quot;HIDDEN&quot;;
private static final String PUBLIC = &quot;PUBLIC&quot;;
private static final String FOLLOWERS = &quot;FOLLOWERS&quot;;

public PostAccessPolicy(FollowRepository followRepository, BlockPolicy blockPolicy) {

// 글 하나를 판정한다 — 팔로우·차단은 필요할 때만 그 자리에서 조회한다
public boolean canView(AuthenticatedMember viewer, Post post) {

            authorId -&gt; followRepository.existsByFollower_IdAndFollowee_Id(viewer.id(), authorId),
            authorId -&gt; blockPolicy.isBlockedEitherWay(viewer.id(), authorId));

// 여러 글을 한꺼번에 판정할 때(6-3 북마크 목록) — 같은 규칙을 쓰되 팔로우·차단 정보를
// 미리 한 번에 읽어 두고 메모리에서 판정한다(항목마다 쿼리를 보내지 않는다).
public Predicate&lt;Post&gt; viewableBy(AuthenticatedMember viewer) {
    Set&lt;Long&gt; followees = viewer == null ? Set.of() : Set.copyOf(followRepository.findFolloweeIds(viewer.id()));
    Set&lt;Long&gt; blocked = viewer == null ? Set.of() : blockPolicy.blockRelatedIds(viewer.id());
    return post -&gt; canView(viewer, post, followees::contains, blocked::contains);
}

// follows/blocked는 viewer가 있을 때만 호출된다
private boolean canView(AuthenticatedMember viewer, Post post,
                        Predicate&lt;Long&gt; follows, Predicate&lt;Long&gt; blocked) {
    // 1. 삭제된 글은 작성자 본인도 못 본다
    if (post.getDeletedAt() != null) {
        return false;
    }

    Long viewerId = viewer == null ? null : viewer.id();
    Long authorId = post.getAuthor().getId();
    boolean isAuthor = viewerId != null &amp;&amp; viewerId.equals(authorId);
    boolean isAdmin = viewer != null &amp;&amp; ADMIN_ROLE.equals(viewer.role());

    // 2. 상태별 허용 — DRAFT는 작성자만, HIDDEN은 운영자만
    boolean statusOk = switch (post.getStatus()) {
        case PUBLISHED -&gt; true;
        case DRAFT -&gt; isAuthor;
        case HIDDEN -&gt; isAdmin;
        default -&gt; false;
    };
    if (!statusOk) {
        return false;
    }

    // 3. PUBLISHED + FOLLOWERS 공개는 작성자 본인이거나 팔로워만
    if (PUBLISHED.equals(post.getStatus()) &amp;&amp; FOLLOWERS.equals(post.getVisibility()) &amp;&amp; !isAuthor) {
        if (viewerId == null || !follows.test(authorId)) {
            return false;
        }

    // 4. 둘 사이에 차단이 있으면(어느 쪽이 차단했든) 못 본다
    if (viewerId != null &amp;&amp; !isAuthor &amp;&amp; blocked.test(authorId)) {
        return false;
    }

    // 5. 발행 글의 공개 범위가 알려진 값인지 검사한다.
    // 비로그인과 일반 열람자는 PUBLIC 또는 위에서 확인한 FOLLOWERS만 볼 수 있다.
    // DB에 PRIVATE/알 수 없는 값이 들어와도 공개로 취급하지 않는다.
    if (PUBLISHED.equals(post.getStatus()) &amp;&amp; !isAuthor
            &amp;&amp; !PUBLIC.equals(post.getVisibility())
            &amp;&amp; !FOLLOWERS.equals(post.getVisibility())) {
        return false;
    }
    return true;
}

@Transactional
public BookmarkFolderResponse create(AuthenticatedMember me, BookmarkFolderRequest request) {
    String name = normalizeName(request.name());
    // 같은 회원의 생성 요청을 직렬화해 10개 한도를 지킨다.
    folders.lockMember(me.id());
    if (folders.countByMember(me.id()) &gt;= MAX_FOLDERS) {
        throw new InvalidBookmarkFolderException(&quot;폴더는 10개까지 만들 수 있습니다.&quot;);
    }
    if (folders.countByName(me.id(), name, null) &gt; 0) throw new DuplicateBookmarkFolderNameException();
    BookmarkFolderInsertRow row = new BookmarkFolderInsertRow(me.id(), name, LocalDateTime.now());
    try {
        folders.insert(row);
    } catch (DataIntegrityViolationException ex) {
        throw new DuplicateBookmarkFolderNameException();
    }
    return new BookmarkFolderResponse(row.id, name);
}

@Transactional
public BookmarkFolderResponse rename(AuthenticatedMember me, Long folderId, BookmarkFolderRequest request) {
    String name = normalizeName(request.name());
    if (folders.findMine(folderId, me.id()) == null) throw new BookmarkFolderNotFoundException();
    if (folders.countByName(me.id(), name, folderId) &gt; 0) throw new DuplicateBookmarkFolderNameException();
    try {
        if (folders.rename(folderId, me.id(), name) != 1) throw new BookmarkFolderNotFoundException();
    } catch (DataIntegrityViolationException ex) {
        throw new DuplicateBookmarkFolderNameException();
    }
    return new BookmarkFolderResponse(folderId, name);
}

@Transactional
public void delete(AuthenticatedMember me, Long folderId) {
    if (folders.findMine(folderId, me.id()) == null) throw new BookmarkFolderNotFoundException();
    folders.moveAllToDefault(folderId, me.id());
    if (folders.deleteMine(folderId, me.id()) != 1) throw new BookmarkFolderNotFoundException();
}

private String normalizeName(String raw) {
    String name = raw == null ? &quot;&quot; : raw.strip();
    if (name.isEmpty() || name.length() &gt; 50) {
        throw new InvalidBookmarkFolderException(&quot;폴더 이름은 1~50자여야 합니다.&quot;);
    }

@Transactional
public BookmarkResponse bookmark(Long postId, AuthenticatedMember me, BookmarkRequest request) {
    if (visiblePosts.findVisibleDetail(postId, me.id(), me.role()) == null
            || !&quot;PUBLISHED&quot;.equals(bookmarks.findPostStatus(postId))) {
        throw new PostNotFoundException();
    }
    Long folderId = request.folderId();
    if (folderId != null &amp;&amp; bookmarks.findOwnedFolder(folderId, me.id()) == null) {
        throw new BookmarkFolderNotFoundException();
    }
    // UNIQUE(member_id,post_id)를 이용해 생성과 변경을 하나의 SQL로 처리한다.
    bookmarks.upsert(new BookmarkInsertRow(me.id(), postId, folderId, request.memo(), LocalDateTime.now()));
    // FOR UPDATE로 현재 행을 읽는다. 반복 PUT도 같은 bookmarkId를 돌려준다.
    BookmarkRow row = bookmarks.findBookmark(me.id(), postId);
    return new BookmarkResponse(row.id, postId, true, row.folderId, row.memo);
}

@Transactional

    bookmarks.deleteByMemberAndPost(me.id(), postId);
}

@Transactional
public BookmarkResponse move(Long bookmarkId, AuthenticatedMember me, BookmarkMoveRequest request) {
    BookmarkRow row = bookmarks.findOwnedBookmark(bookmarkId, me.id());
    if (row == null) throw new BookmarkNotFoundException();
    Long folderId = request.folderId();
    if (folderId != null &amp;&amp; bookmarks.findOwnedFolder(folderId, me.id()) == null) {
        throw new BookmarkFolderNotFoundException();
    }
    if (bookmarks.moveBookmark(bookmarkId, me.id(), folderId) != 1) {
        throw new BookmarkNotFoundException();
    }
    return new BookmarkResponse(bookmarkId, row.postId, true, folderId, row.memo);
}

@Transactional(readOnly = true)
public MyBookmarksResponse myBookmarks(AuthenticatedMember me, String folder, int size, String cursor) {
    String mode;
    Long folderId = null;
    if (folder == null) {
        mode = &quot;ALL&quot;;
    } else if (&quot;default&quot;.equals(folder)) {
        mode = &quot;DEFAULT&quot;;
    } else {
        mode = &quot;ID&quot;;
        try {
            folderId = Long.valueOf(folder);
        } catch (NumberFormatException ex) {
            throw new InvalidBookmarkFolderException(&quot;folder는 생략·default·폴더 id 중 하나여야 합니다.&quot;);
        }
        if (bookmarks.findOwnedFolder(folderId, me.id()) == null) {
            throw new BookmarkFolderNotFoundException();
        }
    }
    Cursor after = Cursor.decode(cursor);
    List&lt;BookmarkListRow&gt; rows = bookmarks.findMyBookmarks(me.id(), me.role(), mode, folderId,
            after == null ? null : after.at(), after == null ? null : after.id(), size + 1);
    boolean hasNext = rows.size() &gt; size;
    List&lt;BookmarkListRow&gt; page = hasNext ? rows.subList(0, size) : rows;
    List&lt;BookmarkItemResponse&gt; items = page.stream().map(row -&gt; {
        boolean available = Boolean.TRUE.equals(row.available);
        return new BookmarkItemResponse(row.id, row.postId, row.folderId, row.memo,
                row.bookmarkedAt, available, available ? row.title : null,
                available ? new AuthorResponse(row.authorId, row.authorNickname) : null);
    }).toList();
    BookmarkListRow last = page.isEmpty() ? null : page.get(page.size() - 1);
    return new MyBookmarksResponse(items,
            hasNext ? new Cursor(last.bookmarkedAt, last.id).encode() : null);
}

public MyLikesResponse myLikes(AuthenticatedMember me, int size, String cursor) {
    LocalDateTime cursorAt = null;
    Long cursorId = null;
    if (cursor != null) {
        String[] parts = decodeCursor(cursor);
        cursorAt = LocalDateTime.parse(parts[0]);
        cursorId = Long.valueOf(parts[1]);
    }
    List&lt;MyLikeRow&gt; rows = likes.findMyLikes(me.id(), cursorAt, cursorId, size + 1);
    boolean hasNext = rows.size() &gt; size;
    List&lt;MyLikeRow&gt; page = hasNext ? rows.subList(0, size) : rows;
    List&lt;MyLikeItemResponse&gt; items = page.stream()
            .map(row -&gt; new MyLikeItemResponse(row.postId, row.title,
                    new AuthorResponse(row.authorId, row.authorNickname), row.likedAt))
            .toList();
    String nextCursor = hasNext ? encodeCursor(page.get(page.size() - 1)) : null;
    return new MyLikesResponse(items, nextCursor);
}

private String encodeCursor(MyLikeRow last) {
    String raw = last.likedAt + &quot;|&quot; + last.likeId;
    return Base64.getUrlEncoder().withoutPadding()
            .encodeToString(raw.getBytes(StandardCharsets.UTF_8));
}

private String[] decodeCursor(String cursor) {
    try {
        String raw = new String(Base64.getUrlDecoder().decode(cursor), StandardCharsets.UTF_8);
        String[] parts = raw.split(&quot;\\|&quot;, -1);
        if (parts.length != 2) throw new IllegalArgumentException();
        LocalDateTime.parse(parts[0]);
        if (Long.parseLong(parts[1]) &lt; 1) throw new IllegalArgumentException();
        return parts;
    } catch (IllegalArgumentException | DateTimeException ex) {
        throw new ResponseStatusException(HttpStatus.BAD_REQUEST, &quot;올바르지 않은 cursor입니다.&quot;);
    }
}

@Transactional
public PostDetailResponse replace(Long id, AuthenticatedMember me, PostReplaceRequest request) {
    PostDetailRow current = visibilityMapper.findVisibleDetail(id, me.id(), me.role());
    if (current == null) {
        throw new PostNotFoundException();
    }
    if (!current.authorId.equals(me.id())) {
        throw new ResponseStatusException(HttpStatus.FORBIDDEN);
    }

    // 이름이 다른 대소문자 표기도 같은 tag 행으로 연결한다.

        PostReplaceTagRow tag = replaceMapper.findTagByName(name);

            tags.add(tag);
        }
    }

        replaceMapper.insertPostTag(id, tag.id);
    }

            request.content(), tags.stream().map(tag -&gt; tag.name).toList(),
            request.visibility(),
