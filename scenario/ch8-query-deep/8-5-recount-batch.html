JPA / Repository·SQL

RecountRepository.java: JPA 조회 조건을 DB 예시와 대조한다.

SELECT p.id AS postId, p.like_count AS beforeCount, COALESCE(l.cnt, 0) AS afterCount
FROM post p
LEFT JOIN (SELECT post_id, COUNT(*) AS cnt FROM post_like GROUP BY post_id) l
       ON l.post_id = p.id
WHERE p.deleted_at IS NULL AND p.like_count <> COALESCE(l.cnt, 0)
ORDER BY p.id;

UPDATE post p
LEFT JOIN (SELECT post_id, COUNT(*) AS cnt FROM post_like GROUP BY post_id) l
       ON l.post_id = p.id
SET p.like_count = COALESCE(l.cnt, 0)
WHERE p.deleted_at IS NULL AND p.like_count <> COALESCE(l.cnt, 0);

댓글은 comment.deleted_at IS NULL을 집계 조건에 추가한다. LEFT JOIN과 COALESCE가 있어야 좋아요·댓글이 0건인 글도 0으로 고칠 수 있다. 실제 @Query·@Modifying 전문은 아래 역할별 코드에 있다.

@Transactional
public RecountResponse recount() {
    List<CountFix> likeFixed = toFixes(recountRepository.findLikeCountMismatches());

package org.example.admin;


import org.example.post.Post;
import org.springframework.data.jpa.repository.Modifying;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.Repository;


import java.util.List;


// 8-5 정합성 배치 — 어긋난 글을 SELECT로 찾고, 글마다 엔티티를 불러 고치는 대신 UPDATE … JOIN(집계)으로 한 번에 고친다.
// (별칭에 before를 쓰지 않는다 — MySQL 예약어)
public interface RecountRepository extends Repository<Post, Long> {


@Query(nativeQuery = true, value = &quot;&quot;&quot;
        SELECT p.id AS postId, p.like_count AS beforeCount, COALESCE(l.cnt, 0) AS afterCount
          FROM post p
          LEFT JOIN (SELECT post_id, COUNT(*) AS cnt FROM post_like GROUP BY post_id) l ON l.post_id = p.id
         WHERE p.deleted_at IS NULL AND p.like_count &lt;&gt; COALESCE(l.cnt, 0)
         ORDER BY p.id
        &quot;&quot;&quot;)
List&lt;CountMismatchRow&gt; findLikeCountMismatches();

@Query(nativeQuery = true, value = &quot;&quot;&quot;
        SELECT p.id AS postId, p.comment_count AS beforeCount, COALESCE(c.cnt, 0) AS afterCount
          FROM post p
          LEFT JOIN (SELECT post_id, COUNT(*) AS cnt FROM comment WHERE deleted_at IS NULL GROUP BY post_id) c
                 ON c.post_id = p.id
         WHERE p.deleted_at IS NULL AND p.comment_count &lt;&gt; COALESCE(c.cnt, 0)
         ORDER BY p.id
        &quot;&quot;&quot;)
List&lt;CountMismatchRow&gt; findCommentCountMismatches();

@Modifying
@Query(nativeQuery = true, value = &quot;&quot;&quot;
        UPDATE post p
          LEFT JOIN (SELECT post_id, COUNT(*) AS cnt FROM post_like GROUP BY post_id) l ON l.post_id = p.id
           SET p.like_count = COALESCE(l.cnt, 0)
         WHERE p.deleted_at IS NULL AND p.like_count &lt;&gt; COALESCE(l.cnt, 0)
        &quot;&quot;&quot;)
int fixLikeCounts();

@Modifying
@Query(nativeQuery = true, value = &quot;&quot;&quot;
        UPDATE post p
          LEFT JOIN (SELECT post_id, COUNT(*) AS cnt FROM comment WHERE deleted_at IS NULL GROUP BY post_id) c
                 ON c.post_id = p.id

         WHERE p.deleted_at IS NULL AND p.comment_count &lt;&gt; COALESCE(c.cnt, 0)

package org.example.admin;


import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;


import java.util.List;


// 비정규화 숫자(like_count·comment_count)를 실제 행 수로 맞춘다. 몇 번을 돌려도 결과가 같다(멱등).
// 이건 "나중에 맞추는" 안전망이다 — 4-2·5-1처럼 숫자를 원자적으로 바꿔 애초에 어긋나지 않게 하는 게 먼저다.
// 수동 실행(AdminController)과 예약 실행(RecountScheduler)이 이 메서드 하나를 같이 쓴다.
@Service
public class RecountService {


private static final Logger log = LoggerFactory.getLogger(RecountService.class);

private final RecountRepository recountRepository;

public RecountService(RecountRepository recountRepository) {
    this.recountRepository = recountRepository;
}

// 찾기와 고치기를 한 트랜잭션에서 한다. 그 사이에 좋아요가 눌리면 응답의 after와 실제 값이
// 조금 다를 수 있지만, UPDATE는 그 시점의 실제 행 수로 맞추므로 결과는 맞다.
@Transactional
public RecountResponse recount() {
    List&lt;CountFix&gt; likeFixed = toFixes(recountRepository.findLikeCountMismatches());
    List&lt;CountFix&gt; commentFixed = toFixes(recountRepository.findCommentCountMismatches());

    if (!likeFixed.isEmpty()) {
        recountRepository.fixLikeCounts();
    }
    if (!commentFixed.isEmpty()) {
        recountRepository.fixCommentCounts();
    }

    likeFixed.forEach(f -&gt; log.info(&quot;like_count 보정 postId={} {} -&gt; {}&quot;, f.postId(), f.before(), f.after()));
    commentFixed.forEach(f -&gt; log.info(&quot;comment_count 보정 postId={} {} -&gt; {}&quot;, f.postId(), f.before(), f.after()));
    log.info(&quot;정합성 배치 완료: like {}건, comment {}건 보정&quot;, likeFixed.size(), commentFixed.size());
    return new RecountResponse(likeFixed, commentFixed);
}

private List&lt;CountFix&gt; toFixes(List&lt;CountMismatchRow&gt; rows) {
    return rows.stream()
            .map(r -&gt; new CountFix(r.getPostId(), r.getBeforeCount(), r.getAfterCount()))
            .toList();
}


}

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;


import java.util.List;


// 비정규화 숫자(like_count·comment_count)를 실제 행 수로 맞춘다. 몇 번을 돌려도 결과가 같다(멱등).
// 이건 "나중에 맞추는" 안전망이다 — 4-2·5-1처럼 숫자를 원자적으로 바꿔 애초에 어긋나지 않게 하는 게 먼저다.
// 수동 실행(AdminController)과 예약 실행(RecountScheduler)이 이 메서드 하나를 같이 쓴다.
@Service
public class RecountService {


private static final Logger log = LoggerFactory.getLogger(RecountService.class);

private final MyBatisRecountMapper recountMapper;

public RecountService(MyBatisRecountMapper recountMapper) {
    this.recountMapper = recountMapper;
}

// 찾기와 고치기를 한 트랜잭션에서 한다. 그 사이에 좋아요가 눌리면 응답의 after와 실제 값이
// 조금 다를 수 있지만, UPDATE는 그 시점의 실제 행 수로 맞추므로 결과는 맞다.
@Transactional
public RecountResponse recount() {
    List&lt;CountFix&gt; likeFixed = toFixes(recountMapper.findLikeCountMismatches());
    List&lt;CountFix&gt; commentFixed = toFixes(recountMapper.findCommentCountMismatches());

    if (!likeFixed.isEmpty()) {
        recountMapper.fixLikeCounts();
    }
    if (!commentFixed.isEmpty()) {
        recountMapper.fixCommentCounts();
    }

    likeFixed.forEach(f -&gt; log.info(&quot;like_count 보정 postId={} {} -&gt; {}&quot;, f.postId(), f.before(), f.after()));
    commentFixed.forEach(f -&gt; log.info(&quot;comment_count 보정 postId={} {} -&gt; {}&quot;, f.postId(), f.before(), f.after()));
    log.info(&quot;정합성 배치 완료: like {}건, comment {}건 보정&quot;, likeFixed.size(), commentFixed.size());
    return new RecountResponse(likeFixed, commentFixed);
}

private List&lt;CountFix&gt; toFixes(List&lt;CountMismatchRow&gt; rows) {
    return rows.stream()
            .map(r -&gt; new CountFix(r.postId, r.beforeCount, r.afterCount))
            .toList();
}


}

package org.example.bookmark;


import org.example.auth.AuthenticatedMember;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.time.LocalDateTime;
import java.util.List;


@Service
public class MyBatisBookmarkFolderService {
    private static final int MAX_FOLDERS = 10;
    private final MyBatisBookmarkFolderMapper folders;


public MyBatisBookmarkFolderService(MyBatisBookmarkFolderMapper folders) {
    this.folders = folders;
}

@Transactional(readOnly = true)
public List&lt;BookmarkFolderSummaryResponse&gt; list(AuthenticatedMember me) {
    return folders.findSummaries(me.id());
}

@Transactional
public BookmarkFolderResponse create(AuthenticatedMember me, BookmarkFolderRequest request) {
    String name = normalizeName(request.name());
    // 같은 회원의 생성 요청을 직렬화해 10개 한도를 지킨다.
    folders.lockMember(me.id());
    if (folders.countByMember(me.id()) &gt;= MAX_FOLDERS) {
        throw new InvalidBookmarkFolderException(&quot;폴더는 10개까지 만들 수 있습니다.&quot;);
    }
    if (folders.countByName(me.id(), name, null) &gt; 0) throw new DuplicateBookmarkFolderNameException();
    BookmarkFolderInsertRow row = new BookmarkFolderInsertRow(me.id(), name, LocalDateTime.now());
    try {
        folders.insert(row);
    } catch (DataIntegrityViolationException ex) {
        throw new DuplicateBookmarkFolderNameException();
    }
    return new BookmarkFolderResponse(row.id, name);
}

@Transactional
public BookmarkFolderResponse rename(AuthenticatedMember me, Long folderId, BookmarkFolderRequest request) {
    String name = normalizeName(request.name());
    if (folders.findMine(folderId, me.id()) == null) throw new BookmarkFolderNotFoundException();
    if (folders.countByName(me.id(), name, folderId) &gt; 0) throw new DuplicateBookmarkFolderNameException();
    try {
        if (folders.rename(folderId, me.id(), name) != 1) throw new BookmarkFolderNotFoundException();
    } catch (DataIntegrityViolationException ex) {
        throw new DuplicateBookmarkFolderNameException();
    }
    return new BookmarkFolderResponse(folderId, name);
}

@Transactional
public void delete(AuthenticatedMember me, Long folderId) {

private String[] decodeCursor(String cursor) {

private LikeResponse withDeadlockRetry(java.util.function.Supplier&lt;LikeResponse&gt; action) {

    if (!current.authorId.equals(me.id())) throw new PostForbiddenException();

    if (request.title() != null) validateTitle(request.title());
    if (request.content() != null) validateContent(request.content());
    if (request.tags() != null) validateTags(request.tags());
    if (request.visibility() != null) validateVisibility(request.visibility());

    List&lt;String&gt; tags;
    if (request.tags() == null) {
        tags = visibilityMapper.findTagsByPostIds(List.of(id)).stream()
                .map(PostTagRow::name).toList();
    } else {
        List&lt;PostReplaceTagRow&gt; resolved = new ArrayList&lt;&gt;();
        Set&lt;Long&gt; seenIds = new HashSet&lt;&gt;();

            replaceMapper.insertTagIfMissing(name);
            PostReplaceTagRow tag = replaceMapper.findTagByName(name);
            if (seenIds.add(tag.id)) resolved.add(tag);
        }
        replaceMapper.deletePostTags(id);
        for (PostReplaceTagRow tag : resolved) replaceMapper.insertPostTag(id, tag.id);
        tags = resolved.stream().map(tag -&gt; tag.name).toList();
    }
    replaceMapper.patchPost(id, request.title(), request.content(), request.visibility());
    return new PostDetailResponse(id, current.boardSlug,
            request.title() == null ? current.title : request.title(),
            request.content() == null ? current.content : request.content(),

public PostDetailResponse replace(Long id, AuthenticatedMember me, PostReplaceRequest request) {
    PostDetailRow current = visibilityMapper.findVisibleDetail(id, me.id(), me.role());
    if (current == null) {
        throw new PostNotFoundException();
    }
    if (!current.authorId.equals(me.id())) {
        throw new ResponseStatusException(HttpStatus.FORBIDDEN);
    }

    // 이름이 다른 대소문자 표기도 같은 tag 행으로 연결한다.

        replaceMapper.insertTagIfMissing(name);
        PostReplaceTagRow tag = replaceMapper.findTagByName(name);
        if (seenTagIds.add(tag.id)) {
            tags.add(tag);
        }
    }
    replaceMapper.updatePost(id, request.title(), request.content(), request.visibility());
    replaceMapper.deletePostTags(id);
    for (PostReplaceTagRow tag : tags) {
        replaceMapper.insertPostTag(id, tag.id);
    }
    return new PostDetailResponse(id, current.boardSlug, request.title(),
            request.content(), tags.stream().map(tag -&gt; tag.name).toList(),
            request.visibility(),
            new AuthorResponse(current.authorId, current.authorNickname),
