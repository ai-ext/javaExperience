MyBatis — 페이지 SELECT·COUNT·태그 일괄 조회

아래 MyBatis 대체 구현 코드의 8개 파일을 이 단계 전체 소스에 적용한다. 2-1의 상세 SELECT·조회수 UPDATE는 유지하고, PostReadMapper에 게시판 ID 조회와 목록·COUNT SQL을 더한다.

findPublishedByBoard와 countPublishedByBoard는 둘 다 board_id·status='PUBLISHED'·deleted_at IS NULL 조건을 써야 총 건수와 페이지 항목이 맞는다. ORDER BY published_at DESC, id DESC로 동률 순서를 고정하고, LIMIT #{limit} OFFSET #{offset}으로 페이지를 자른다. #{...}는 숫자를 바인딩한다.

목록 SQL 결과는 PostListRow에 담고, 페이지에 나온 글 ID만 모아 태그를 한 번 더 조회해 PostListItemResponse로 변환한다. content는 목록 SELECT와 DTO에 넣지 않는다.

Service는 COUNT로 totalElements·totalPages·hasNext를 계산한다. 이 단계는 아직 2-3의 팔로우·차단 조건을 적용하지 않으므로 6번·9번 글이 보여야 한다. JPA와 MyBatis 양쪽에서 L1~L5의 글 순서·14건/3페이지·잘못된 입력 오류를 대조한다.

package org.example.post;

import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

// 메서드 매개변수 제약(@Min/@Max)은 @Validated 없이도 Spring MVC가 직접 검증해
// HandlerMethodValidationException으로 바꿔 준다(Spring 6.1+). @Validated를 더하면
// 옛 AOP 방식(ConstraintViolationException)과 충돌해 500이 났다 — 붙이지 않는다.
@RestController
@RequestMapping("/api/boards/{slug}/posts")
public class BoardPostController {

    private final PostQueryService postQueryService;

    public BoardPostController(PostQueryService postQueryService) {
        this.postQueryService = postQueryService;
    }

    @GetMapping
    public PostListResponse list(@PathVariable String slug,
                                  @RequestParam(defaultValue = "0") @Min(0) int page,
                                  @RequestParam(defaultValue = "10") @Min(1) @Max(50) int size) {
        return postQueryService.list(slug, page, size);
    }
}

Service / 규칙 — src/main/java/org/example/post/PostQueryService.java
package org.example.post;

import org.example.board.Board;
import org.example.board.BoardNotFoundException;
import org.example.board.BoardRepository;
import org.example.tag.Tag;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

// 글 조회 전용 서비스 — 1-3/1-4의 PostService(쓰기)와 책임을 나눈다
@Service
public class PostQueryService {

    private static final String PUBLISHED = "PUBLISHED";

    private final PostRepository postRepository;
    private final BoardRepository boardRepository;

    public PostQueryService(PostRepository postRepository, BoardRepository boardRepository) {
        this.postRepository = postRepository;
        this.boardRepository = boardRepository;
    }

    @Transactional
    public PostDetailResponse getDetail(Long id) {
        Post post = postRepository.findById(id)
                .orElseThrow(PostNotFoundException::new);

        post.increaseViewCount();
        return toDetailResponse(post);
    }

    @Transactional(readOnly = true)
    public PostListResponse list(String boardSlug, int page, int size) {
        Board board = boardRepository.findBySlug(boardSlug)
                .orElseThrow(() -> new BoardNotFoundException(boardSlug));

        PageRequest pageRequest = PageRequest.of(page, size,
                Sort.by(Sort.Order.desc("publishedAt"), Sort.Order.desc("id")));
        Page<Post> result = postRepository.findByBoardAndStatusAndDeletedAtIsNull(board, PUBLISHED, pageRequest);

        List<PostListItemResponse> items = result.getContent().stream()
                .map(this::toListItem)
                .toList();

        return new PostListResponse(
                items,
                result.getNumber(),
                result.getSize(),
                result.getTotalElements(),
                result.getTotalPages(),
                result.hasNext()
        );
    }

    private PostListItemResponse toListItem(Post post) {
        return new PostListItemResponse(
                post.getId(),
                post.getTitle(),
                new AuthorResponse(post.getAuthor().getId(), post.getAuthor().getNickname()),
                post.getTags().stream().map(Tag::getName).toList(),
                post.getLikeCount(),
                post.getCommentCount(),
                post.getViewCount(),
                post.getPublishedAt()
        );
    }

    private PostDetailResponse toDetailResponse(Post post) {
        // 트랜잭션 안에서 지연 로딩 연관을 읽어 DTO로 옮긴다(open-in-view: false)
        return new PostDetailResponse(
                post.getId(),
                post.getBoard().getSlug(),
                post.getTitle(),
                post.getContent(),
                post.getTags().stream().map(Tag::getName).toList(),
                post.getVisibility(),
                new AuthorResponse(post.getAuthor().getId(), post.getAuthor().getNickname()),
                post.getViewCount(),
                post.getLikeCount(),
                post.getCommentCount(),
                post.getCreatedAt()
        );
    }
}

package org.example.post;

import org.example.board.BoardNotFoundException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

// 상세·목록·COUNT·태그를 Mapper SQL에서 읽는다.
@Service
public class PostQueryService {
    private final PostReadMapper mapper;

    public PostQueryService(PostReadMapper mapper) {
        this.mapper = mapper;
    }

    @Transactional
    public PostDetailResponse getDetail(Long id) {
        PostDetailRow row = mapper.findDetail(id);
        if (row == null) throw new PostNotFoundException();
        mapper.increaseViewCount(id);
        List<String> tags = mapper.findTagsByPostIds(List.of(id)).stream()
                .map(PostTagRow::name).toList();
        return new PostDetailResponse(row.id, row.boardSlug, row.title, row.content,
                tags, row.visibility, new AuthorResponse(row.authorId, row.authorNickname),
                row.viewCount + 1, row.likeCount, row.commentCount, row.createdAt);
    }

    @Transactional(readOnly = true)
    public PostListResponse list(String boardSlug, int page, int size) {
        Long boardId = mapper.findBoardIdBySlug(boardSlug);
        if (boardId == null) throw new BoardNotFoundException(boardSlug);
        long offset = (long) page * size;
        List<PostListRow> rows = mapper.findPublishedByBoard(boardId, size, offset);
        long total = mapper.countPublishedByBoard(boardId);
        Map<Long, List<String>> tags = rows.isEmpty() ? Map.of()
                : mapper.findTagsByPostIds(rows.stream().map(row -> row.id).toList()).stream()
                    .collect(Collectors.groupingBy(PostTagRow::postId,
                            Collectors.mapping(PostTagRow::name, Collectors.toList())));
        List<PostListItemResponse> items = rows.stream()
                .map(row -> new PostListItemResponse(row.id, row.title,
                        new AuthorResponse(row.authorId, row.authorNickname),
                        tags.getOrDefault(row.id, List.of()), row.likeCount,
                        row.commentCount, row.viewCount, row.publishedAt))
                .toList();
        return new PostListResponse(items, page, size, total,
                (int) ((total + size - 1) / size), offset + rows.size() < total);
    }
}

@Mapper
public interface PostReadMapper {
    PostDetailRow findDetail(@Param("id") Long id);
    int increaseViewCount(@Param("id") Long id);
    Long findBoardIdBySlug(@Param("slug") String slug);
    List<PostListRow> findPublishedByBoard(@Param("boardId") Long boardId,
            @Param("limit") int limit, @Param("offset") long offset);
    long countPublishedByBoard(@Param("boardId") Long boardId);
    List<PostTagRow> findTagsByPostIds(@Param("postIds") List<Long> postIds);
}

<mapper namespace="org.example.post.PostReadMapper">
    <select id="findDetail" resultType="org.example.post.PostDetailRow">
        SELECT p.id, b.slug AS board_slug, p.title, p.content, p.visibility,
               m.id AS author_id, m.nickname AS author_nickname,
               p.view_count, p.like_count, p.comment_count, p.created_at
        FROM post p JOIN board b ON b.id = p.board_id
                    JOIN member m ON m.id = p.author_id
        WHERE p.id = #{id}
    </select>
    <update id="increaseViewCount">
        UPDATE post SET view_count = view_count + 1 WHERE id = #{id}
    </update>
    <select id="findBoardIdBySlug" resultType="long">
        SELECT id FROM board WHERE slug = #{slug}
    </select>
    <select id="findPublishedByBoard" resultType="org.example.post.PostListRow">
        SELECT p.id, p.title, p.like_count, p.comment_count, p.view_count, p.published_at,
               m.id AS author_id, m.nickname AS author_nickname
        FROM post p JOIN member m ON m.id = p.author_id
        WHERE p.board_id = #{boardId} AND p.status = 'PUBLISHED' AND p.deleted_at IS NULL
        ORDER BY p.published_at DESC, p.id DESC
        LIMIT #{limit} OFFSET #{offset}
    </select>
    <select id="countPublishedByBoard" resultType="long">
        SELECT COUNT(*) FROM post p
        WHERE p.board_id = #{boardId} AND p.status = 'PUBLISHED' AND p.deleted_at IS NULL
    </select>
    <select id="findTagsByPostIds" resultType="org.example.post.PostTagRow">
        SELECT pt.post_id AS postId, t.name FROM post_tag pt
        JOIN tag t ON t.id = pt.tag_id
        WHERE pt.post_id IN
        <foreach collection="postIds" item="id" open="(" separator="," close=")">#{id}</foreach>
        ORDER BY t.name
    </select>
