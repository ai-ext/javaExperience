화면 시나리오의 서버·DB 흐름을 이 단계에서 바뀐 파일에 연결한다. 먼저 만들 것의 계약대로 직접 작성하고, 필요한 역할만 펼쳐 파일 전문과 비교한다.

이 단계의 처리 흐름:

  • PostQueryService(@Transactional): 글을 찾고, 조회수를 1 올리고, 트랜잭션 안에서 작성자·태그를 DTO로 옮긴다

PostQueryService.java. 위 서버 처리 순서가 어디에서 검증·조회·상태 변경으로 이어지는지 따라간다. 트랜잭션 경계는 아래 코드의 @Transactional 위치에서 확인한다.

Entity — 테이블·상태

Post.java. 실제 테이블의 컬럼·외래키와 엔티티 매핑을 대조한다. 변경된 메서드: increaseViewCount.

예외 처리 — 실패 응답

ApiExceptionHandler.java, PostNotFoundException.java. 예외가 어떤 HTTP 상태와 오류 본문으로 바뀌는지 실패 계약과 대조한다. 변경된 메서드: handlePostNotFound.

MyBatis — 상세 SELECT와 조회수 UPDATE

아래 MyBatis 대체 구현 코드의 7개 파일을 이 단계 전체 소스에 적용한다. PostReadMapper.findDetail은 post·board·member를 JOIN해 상세 응답에 필요한 열을 PostDetailRow로 읽는다.

결과가 없으면 기존 PostNotFoundException으로 404를 돌린다. 태그는 post_tag·tag JOIN으로 읽어 PostDetailResponse에 넣는다. #{id}는 요청 ID를 바인딩하므로 SQL 문자열에 직접 이어 붙이지 않는다.

PostQueryService.getDetail의 @Transactional 안에서 UPDATE post SET view_count = view_count + 1 WHERE id = #{id}를 실행하고, 응답 viewCount는 조회 직전 값에 1을 더한다. JPA의 변경 감지와 달리 MyBatis는 UPDATE를 명시적으로 쓴다.

G1·G2를 두 번 실행해 응답과 DB의 조회수가 각각 1씩 증가하는지 D1/D2로 확인한다. 이 단계에는 아직 2-3의 공개·차단 조건을 추가하지 않는다.

package org.example.common;

import jakarta.servlet.http.HttpServletRequest;
import org.example.auth.AuthenticationRequiredException;
import org.example.auth.MemberNotActiveException;
import org.example.board.BoardNotFoundException;
import org.example.board.BoardWriteForbiddenException;
import org.example.post.PostNotFoundException;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.net.URI;
import java.util.List;
import java.util.Map;

// 모든 컨트롤러의 에러 응답을 한 곳에서 RFC 9457 Problem Details로 만든다.
// 컨트롤러마다 try-catch를 두지 않는다.
@RestControllerAdvice
public class ApiExceptionHandler {

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ProblemDetail handleValidation(MethodArgumentNotValidException ex, HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, "입력값이 올바르지 않습니다.");
        problem.setInstance(URI.create(request.getRequestURI()));
        List<Map<String, String>> errors = ex.getBindingResult().getFieldErrors().stream()
                .map(fieldError -> Map.of(
                        "field", fieldError.getField(),
                        "message", fieldError.getDefaultMessage() == null ? "" : fieldError.getDefaultMessage()
                ))
                .toList();
        problem.setProperty("errors", errors);
        return problem;
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ProblemDetail handleUnreadable(HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, "요청 본문을 읽을 수 없습니다.");
        problem.setInstance(URI.create(request.getRequestURI()));
        return problem;
    }

    @ExceptionHandler(BoardNotFoundException.class)
    public ProblemDetail handleBoardNotFound(BoardNotFoundException ex, HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, ex.getMessage());
        problem.setInstance(URI.create(request.getRequestURI()));
        return problem;
    }

    @ExceptionHandler(PostNotFoundException.class)
    public ProblemDetail handlePostNotFound(PostNotFoundException ex, HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, ex.getMessage());
        problem.setInstance(URI.create(request.getRequestURI()));
        return problem;
    }

    @ExceptionHandler(BoardWriteForbiddenException.class)
    public ProblemDetail handleBoardWriteForbidden(BoardWriteForbiddenException ex, HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.FORBIDDEN, ex.getMessage());
        problem.setInstance(URI.create(request.getRequestURI()));
        return problem;
    }

    // 누구인지 모름 — WWW-Authenticate 헤더를 함께 내려준다
    @ExceptionHandler(AuthenticationRequiredException.class)
    public ResponseEntity<ProblemDetail> handleAuthenticationRequired(AuthenticationRequiredException ex,
                                                                       HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.UNAUTHORIZED, ex.getMessage());
        problem.setInstance(URI.create(request.getRequestURI()));
        return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                .header(HttpHeaders.WWW_AUTHENTICATE, "Bearer")
                .body(problem);
    }

    // 누구인지는 알지만(토큰은 유효) 회원 상태 때문에 안 됨
    @ExceptionHandler(MemberNotActiveException.class)
    public ProblemDetail handleMemberNotActive(MemberNotActiveException ex, HttpServletRequest request) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.FORBIDDEN, ex.getMessage());
        problem.setInstance(URI.create(request.getRequestURI()));
        return problem;
    }
}

package org.example.post;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.FetchType;

// post 테이블 엔티티 — like_count·comment_count는 목록 성능용 비정규화 컬럼
@Entity
@Table(name = "post")
public class Post {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "board_id", nullable = false)
    private Board board;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "author_id", nullable = false)
    private Member author;

    @Column(nullable = false)
    private String title;

    @Column(nullable = false, columnDefinition = "text")
    private String content;

    @Column(nullable = false)
    private String status;

    @Column(nullable = false)
    private String visibility;

    @Column(name = "view_count", nullable = false)
    private Integer viewCount;

    @Column(name = "like_count", nullable = false)
    private Integer likeCount;

    @Column(name = "comment_count", nullable = false)
    private Integer commentCount;

    @Column(name = "published_at")
    private LocalDateTime publishedAt;

    @Column(name = "created_at", nullable = false)
    private LocalDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private LocalDateTime updatedAt;

    @Column(name = "deleted_at")
    private LocalDateTime deletedAt;

    @ManyToMany(fetch = FetchType.LAZY)
    @JoinTable(
            name = "post_tag",
            joinColumns = @JoinColumn(name = "post_id"),
            inverseJoinColumns = @JoinColumn(name = "tag_id")
    )
    private Set<Tag> tags = new LinkedHashSet<>();

    protected Post() {
        // JPA
    }

    public Post(Board board, Member author, String title, String content,
                String status, String visibility, LocalDateTime now) {
        this.board = board;
        this.author = author;
        this.title = title;
        this.content = content;
        this.status = status;
        this.visibility = visibility;
        this.viewCount = 0;
        this.likeCount = 0;
        this.commentCount = 0;
        this.publishedAt = now;
        this.createdAt = now;
        this.updatedAt = now;
    }

    public void addTag(Tag tag) {
        tags.add(tag);
    }

    // 변경 감지로 조회수를 올린다 — 트랜잭션 커밋 시 UPDATE 한 번이 나간다
    public void increaseViewCount() {
        this.viewCount = this.viewCount + 1;
    }

    public Long getId() {
        return id;
    }

    public Board getBoard() {
        return board;
    }

    public Member getAuthor() {
        return author;
    }

    public String getTitle() {
        return title;
    }

    public String getContent() {
        return content;
    }

    public String getStatus() {
        return status;
    }

    public String getVisibility() {
        return visibility;
    }

    public Integer getViewCount() {
        return viewCount;
    }

    public Integer getLikeCount() {
        return likeCount;
    }

    public Integer getCommentCount() {
        return commentCount;
    }

    public LocalDateTime getPublishedAt() {
        return publishedAt;
    }

    public LocalDateTime getCreatedAt() {
        return createdAt;
    }

    public LocalDateTime getUpdatedAt() {
        return updatedAt;

package org.example.post;

import jakarta.validation.Valid;
import org.example.auth.AuthenticatedMember;
import org.example.auth.LoginMember;
import org.example.tag.Tag;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.net.URI;
import java.util.List;

@RestController
@RequestMapping("/api/posts")
public class PostController {

    private static final Logger log = LoggerFactory.getLogger(PostController.class);

    private final PostService postService;
    private final PostQueryService postQueryService;

    public PostController(PostService postService, PostQueryService postQueryService) {
        this.postService = postService;
        this.postQueryService = postQueryService;
    }

    @GetMapping("/{id}")
    public PostDetailResponse getOne(@PathVariable Long id) {
        return postQueryService.getDetail(id);
    }

    @PostMapping
    public ResponseEntity<PostResponse> create(@LoginMember AuthenticatedMember loginMember,
                                                @Valid @RequestBody PostCreateRequest request) {
        log.info("게시글 작성 요청 제목: {}", request.title());

        Post post = postService.create(loginMember.id(), request);
        PostResponse response = toResponse(post);

        return ResponseEntity.created(URI.create("/api/posts/" + post.getId())).body(response);
    }

    @PostMapping("/{sourceId}/copies")
    public ResponseEntity<PostResponse> copy(@PathVariable Long sourceId,
                                             @RequestParam(defaultValue = "false") boolean copyTags,
                                             @LoginMember AuthenticatedMember loginMember,
                                             @Valid @RequestBody CopyPostRequest request) {
        Post post = postService.copy(sourceId, loginMember.id(), copyTags, request);
        return ResponseEntity.created(URI.create("/api/posts/" + post.getId()))
                .body(toResponse(post));
    }

    private PostResponse toResponse(Post post) {
        List<String> tagNames = post.getTags().stream().map(Tag::getName).toList();
        AuthorResponse author = new AuthorResponse(post.getAuthor().getId(), post.getAuthor().getNickname());
        return new PostResponse(
                post.getId(),
                post.getBoard().getSlug(),
                post.getTitle(),
                post.getContent(),
                tagNames,
                post.getVisibility(),
                author,
                post.getCreatedAt()
        );
    }
}

    @Transactional
    public PostDetailResponse getDetail(Long id) {

                post.getTitle(),
                post.getContent(),
                post.getTags().stream().map(Tag::getName).toList(),
                post.getVisibility(),
                new AuthorResponse(post.getAuthor().getId(), post.getAuthor().getNickname()),
                post.getViewCount(),
                post.getLikeCount(),
                post.getCommentCount(),
                post.getCreatedAt()
        );

package org.example.post;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.List;

// 글 상세 조회와 조회수 증가를 같은 트랜잭션에서 실행한다.
@Service
public class PostQueryService {
    private final PostReadMapper mapper;

    public PostQueryService(PostReadMapper mapper) {
        this.mapper = mapper;
    }

    @Transactional
    public PostDetailResponse getDetail(Long id) {
        PostDetailRow row = mapper.findDetail(id);
        if (row == null) throw new PostNotFoundException();
        mapper.increaseViewCount(id);
        List<String> tags = mapper.findTagsByPostIds(List.of(id)).stream()
                .map(PostTagRow::name).toList();
        return new PostDetailResponse(row.id, row.boardSlug, row.title, row.content,
                tags, row.visibility, new AuthorResponse(row.authorId, row.authorNickname),
                row.viewCount + 1, row.likeCount, row.commentCount, row.createdAt);
    }
}

<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN" "https://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="org.example.post.PostReadMapper">
    <select id="findDetail" resultType="org.example.post.PostDetailRow">
        SELECT p.id, b.slug AS board_slug, p.title, p.content, p.visibility,
               m.id AS author_id, m.nickname AS author_nickname,
               p.view_count, p.like_count, p.comment_count, p.created_at
        FROM post p JOIN board b ON b.id = p.board_id
                    JOIN member m ON m.id = p.author_id
        WHERE p.id = #{id}
    </select>
    <update id="increaseViewCount">
        UPDATE post SET view_count = view_count + 1 WHERE id = #{id}
    </update>
    <select id="findTagsByPostIds" resultType="org.example.post.PostTagRow">
        SELECT pt.post_id AS postId, t.name FROM post_tag pt
        JOIN tag t ON t.id = pt.tag_id
        WHERE pt.post_id IN
        <foreach collection="postIds" item="id" open="(" separator="," close=")">#{id}</foreach>
        ORDER BY t.name
    </select>
</mapper>
