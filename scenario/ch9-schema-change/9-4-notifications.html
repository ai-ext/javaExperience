새 기능에는 새 테이블이 필요하다. 테이블을 마이그레이션으로 만들고, 기존 기능(좋아요·댓글)이 성공할 때 알림이 함께 생기게 연결한다.

기존 기능을 건드리는 일이라 같은 알림이 두 번 생기면 안 된다. 이 단계는 알림을 좋아요·댓글과 같은 트랜잭션에서 함께 저장하는 것까지다 — 그래서 알림 저장이 실패하면 좋아요도 함께 되돌려진다. "알림이 실패해도 좋아요는 성공"으로 떼어 내는 일(커밋 뒤 이벤트·별도 트랜잭션·재시도)은 R1에서 차이를 설명하는 데까지만 보고, 구현은 다음 과제로 남긴다.

        Post post = postRepository.findById(postId)
                .orElseThrow(PostNotFoundException::new);
        if (!accessPolicy.canView(viewer, post)) {
            throw new PostNotFoundException();
        }

        List<Comment> all = commentRepository.findByPostIdWithMember(postId);
        Long viewerId = viewer == null ? null : viewer.id();
        Set<Long> blockedMemberIds = viewerId == null ? Set.of() : blockPolicy.blockRelatedIds(viewerId);

        Map<Long, List<Comment>> repliesByParent = all.stream()
                .filter(c -> c.getParent() != null)
                .collect(Collectors.groupingBy(c -> c.getParent().getId(), LinkedHashMap::new, Collectors.toList()));

        List<CommentThreadResponse> result = new ArrayList<>();
        for (Comment top : all) {
            if (top.getParent() != null) {
                continue; // 답글은 부모와 함께 처리
            }
            if (blockedMemberIds.contains(top.getMember().getId())) {
                continue; // 차단 — 최상위째 뺀다(답글도 함께)
            }

            // "살아 있는 답글"은 지운 것·차단으로 빠지는 것을 제외하고 센다
            List<Comment> visibleReplies = repliesByParent.getOrDefault(top.getId(), List.of()).stream()
                    .filter(r -> r.getDeletedAt() == null)
                    .filter(r -> !blockedMemberIds.contains(r.getMember().getId()))
                    .toList();

            boolean deleted = top.getDeletedAt() != null;
            if (deleted && visibleReplies.isEmpty()) {
                continue; // 지웠고 남은 답글도 없으면 자리도 남기지 않는다
            }

            List<CommentReplyResponse> replies = visibleReplies.stream()
                    .map(r -> new CommentReplyResponse(r.getId(), toAuthor(r.getMember()), r.getContent(), r.getCreatedAt()))
                    .toList();

            result.add(new CommentThreadResponse(
                    top.getId(),
                    deleted ? null : toAuthor(top.getMember()),
                    deleted ? "삭제된 댓글입니다" : top.getContent(),
                    top.getCreatedAt(),
                    deleted,
                    replies
            ));
        }
        return result;
    }

    private Post findCommentablePost(Long postId, AuthenticatedMember loginMember) {
        Post post = postRepository.findById(postId)
                .orElseThrow(PostNotFoundException::new);
        if (!accessPolicy.canView(loginMember, post) || !PUBLISHED.equals(post.getStatus())) {
            throw new PostNotFoundException();
        }
        return post;
    }

    // 부모 댓글 검증 — 같은 글의, 최상위, 지워지지 않은 댓글이어야 한다.
    // 부모가 없으면 404(CommentNotFoundException), 있는데 규칙에 안 맞으면 400(InvalidParentCommentException).
    private Comment resolveParent(Long postId, Long parentId) {
        if (parentId == null) {
            return null;
        }
        Comment parent = commentRepository.findById(parentId)
                .orElseThrow(CommentNotFoundException::new);
        if (!parent.getPost().getId().equals(postId)) {
            throw new InvalidParentCommentException("다른 글의 댓글에는 답글을 달 수 없습니다.");
        }
        if (parent.getParent() != null) {
            throw new InvalidParentCommentException("답글에는 답글을 달 수 없습니다.");
        }
        if (parent.getDeletedAt() != null) {
            throw new InvalidParentCommentException("지워진 댓글에는 답글을 달 수 없습니다.");
        }
        return parent;
    }

    private AuthorResponse toAuthor(Member member) {
        return new AuthorResponse(member.getId(), member.getNickname());
    }

@Service
public class LikeService {

    private static final String PUBLISHED = "PUBLISHED";
    private static final int MAX_ATTEMPTS = 10;

    private final PostRepository postRepository;
    private final PostLikeRepository postLikeRepository;
    private final PostAccessPolicy accessPolicy;
    private final TransactionTemplate transactionTemplate;
    private final ApplicationEventPublisher eventPublisher;

    public LikeService(PostRepository postRepository, PostLikeRepository postLikeRepository,
                        PostAccessPolicy accessPolicy, PlatformTransactionManager transactionManager,
                        ApplicationEventPublisher eventPublisher) {
        this.postRepository = postRepository;
        this.postLikeRepository = postLikeRepository;
        this.accessPolicy = accessPolicy;
        this.transactionTemplate = new TransactionTemplate(transactionManager);
        this.eventPublisher = eventPublisher;
    }

    public LikeResponse like(Long postId, AuthenticatedMember loginMember) {
        return withDeadlockRetry(() -> transactionTemplate.execute(status -> doLike(postId, loginMember)));
    }

    public LikeResponse unlike(Long postId, AuthenticatedMember loginMember) {
        return withDeadlockRetry(() -> transactionTemplate.execute(status -> doUnlike(postId, loginMember)));
    }

    private LikeResponse doLike(Long postId, AuthenticatedMember loginMember) {
        Post post = findLikeablePost(postId, loginMember);
        if (post.getAuthor().getId().equals(loginMember.id())) {
            throw new SelfLikeNotAllowedException();
        }

        int inserted = postLikeRepository.insertIgnore(postId, loginMember.id());
        if (inserted > 0) {
            postRepository.incrementLikeCount(postId);
            // 9-4: 좋아요가 "새로" 생겼을 때만(4-1의 멱등 판단 결과 재사용). 같은 트랜잭션의 동기 리스너가 알림을 만든다
            eventPublisher.publishEvent(new PostLikedEvent(postId, loginMember.id()));
        }
        return new LikeResponse(postId, true, postRepository.findLikeCountById(postId));
    }

    private LikeResponse doUnlike(Long postId, AuthenticatedMember loginMember) {
        findLikeablePost(postId, loginMember);

        long deleted = postLikeRepository.deleteByPost_IdAndMember_Id(postId, loginMember.id());
        if (deleted > 0) {
            postRepository.decrementLikeCount(postId);
        }
        return new LikeResponse(postId, false, postRepository.findLikeCountById(postId));
    }

    // 4-3: 내가 좋아요한 글 — 좋아요 시각 최신순 커서 페이징(OFFSET 아님)
    @Transactional(readOnly = true)
    public MyLikesResponse myLikes(AuthenticatedMember loginMember, int size, String cursor) {
        Cursor after = Cursor.decode(cursor);
        List<MyLikeRow> rows = postLikeRepository.findMyLikes(
                loginMember.id(),
                after == null ? null : after.at(),
                after == null ? null : after.id(),
                size + 1);

        boolean hasNext = rows.size() > size;
        List<MyLikeRow> page = hasNext ? rows.subList(0, size) : rows;

        List<MyLikeItemResponse> items = page.stream()
                .map(like -> new MyLikeItemResponse(
                        like.getPostId(),
                        like.getTitle(),
                        new AuthorResponse(like.getAuthorId(), like.getAuthorNickname()),
                        like.getLikedAt()
                ))
                .toList();

        MyLikeRow last = page.isEmpty() ? null : page.get(page.size() - 1);
        String nextCursor = hasNext ? new Cursor(last.getLikedAt(), last.getLikeId()).encode() : null;
        return new MyLikesResponse(items, nextCursor);
    }

    // 볼 수 있는 글인가(2-3 규칙 재사용) + PUBLISHED인가 — 둘 다 아니면 404
    private Post findLikeablePost(Long postId, AuthenticatedMember loginMember) {
        Post post = postRepository.findById(postId)
                .orElseThrow(PostNotFoundException::new);
        if (!accessPolicy.canView(loginMember, post) || !PUBLISHED.equals(post.getStatus())) {
            throw new PostNotFoundException();
        }
        return post;
    }

    private LikeResponse withDeadlockRetry(java.util.function.Supplier<LikeResponse> action) {
        for (int attempt = 1; ; attempt++) {
            try {
                return action.get();
            } catch (ConcurrencyFailureException e) {
                if (attempt >= MAX_ATTEMPTS) {
                    throw e;
                }
                // 이 트랜잭션은 이미 롤백됐다 — 처음부터(새 트랜잭션으로) 다시 시도한다.
                // 바로 재시도하면 같은 상대와 또 부딪히기 쉬우니 짧게, 무작위로 간격을 둔다.
                try {
                    Thread.sleep((long) (Math.random() * 20 * attempt));
                } catch (InterruptedException interrupted) {
                    Thread.currentThread().interrupt();
                    throw e;
                }
            }
        }
    }

import org.example.auth.AuthenticatedMember;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

import java.time.Clock;
import java.time.LocalDateTime;
import java.util.List;

@Service
public class NotificationService {

    private final NotificationRepository notificationRepository;
    private final PostRepository postRepository;
    private final CommentRepository commentRepository;
    private final MemberRepository memberRepository;
    private final Clock clock;

    public NotificationService(NotificationRepository notificationRepository, PostRepository postRepository,
                               CommentRepository commentRepository, MemberRepository memberRepository, Clock clock) {
        this.notificationRepository = notificationRepository;
        this.postRepository = postRepository;
        this.commentRepository = commentRepository;
        this.memberRepository = memberRepository;
        this.clock = clock;
    }

    // 동기 이벤트 리스너가 좋아요·댓글의 기존 트랜잭션 안에서 호출한다.
    // 호출자가 트랜잭션 없이 이벤트를 발행하면 실패시켜 알림만 따로 저장되는 일을 막는다.
    @Transactional(propagation = Propagation.MANDATORY)
    public void notifyLike(Long postId, Long actorId) {
        create(postId, actorId, Notification.LIKE, null);
    }

    @Transactional(propagation = Propagation.MANDATORY)
    public void notifyComment(Long postId, Long commentId, Long actorId) {
        create(postId, actorId, Notification.COMMENT, commentId);
    }

    private void create(Long postId, Long actorId, String type, Long commentId) {
        Post post = postRepository.findById(postId).orElse(null);
        if (post == null) {
            return;
        }
        Long receiverId = post.getAuthor().getId();
        if (receiverId.equals(actorId)) {
            return; // 자기 글에 자기가 한 일은 알리지 않는다
        }
        notificationRepository.save(new Notification(
                memberRepository.getReferenceById(receiverId),
                memberRepository.getReferenceById(actorId),
                type,
                post,
                commentId == null ? null : commentRepository.getReferenceById(commentId),
                LocalDateTime.now(clock)));
    }

    @Transactional(readOnly = true)
    public NotificationListResponse list(AuthenticatedMember me, boolean unreadOnly, int size) {
        List<NotificationResponse> items = notificationRepository
                .findMine(me.id(), unreadOnly, PageRequest.of(0, size)).stream()
                .map(n -> new NotificationResponse(
                        n.getId(),
                        n.getType(),
                        new AuthorResponse(n.getActor().getId(), n.getActor().getNickname()),
                        n.getPost().getId(),
                        n.getComment() == null ? null : n.getComment().getId(),
                        n.getReadAt() != null,
                        n.getCreatedAt()))
                .toList();
        return new NotificationListResponse(items, notificationRepository.countByMember_IdAndReadAtIsNull(me.id()));
    }

    @Transactional
    public void markRead(AuthenticatedMember me, Long notificationId) {
        notificationRepository.findByIdAndMember_Id(notificationId, me.id())
                .orElseThrow(NotificationNotFoundException::new)
                .markRead(LocalDateTime.now(clock));
    }
}

// 이건 "나중에 맞추는" 안전망이다 — 4-2·5-1처럼 숫자를 원자적으로 바꿔 애초에 어긋나지 않게 하는 게 먼저다.

@Service
public class RecountService {

    private static final Logger log = LoggerFactory.getLogger(RecountService.class);

    private final MyBatisRecountMapper recountMapper;

    public RecountService(MyBatisRecountMapper recountMapper) {
        this.recountMapper = recountMapper;
    }

    // 찾기와 고치기를 한 트랜잭션에서 한다. 그 사이에 좋아요가 눌리면 응답의 after와 실제 값이
    // 조금 다를 수 있지만, UPDATE는 그 시점의 실제 행 수로 맞추므로 결과는 맞다.
    @Transactional
    public RecountResponse recount() {
        List<CountFix> likeFixed = toFixes(recountMapper.findLikeCountMismatches());
        List<CountFix> commentFixed = toFixes(recountMapper.findCommentCountMismatches());

        if (!likeFixed.isEmpty()) {
            recountMapper.fixLikeCounts();
        }
        if (!commentFixed.isEmpty()) {
            recountMapper.fixCommentCounts();
        }

        likeFixed.forEach(f -> log.info("like_count 보정 postId={} {} -> {}", f.postId(), f.before(), f.after()));
        commentFixed.forEach(f -> log.info("comment_count 보정 postId={} {} -> {}", f.postId(), f.before(), f.after()));
        log.info("정합성 배치 완료: like {}건, comment {}건 보정", likeFixed.size(), commentFixed.size());
        return new RecountResponse(likeFixed, commentFixed);
    }

    private List<CountFix> toFixes(List<CountMismatchRow> rows) {
        return rows.stream()
                .map(r -> new CountFix(r.postId, r.beforeCount, r.afterCount))
                .toList();
    }

package org.example.bookmark;

import org.example.auth.AuthenticatedMember;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.time.LocalDateTime;
import java.util.List;

@Service
public class MyBatisBookmarkFolderService {
    private static final int MAX_FOLDERS = 10;
    private final MyBatisBookmarkFolderMapper folders;

    public MyBatisBookmarkFolderService(MyBatisBookmarkFolderMapper folders) {
        this.folders = folders;
    }

    @Transactional(readOnly = true)
    public List<BookmarkFolderSummaryResponse> list(AuthenticatedMember me) {
        return folders.findSummaries(me.id());
    }

    @Transactional
    public BookmarkFolderResponse create(AuthenticatedMember me, BookmarkFolderRequest request) {
        String name = normalizeName(request.name());
        // 같은 회원의 생성 요청을 직렬화해 10개 한도를 지킨다.
        folders.lockMember(me.id());
        if (folders.countByMember(me.id()) >= MAX_FOLDERS) {
            throw new InvalidBookmarkFolderException("폴더는 10개까지 만들 수 있습니다.");
        }
        if (folders.countByName(me.id(), name, null) > 0) throw new DuplicateBookmarkFolderNameException();
        BookmarkFolderInsertRow row = new BookmarkFolderInsertRow(me.id(), name, LocalDateTime.now());
        try {
            folders.insert(row);
        } catch (DataIntegrityViolationException ex) {
            throw new DuplicateBookmarkFolderNameException();
        }
        return new BookmarkFolderResponse(row.id, name);
    }

    @Transactional
    public BookmarkFolderResponse rename(AuthenticatedMember me, Long folderId, BookmarkFolderRequest request) {
        String name = normalizeName(request.name());
        if (folders.findMine(folderId, me.id()) == null) throw new BookmarkFolderNotFoundException();
        if (folders.countByName(me.id(), name, folderId) > 0) throw new DuplicateBookmarkFolderNameException();
        try {
            if (folders.rename(folderId, me.id(), name) != 1) throw new BookmarkFolderNotFoundException();
        } catch (DataIntegrityViolationException ex) {
            throw new DuplicateBookmarkFolderNameException();
        }
        return new BookmarkFolderResponse(folderId, name);
    }

    @Transactional
    public void delete(AuthenticatedMember me, Long folderId) {
        if (folders.findMine(folderId, me.id()) == null) throw new BookmarkFolderNotFoundException();
        folders.moveAllToDefault(folderId, me.id());
        if (folders.deleteMine(folderId, me.id()) != 1) throw new BookmarkFolderNotFoundException();
    }

    private String normalizeName(String raw) {
        String name = raw == null ? "" : raw.strip();
        if (name.isEmpty() || name.length() > 50) {
            throw new InvalidBookmarkFolderException("폴더 이름은 1~50자여야 합니다.");
        }
        return name;
    }

import tools.jackson.databind.JsonNode;
import org.springframework.transaction.annotation.Transactional;
import java.time.LocalDateTime;

@Service
public class MyBatisBookmarkService {
    private final PostVisibilityMapper visiblePosts;
    private final MyBatisBookmarkMapper bookmarks;

    public MyBatisBookmarkService(PostVisibilityMapper visiblePosts, MyBatisBookmarkMapper bookmarks) {
        this.visiblePosts = visiblePosts;
        this.bookmarks = bookmarks;
    }

    @Transactional
    public BookmarkResponse bookmark(Long postId, AuthenticatedMember me, BookmarkRequest request) {
        if (visiblePosts.findVisibleDetail(postId, me.id(), me.role()) == null
                || !"PUBLISHED".equals(bookmarks.findPostStatus(postId))) {
            throw new PostNotFoundException();
        }
        Long folderId = request.folderId();
        if (folderId != null && bookmarks.findOwnedFolder(folderId, me.id()) == null) {
            throw new BookmarkFolderNotFoundException();
        }
        // UNIQUE(member_id,post_id)를 이용해 생성과 변경을 하나의 SQL로 처리한다.
        bookmarks.upsert(new BookmarkInsertRow(me.id(), postId, folderId, request.memo(), LocalDateTime.now()));
        // FOR UPDATE로 현재 행을 읽는다. 반복 PUT도 같은 bookmarkId를 돌려준다.
        BookmarkRow row = bookmarks.findBookmark(me.id(), postId);
        return new BookmarkResponse(row.id, postId, true, row.folderId, row.memo);
    }

    @Transactional
    public BookmarkResponse patchMemo(Long postId, AuthenticatedMember me,
                                      BookmarkMemoPatchRequest request) {
        if (request == null) throw new ResponseStatusException(HttpStatus.BAD_REQUEST);
        // 회원과 글 조건으로 행을 잠그고, 생략된 memo는 UPDATE하지 않는다.
        BookmarkRow row = bookmarks.findBookmark(me.id(), postId);
        if (row == null) throw new BookmarkNotFoundException();
        if (request.isMemoPresent()) {
            JsonNode value = request.getMemo();
            if (value != null && !value.isNull() && !value.isTextual()) {
                throw new ResponseStatusException(HttpStatus.BAD_REQUEST);
            }
            String memo = value == null || value.isNull() ? null : value.asText();
            if (memo != null && memo.length() > 200) {
                throw new ResponseStatusException(HttpStatus.BAD_REQUEST);
            }
            if (bookmarks.updateMemo(me.id(), postId, memo) != 1) {
                throw new BookmarkNotFoundException();
            }
            row.memo = memo;
        }
        return new BookmarkResponse(row.id, postId, true, row.folderId, row.memo);
    }

    @Transactional
    public void unbookmark(Long postId, AuthenticatedMember me) {
        bookmarks.deleteByMemberAndPost(me.id(), postId);
    }

    @Transactional
    public BookmarkResponse move(Long bookmarkId, AuthenticatedMember me, BookmarkMoveRequest request) {
        BookmarkRow row = bookmarks.findOwnedBookmark(bookmarkId, me.id());
        if (row == null) throw new BookmarkNotFoundException();
        Long folderId = request.folderId();
        if (folderId != null && bookmarks.findOwnedFolder(folderId, me.id()) == null) {
            throw new BookmarkFolderNotFoundException();
        }
        if (bookmarks.moveBookmark(bookmarkId, me.id(), folderId) != 1) {
            throw new BookmarkNotFoundException();
        }
        return new BookmarkResponse(bookmarkId, row.postId, true, folderId, row.memo);
    }

    public MyBookmarksResponse myBookmarks(AuthenticatedMember me, String folder, int size, String cursor) {
        String mode;
        Long folderId = null;
        if (folder == null) {
            mode = "ALL";
        } else if ("default".equals(folder)) {
            mode = "DEFAULT";
        } else {
            mode = "ID";
            try {
                folderId = Long.valueOf(folder);
            } catch (NumberFormatException ex) {
                throw new InvalidBookmarkFolderException("folder는 생략·default·폴더 id 중 하나여야 합니다.");
            }
            if (bookmarks.findOwnedFolder(folderId, me.id()) == null) {
                throw new BookmarkFolderNotFoundException();
            }
        }
        Cursor after = Cursor.decode(cursor);
        List<BookmarkListRow> rows = bookmarks.findMyBookmarks(me.id(), me.role(), mode, folderId,
                after == null ? null : after.at(), after == null ? null : after.id(), size + 1);
        boolean hasNext = rows.size() > size;
        List<BookmarkListRow> page = hasNext ? rows.subList(0, size) : rows;
        List<BookmarkItemResponse> items = page.stream().map(row -> {
            boolean available = Boolean.TRUE.equals(row.available);
            return new BookmarkItemResponse(row.id, row.postId, row.folderId, row.memo,
                    row.bookmarkedAt, available, available ? row.title : null,
                    available ? new AuthorResponse(row.authorId, row.authorNickname) : null);
        }).toList();
        BookmarkListRow last = page.isEmpty() ? null : page.get(page.size() - 1);
        return new MyBookmarksResponse(items,
                hasNext ? new Cursor(last.bookmarkedAt, last.id).encode() : null);
    }

    private final MyBatisCommentMapper comments;

    public MyBatisCommentQueryService(PostVisibilityMapper posts, MyBatisCommentMapper comments) {
        this.posts = posts;
        this.comments = comments;
    }

    @Transactional(readOnly = true)
    public List<CommentThreadResponse> list(Long postId, AuthenticatedMember viewer) {
        Long viewerId = viewer == null ? null : viewer.id();
        String role = viewer == null ? null : viewer.role();
        if (posts.findVisibleDetail(postId, viewerId, role) == null) throw new PostNotFoundException();
        List<CommentTreeRow> all = comments.selectThreadRows(postId);
        Set<Long> blocked = viewerId == null ? Set.of() :
                Set.copyOf(comments.selectBlockedMemberIds(viewerId));
        Map<Long, List<CommentTreeRow>> replies = all.stream()
                .filter(r -> r.parentId != null)
                .collect(Collectors.groupingBy(r -> r.parentId, LinkedHashMap::new, Collectors.toList()));
        List<CommentThreadResponse> result = new ArrayList<>();
        for (CommentTreeRow top : all) {
            if (top.parentId != null || blocked.contains(top.authorId)) continue;
            List<CommentReplyResponse> visible = replies.getOrDefault(top.id, List.of()).stream()
                    .filter(r -> r.deletedAt == null && !blocked.contains(r.authorId))
                    .map(r -> new CommentReplyResponse(r.id,
                            new AuthorResponse(r.authorId, r.authorNickname), r.content, r.createdAt))
                    .toList();
            boolean deleted = top.deletedAt != null;
            if (deleted && visible.isEmpty()) continue;
            result.add(new CommentThreadResponse(top.id,
                    deleted ? null : new AuthorResponse(top.authorId, top.authorNickname),
                    deleted ? "삭제된 댓글입니다" : top.content, top.createdAt, deleted, visible));
        }
        return result;
    }

@Service
public class LikeService {

    private static final String PUBLISHED = "PUBLISHED";
    private static final int MAX_ATTEMPTS = 10;

    private final PostRepository postRepository;
    private final PostLikeRepository postLikeRepository;
    private final PostAccessPolicy accessPolicy;
    private final TransactionTemplate transactionTemplate;
    private final ApplicationEventPublisher eventPublisher;

    public LikeService(PostRepository postRepository, PostLikeRepository postLikeRepository,
                        PostAccessPolicy accessPolicy, PlatformTransactionManager transactionManager,
                        ApplicationEventPublisher eventPublisher) {
        this.postRepository = postRepository;
        this.postLikeRepository = postLikeRepository;
        this.accessPolicy = accessPolicy;
        this.transactionTemplate = new TransactionTemplate(transactionManager);
        this.eventPublisher = eventPublisher;
    }

    public LikeResponse like(Long postId, AuthenticatedMember loginMember) {
        return withDeadlockRetry(() -> transactionTemplate.execute(status -> doLike(postId, loginMember)));
    }

    public LikeResponse unlike(Long postId, AuthenticatedMember loginMember) {
        return withDeadlockRetry(() -> transactionTemplate.execute(status -> doUnlike(postId, loginMember)));
    }

    private LikeResponse doLike(Long postId, AuthenticatedMember loginMember) {
        Post post = findLikeablePost(postId, loginMember);
        if (post.getAuthor().getId().equals(loginMember.id())) {
            throw new SelfLikeNotAllowedException();
        }

        int inserted = postLikeRepository.insertIgnore(postId, loginMember.id());
        if (inserted > 0) {
            postRepository.incrementLikeCount(postId);
            eventPublisher.publishEvent(new PostLikedEvent(postId, loginMember.id()));
        }
        return new LikeResponse(postId, true, postRepository.findLikeCountById(postId));
    }

    private LikeResponse doUnlike(Long postId, AuthenticatedMember loginMember) {
        findLikeablePost(postId, loginMember);

        long deleted = postLikeRepository.deleteByPost_IdAndMember_Id(postId, loginMember.id());
        if (deleted > 0) {
            postRepository.decrementLikeCount(postId);
        }
        return new LikeResponse(postId, false, postRepository.findLikeCountById(postId));
    }

    // 4-3: 내가 좋아요한 글 — 좋아요 시각 최신순 커서 페이징(OFFSET 아님)
    @Transactional(readOnly = true)
    public MyLikesResponse myLikes(AuthenticatedMember loginMember, int size, String cursor) {
        Cursor after = Cursor.decode(cursor);
        List<PostLike> rows = postLikeRepository.findMyLikes(
                loginMember.id(),
                after == null ? null : after.at(),
                after == null ? null : after.id(),
                PageRequest.of(0, size + 1));

        boolean hasNext = rows.size() > size;
        List<PostLike> page = hasNext ? rows.subList(0, size) : rows;

        List<MyLikeItemResponse> items = page.stream()
                .map(like -> new MyLikeItemResponse(
                        like.getPost().getId(),
                        like.getPost().getTitle(),
                        new AuthorResponse(like.getPost().getAuthor().getId(), like.getPost().getAuthor().getNickname()),
                        like.getCreatedAt()
                ))
                .toList();

        PostLike last = page.isEmpty() ? null : page.get(page.size() - 1);
        String nextCursor = hasNext ? new Cursor(last.getCreatedAt(), last.getId()).encode() : null;
        return new MyLikesResponse(items, nextCursor);
    }

    // 볼 수 있는 글인가(2-3 규칙 재사용) + PUBLISHED인가 — 둘 다 아니면 404
    private Post findLikeablePost(Long postId, AuthenticatedMember loginMember) {
        Post post = postRepository.findById(postId)
                .orElseThrow(PostNotFoundException::new);
        if (!accessPolicy.canView(loginMember, post) || !PUBLISHED.equals(post.getStatus())) {
            throw new PostNotFoundException();
        }
        return post;
    }

    private LikeResponse withDeadlockRetry(java.util.function.Supplier<LikeResponse> action) {
        for (int attempt = 1; ; attempt++) {
            try {
                return action.get();
            } catch (ConcurrencyFailureException e) {
                if (attempt >= MAX_ATTEMPTS) {
                    throw e;
                }
                // 이 트랜잭션은 이미 롤백됐다 — 처음부터(새 트랜잭션으로) 다시 시도한다.
                // 바로 재시도하면 같은 상대와 또 부딪히기 쉬우니 짧게, 무작위로 간격을 둔다.
                try {
                    Thread.sleep((long) (Math.random() * 20 * attempt));
                } catch (InterruptedException interrupted) {
                    Thread.currentThread().interrupt();
                    throw e;
                }
            }
        }
    }

public class MyBatisLikeService {
    private static final int MAX_ATTEMPTS = 10;
    private final PostVisibilityMapper visiblePosts;
    private final MyBatisLikeMapper likes;
    private final TransactionTemplate transactions;
    private final ApplicationEventPublisher eventPublisher;

    public MyBatisLikeService(PostVisibilityMapper visiblePosts, MyBatisLikeMapper likes,

        this.visiblePosts = visiblePosts;
        this.likes = likes;
        this.transactions = new TransactionTemplate(manager);
        this.eventPublisher = eventPublisher;
    }

    public LikeResponse like(Long postId, AuthenticatedMember me) {
        return withDeadlockRetry(() -> transactions.execute(status -> doLike(postId, me)));
    }

    public LikeResponse unlike(Long postId, AuthenticatedMember me) {
        return withDeadlockRetry(() -> transactions.execute(status -> doUnlike(postId, me)));
    }

    private LikeResponse doLike(Long postId, AuthenticatedMember me) {
        PostDetailRow post = findLikeablePost(postId, me);
        if (post.authorId.equals(me.id())) throw new SelfLikeNotAllowedException();
        if (likes.insertIgnoreLike(postId, me.id()) == 1) {
            likes.increaseLikeCount(postId);
            eventPublisher.publishEvent(new PostLikedEvent(postId, me.id()));
        }
        return new LikeResponse(postId, true, likes.findLikeCount(postId));
    }

    private LikeResponse doUnlike(Long postId, AuthenticatedMember me) {
        findLikeablePost(postId, me);
        if (likes.deleteLike(postId, me.id()) == 1) likes.decreaseLikeCount(postId);
        return new LikeResponse(postId, false, likes.findLikeCount(postId));
    }

    @Transactional(readOnly = true)
    public MyLikesResponse myLikes(AuthenticatedMember me, int size, String cursor) {
        LocalDateTime cursorAt = null;
        Long cursorId = null;
        if (cursor != null) {
            String[] parts = decodeCursor(cursor);
            cursorAt = LocalDateTime.parse(parts[0]);
            cursorId = Long.valueOf(parts[1]);
        }
        List<MyLikeRow> rows = likes.findMyLikes(me.id(), cursorAt, cursorId, size + 1);
        boolean hasNext = rows.size() > size;
        List<MyLikeRow> page = hasNext ? rows.subList(0, size) : rows;
        List<MyLikeItemResponse> items = page.stream()
                .map(row -> new MyLikeItemResponse(row.postId, row.title,
                        new AuthorResponse(row.authorId, row.authorNickname), row.likedAt))
                .toList();
        String nextCursor = hasNext ? encodeCursor(page.get(page.size() - 1)) : null;
        return new MyLikesResponse(items, nextCursor);
    }

    private String encodeCursor(MyLikeRow last) {
        String raw = last.likedAt + "|" + last.likeId;
        return Base64.getUrlEncoder().withoutPadding()
                .encodeToString(raw.getBytes(StandardCharsets.UTF_8));
    }

    private String[] decodeCursor(String cursor) {
        try {
            String raw = new String(Base64.getUrlDecoder().decode(cursor), StandardCharsets.UTF_8);
            String[] parts = raw.split("\\|", -1);
            if (parts.length != 2) throw new IllegalArgumentException();
            LocalDateTime.parse(parts[0]);
            if (Long.parseLong(parts[1]) < 1) throw new IllegalArgumentException();
            return parts;
        } catch (IllegalArgumentException | DateTimeException ex) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "올바르지 않은 cursor입니다.");
        }
    }

    private PostDetailRow findLikeablePost(Long postId, AuthenticatedMember me) {
        PostDetailRow post = visiblePosts.findVisibleDetail(postId, me.id(), me.role());
        if (post == null || !"PUBLISHED".equals(likes.findPostStatus(postId)))
            throw new PostNotFoundException();
        return post;

        }
    }

            JOIN FETCH p.author

        if ("true".equals(value)) return true;

        }
        try {

            if (parsed >= min && parsed <= max) return parsed;
        } catch (NumberFormatException ignored) {
            // int 범위를 벗어나는 값도 동일한 요청 오류다.

import org.example.auth.AuthenticatedMember;

import java.util.ArrayList;

// PATCH는 전달된 값만 변경한다. null은 유지하고 빈 태그 목록은 연결을 비운다.
@Service
public class PostPatchMyBatisService {
    private static final int TITLE_MAX_LENGTH = 200;
    private static final int TAGS_MAX_COUNT = 5;
    private static final int TAG_NAME_MAX_LENGTH = 30;

    private final PostVisibilityMapper visibilityMapper;
    private final PostReplaceMapper replaceMapper;

    public PostPatchMyBatisService(PostVisibilityMapper visibilityMapper,
                                   PostReplaceMapper replaceMapper) {
        this.visibilityMapper = visibilityMapper;
        this.replaceMapper = replaceMapper;
    }

    @Transactional
    public PostDetailResponse patch(Long id, AuthenticatedMember me, PostUpdateRequest request) {
        PostDetailRow current = visibilityMapper.findVisibleDetail(id, me.id(), me.role());
        if (current == null) throw new PostNotFoundException();
        if (!current.authorId.equals(me.id())) throw new PostForbiddenException();

        // 모든 입력을 확인한 뒤 쓰기를 시작한다.
        if (request.title() != null) validateTitle(request.title());
        if (request.content() != null) validateContent(request.content());
        if (request.tags() != null) validateTags(request.tags());
        if (request.visibility() != null) validateVisibility(request.visibility());

        List<String> tags;
        if (request.tags() == null) {
            tags = visibilityMapper.findTagsByPostIds(List.of(id)).stream()
                    .map(PostTagRow::name).toList();
        } else {
            List<PostReplaceTagRow> resolved = new ArrayList<>();
            Set<Long> seenIds = new HashSet<>();
            for (String name : request.tags()) {
                replaceMapper.insertTagIfMissing(name);
                PostReplaceTagRow tag = replaceMapper.findTagByName(name);
                if (seenIds.add(tag.id)) resolved.add(tag);
            }
            replaceMapper.deletePostTags(id);
            for (PostReplaceTagRow tag : resolved) replaceMapper.insertPostTag(id, tag.id);
            tags = resolved.stream().map(tag -> tag.name).toList();
        }
        replaceMapper.patchPost(id, request.title(), request.content(), request.visibility());
        return new PostDetailResponse(id, current.boardSlug,
                request.title() == null ? current.title : request.title(),
                request.content() == null ? current.content : request.content(),
                tags,
                request.visibility() == null ? current.visibility : request.visibility(),
                new AuthorResponse(current.authorId, current.authorNickname),
                current.viewCount, current.likeCount, current.commentCount, current.createdAt);
    }

    private void validateTitle(String title) {
        if (title.isBlank()) throw new InvalidPostFieldException("title", "제목을 입력해 주세요.");
        if (title.length() > TITLE_MAX_LENGTH)
            throw new InvalidPostFieldException("title", "제목은 " + TITLE_MAX_LENGTH + "자 이하여야 합니다.");
    }

    private void validateContent(String content) {
        if (content.isBlank()) throw new InvalidPostFieldException("content", "본문을 입력해 주세요.");
    }

    private void validateTags(List<String> tags) {
        if (tags.size() > TAGS_MAX_COUNT)
            throw new InvalidPostFieldException("tags", "태그는 " + TAGS_MAX_COUNT + "개까지입니다.");
        for (String tag : tags) {
            if (tag == null || tag.isBlank() || tag.length() > TAG_NAME_MAX_LENGTH)
                throw new InvalidPostFieldException("tags", "태그 이름은 1~" + TAG_NAME_MAX_LENGTH + "자여야 합니다.");
        }
    }

    private void validateVisibility(String visibility) {
        if (!"PUBLIC".equals(visibility) && !"FOLLOWERS".equals(visibility))
            throw new InvalidPostFieldException("visibility", "공개 범위는 PUBLIC 또는 FOLLOWERS여야 합니다.");
    }

    public PostListResponse search(String boardSlug, String tag, Long authorId, LocalDate from, LocalDate to,

            throw new InvalidSearchConditionException("from이 to보다 늦을 수 없습니다.");
        }

            boardId = mapper.findBoardIdBySlug(boardSlug);
            if (boardId == null) {
                return new PostListResponse(List.of(), page, size, 0, 0, false);
            }
        }
        String normalizedTag = tag == null || tag.isBlank() ? null : tag;
        PostSearchCondition condition = new PostSearchCondition(boardId, normalizedTag, authorId, from, to, q);
        LocalDateTime fromAt = from == null ? null : from.atStartOfDay();
        LocalDateTime toExclusive = to == null || to.equals(LocalDate.of(9999, 12, 31))
                ? null : to.plusDays(1).atStartOfDay();
        String qPattern = q == null || q.isBlank() ? null
                : "%" + q.strip().toLowerCase().replace("!", "!!").replace("%", "!%")
                    .replace("_", "!_") + "%";
        long offset = (long) page * size;
        Long viewerId = viewer == null ? null : viewer.id();
        List<PostListRow> rows = mapper.search(condition, viewerId, fromAt, toExclusive, qPattern, size, offset);
        long total = mapper.countSearch(condition, viewerId, fromAt, toExclusive, qPattern);
        Map<Long, List<String>> tags = rows.isEmpty() ? Map.of()
                : mapper.findTagsByPostIds(rows.stream().map(row -> row.id).toList()).stream()
                    .collect(Collectors.groupingBy(PostTagRow::postId,

        List<PostListItemResponse> items = rows.stream()
                .map(row -> toListItem(row, tags.getOrDefault(row.id, List.of()))).toList();
        return new PostListResponse(items, page, size, total,
                (int) ((total + size - 1) / size), offset + rows.size() < total);

        if (current == null) {

            replaceMapper.insertPostTag(id, tag.id);
        }

                request.content(), tags.stream().map(tag -> tag.name).toList(),
