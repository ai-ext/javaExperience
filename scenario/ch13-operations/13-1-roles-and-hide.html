#상황

"신고 처리를 운영자 한 명이 다 하니까 밀려요. 커뮤니티 매니저 몇 분께 '글 숨기기·복구'만 맡기고 싶어요. 대신 정합성 배치나 권한 주기 같은 건 운영자만 하게요. 그리고 숨긴 글은 나중에 '누가 언제 숨겼는지' 꼭 알아야 해요."

지금 코드에는 운영 권한이 문자열 비교로 흩어져 있다. 8-5의 if (!"ADMIN".equals(me.role())), 2-3 PostAccessPolicy의 ADMIN_ROLE, 글 삭제의 isAdmin이 그렇다. 역할이 하나(ADMIN) 늘 때마다 이런 if를 다 찾아 고쳐야 한다. 이번에는 두 가지를 한다.

  • 역할 → 권한 표를 한 곳에 둔다. 컨트롤러에는 "운영자인가"가 아니라 "이 일을 할 권한이 있나"만 적는다. 확인은 인터셉터 한 곳에서 한다.
  • 모더레이터(MODERATOR) 역할을 더하고, 글 숨김·복구 API를 만든다. 숨길 때 누가·언제 숨겼는지 남긴다.

티켓 T-013이 같은 숨김·복구 API를 운영자 하나 기준으로 다룬다. 이 단계는 그 API 모양(PATCH /api/admin/posts/{id}/status)을 그대로 쓰고, 그 위에 역할·권한 체계와 숨김 기록을 더한다. T-013의 R3("숨긴 시각이 없다")에 대한 답이 V41이다.

API

메서드·경로 권한 응답 200
PATCH /api/admin/members/{id}/role MANAGE_ROLE { "id", "nickname", "role" }
PATCH /api/admin/posts/{id}/status MODERATE_POST { "id", "status", "hiddenBy": { "id", "nickname" } 또는 null, "hiddenAt": 시각 또는 null }
상황 응답
숨김 글을 또 숨김 200, 처음 숨긴 사람·시각 그대로(멱등) — 첫 응답의 hiddenAt과 두 번째 응답의 값이 같아야 한다
복구 200, PUBLISHED, hiddenBy·hiddenAt=null
임시저장 글 숨기기·복구 409
없는 글·삭제된 글 404
운영자(ADMIN)의 역할 바꾸기 409 — 운영자가 0명이 되는 사고를 막는다
없는 회원·탈퇴 회원 404

필드 규칙

필드 필수 null·빈 값 허용 값 틀렸을 때
status(글 상태) 필수 안 됨 HIDDEN·PUBLISHED(대문자 그대로) 400, errors에 status
role(역할) 필수 안 됨 USER·MODERATOR — ADMIN은 API로 줄 수 없다 400, errors에 role
{id}(경로) 필수 — 숫자 숫자가 아니면 400
본문 전체 필수 — JSON JSON이 아니면 400

여기서 숨김은 사진 속 isHide boolean 대신 기존 post.status = 'HIDDEN'으로 표현한다. deleted_at은 작성자·운영자의 삭제 시각이고, hidden_at·hidden_by는 운영자가 숨긴 기록이다. 숨긴 글은 권한자가 상세를 볼 수 있지만, 삭제된 글은 권한자도 일반 상세에서 볼 수 없다.

package org.example.admin;

import java.util.EnumSet;
import java.util.Set;

// 13-1 역할 → 권한 표. 새 역할·권한이 생기면 여기만 고친다(문자열 "ADMIN" 비교를 여기저기 두지 않는다)
public enum Role {
    USER(EnumSet.noneOf(Permission.class)),
    MODERATOR(EnumSet.of(Permission.MODERATE_POST)),
    ADMIN(EnumSet.allOf(Permission.class));

    private final Set<Permission> permissions;

    Role(Set<Permission> permissions) {
        this.permissions = permissions;
    }

    public boolean can(Permission permission) {
        return permissions.contains(permission);
    }

    // member.role 문자열(DB 값)로 판정 — 모르는 값이면 아무 권한도 없다
    public static boolean has(String role, Permission permission) {
        for (Role r : values()) {
            if (r.name().equals(role)) {
                return r.can(permission);
            }
        }
        return false;
    }
}

package org.example.admin;

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.example.auth.AuthInterceptor;
import org.example.auth.AuthenticatedMember;
import org.example.auth.AuthenticationRequiredException;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;

// 13-1 권한 확인 한 곳 — AuthInterceptor(누구인가) 다음에 돈다.
// @RequirePermission이 붙은 핸들러만 본다: 로그인 안 함 → 401, 권한 없음 → 403.
// GET은 AuthInterceptor가 토큰 없이도 통과시키므로 여기서 401을 내야 한다(컨트롤러마다 null 확인을 반복하지 않는다).
@Component
public class PermissionInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        if (!(handler instanceof HandlerMethod method)) {
            return true;
        }
        RequirePermission required = method.getMethodAnnotation(RequirePermission.class);
        if (required == null) {
            required = method.getBeanType().getAnnotation(RequirePermission.class);
        }
        if (required == null) {
            return true;
        }
        AuthenticatedMember me = (AuthenticatedMember) request.getAttribute(AuthInterceptor.LOGIN_MEMBER_ATTRIBUTE);
        if (me == null) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }
        if (!Role.has(me.role(), required.value())) {
            throw new AdminOnlyException("이 작업을 할 권한이 없습니다.");
        }
        return true;
    }
}

package org.example.admin;

import org.example.auth.AuthenticatedMember;
import org.example.member.Member;
import org.example.member.MemberNotFoundException;
import org.example.member.MemberRepository;
import org.example.post.AuthorResponse;
import org.example.post.Post;
import org.example.post.PostNotFoundException;
import org.example.post.PostRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Clock;
import java.time.LocalDateTime;
import java.time.temporal.ChronoUnit;

// 13-1 운영 — 권한 확인은 PermissionInterceptor가 끝냈다. 여기는 상태 규칙만 본다.
@Service
public class ModerationService {

    private static final String WITHDRAWN = "WITHDRAWN";

    private final PostRepository postRepository;
    private final MemberRepository memberRepository;
    private final Clock clock;

    public ModerationService(PostRepository postRepository, MemberRepository memberRepository, Clock clock) {
        this.postRepository = postRepository;
        this.memberRepository = memberRepository;
        this.clock = clock;
    }

    // 없는 글·삭제된 글 404 → 임시저장 409 → 숨기기/복구(이미 그 상태면 그대로 — 처음 숨긴 사람·시각 유지)
    @Transactional
    public PostStatusResponse changePostStatus(AuthenticatedMember me, Long postId, String status) {
        // 숨김 요청 둘이 동시에 와도 첫 hiddenBy/hiddenAt을 보존하도록 글 행을 잠근다.
        Post post = postRepository.findByIdForUpdate(postId)
                .filter(p -> p.getDeletedAt() == null)
                .orElseThrow(PostNotFoundException::new);
        if ("HIDDEN".equals(status)) {
            // DATETIME은 초 단위다 — 소수 초를 그대로 넘기면 MySQL이 반올림해 저장해서(…:09.51 → :10)
            // 이번 응답의 hiddenAt과 다음에 읽은 값이 달라진다. 저장할 값을 미리 초로 자른다
            post.hide(memberRepository.getReferenceById(me.id()), LocalDateTime.now(clock).truncatedTo(ChronoUnit.SECONDS));
        } else {
            post.unhide();
        }
        return toStatusResponse(post);
    }

    // 운영자(ADMIN)의 역할은 API로 바꾸지 않는다(자기 강등으로 운영자가 0명이 되는 사고 방지) — 409
    @Transactional
    public MemberRoleResponse changeRole(Long memberId, String role) {
        Member member = memberRepository.findById(memberId)
                .filter(m -> !WITHDRAWN.equals(m.getMemberStatus()) && m.getDeletedAt() == null)
                .orElseThrow(MemberNotFoundException::new);
        if (Role.ADMIN.name().equals(member.getRole())) {

    problem.setInstance(URI.create(request.getRequestURI()));
    // 11-1 JSON 문법은 맞는데 한 필드의 형식이 틀린 경우(숫자 자리에 &quot;abc&quot;, 목록 자리에 문자열) —
    // Jackson이 남긴 경로로 어느 필드인지 errors에 담는다. 목록 안이면 Bean Validation과 같은 모양(imageIds[0]).
    // JSON 문법 자체가 깨졌으면 경로가 비어 있어 errors 없이 400만 나간다. Jackson은 첫 오류에서 멈추므로 하나만 들어간다.
    if (ex.getCause() instanceof JacksonException jackson &amp;&amp; !jackson.getPath().isEmpty()) {
        problem.setProperty(&quot;errors&quot;, List.of(Map.of(&quot;field&quot;, fieldPath(jackson), &quot;message&quot;, &quot;형식이 올바르지 않습니다.&quot;)));
    }
    return problem;
}

    return problem;
}

private static String fieldPath(JacksonException jackson) {
    StringBuilder field = new StringBuilder();
    for (JacksonException.Reference ref : jackson.getPath()) {
        if (ref.getPropertyName() != null) {
            if (!field.isEmpty()) {
                field.append('.');
            }
            field.append(ref.getPropertyName());
        } else {
            field.append('[').append(ref.getIndex()).append(']');
        }
    }
    return field.toString();
}

@Component
public class PostAccessPolicy {


private static final String PUBLISHED = &quot;PUBLISHED&quot;;
private static final String DRAFT = &quot;DRAFT&quot;;
private static final String HIDDEN = &quot;HIDDEN&quot;;
private static final String PUBLIC = &quot;PUBLIC&quot;;
private static final String FOLLOWERS = &quot;FOLLOWERS&quot;;

private final FollowRepository followRepository;
private final BlockPolicy blockPolicy;

public PostAccessPolicy(FollowRepository followRepository, BlockPolicy blockPolicy) {
    this.followRepository = followRepository;
    this.blockPolicy = blockPolicy;
}

// 글 하나를 판정한다 — 팔로우·차단은 필요할 때만 그 자리에서 조회한다
public boolean canView(AuthenticatedMember viewer, Post post) {
    return canView(viewer, post,
            authorId -&gt; followRepository.existsByFollower_IdAndFollowee_Id(viewer.id(), authorId),
            authorId -&gt; blockPolicy.isBlockedEitherWay(viewer.id(), authorId));
}

// 여러 글을 한꺼번에 판정할 때(6-3 북마크 목록) — 같은 규칙을 쓰되 팔로우·차단 정보를
// 미리 한 번에 읽어 두고 메모리에서 판정한다(항목마다 쿼리를 보내지 않는다).
public Predicate&lt;Post&gt; viewableBy(AuthenticatedMember viewer) {
    Set&lt;Long&gt; followees = viewer == null ? Set.of() : Set.copyOf(followRepository.findFolloweeIds(viewer.id()));
    Set&lt;Long&gt; blocked = viewer == null ? Set.of() : blockPolicy.blockRelatedIds(viewer.id());
    return post -&gt; canView(viewer, post, followees::contains, blocked::contains);
}

// follows/blocked는 viewer가 있을 때만 호출된다
private boolean canView(AuthenticatedMember viewer, Post post,
                        Predicate&lt;Long&gt; follows, Predicate&lt;Long&gt; blocked) {
    // 1. 삭제된 글은 작성자 본인도 못 본다
    if (post.getDeletedAt() != null) {
        return false;
    }

    Long viewerId = viewer == null ? null : viewer.id();
    Long authorId = post.getAuthor().getId();
    boolean isAuthor = viewerId != null &amp;&amp; viewerId.equals(authorId);
    // 13-1: &quot;운영자인가&quot;가 아니라 &quot;숨김 글을 다룰 권한이 있나&quot;(ADMIN·MODERATOR)
    boolean isAdmin = viewer != null &amp;&amp; Role.has(viewer.role(), Permission.MODERATE_POST);

    // 2. 상태별 허용 — DRAFT는 작성자만, HIDDEN은 운영 권한(ADMIN·MODERATOR)만
    boolean statusOk = switch (post.getStatus()) {
        case PUBLISHED -&gt; true;
        case DRAFT -&gt; isAuthor;
        case HIDDEN -&gt; isAdmin;
        default -&gt; false;
    };
    if (!statusOk) {
        return false;
    }

    // 3. PUBLISHED + FOLLOWERS 공개는 작성자 본인이거나 팔로워만
    if (PUBLISHED.equals(post.getStatus()) &amp;&amp; FOLLOWERS.equals(post.getVisibility()) &amp;&amp; !isAuthor) {
        if (viewerId == null || !follows.test(authorId)) {
            return false;
        }
    }

    // 4. 둘 사이에 차단이 있으면(어느 쪽이 차단했든) 못 본다
    if (viewerId != null &amp;&amp; !isAuthor &amp;&amp; blocked.test(authorId)) {
        return false;
    }

    // 5. 발행 글의 공개 범위가 알려진 값인지 검사한다.
    // 비로그인과 일반 열람자는 PUBLIC 또는 위에서 확인한 FOLLOWERS만 볼 수 있다.
    // DB에 PRIVATE/알 수 없는 값이 들어와도 공개로 취급하지 않는다.
    if (PUBLISHED.equals(post.getStatus()) &amp;&amp; !isAuthor
            &amp;&amp; !PUBLIC.equals(post.getVisibility())
            &amp;&amp; !FOLLOWERS.equals(post.getVisibility())) {
        return false;
    }
    return true;
}


}

private final PostRepository postRepository;
private final MemberRepository memberRepository;
private final Clock clock;
private final ModerationCommandMapper commandMapper;
private final EntityManager entityManager;

public ModerationService(PostRepository postRepository, MemberRepository memberRepository, Clock clock,
                         ModerationCommandMapper commandMapper, EntityManager entityManager) {
    this.postRepository = postRepository;
    this.memberRepository = memberRepository;
    this.clock = clock;
    this.commandMapper = commandMapper;

// 없는 글·삭제된 글 404 → 임시저장 409 → 숨기기/복구(이미 그 상태면 그대로 — 처음 숨긴 사람·시각 유지)
@Transactional
public PostStatusResponse changePostStatus(AuthenticatedMember me, Long postId, String status) {
    // 먼저 행을 잠가 동시 숨김의 응답도 첫 hiddenBy/hiddenAt과 일치시킨다.
    Post post = postRepository.findByIdForUpdate(postId)
            .filter(p -&gt; p.getDeletedAt() == null)
            .orElseThrow(PostNotFoundException::new);
    if (&quot;HIDDEN&quot;.equals(status)) {
        // DATETIME은 초 단위다 — 소수 초를 그대로 넘기면 MySQL이 반올림해 저장해서(…:09.51 → :10)
        // 이번 응답의 hiddenAt과 다음에 읽은 값이 달라진다. 저장할 값을 미리 초로 자른다
        if (&quot;DRAFT&quot;.equals(post.getStatus())) {
            throw new org.example.post.PostStatusConflictException(&quot;임시저장 글은 숨기거나 복구할 수 없습니다.&quot;);
        }
        commandMapper.hideIfPublished(postId, me.id(), LocalDateTime.now(clock).truncatedTo(ChronoUnit.SECONDS));
    } else {
        if (&quot;DRAFT&quot;.equals(post.getStatus())) {
            throw new org.example.post.PostStatusConflictException(&quot;임시저장 글은 숨기거나 복구할 수 없습니다.&quot;);
        }
        commandMapper.unhide(postId);
    }
    entityManager.refresh(post);
    if (post.getDeletedAt() != null) {
        throw new PostNotFoundException();
    }
    return toStatusResponse(post);
}

@Transactional
public MemberRoleResponse changeRole(Long memberId, String role) {
    Member member = memberRepository.findById(memberId)
            .filter(m -&gt; !WITHDRAWN.equals(m.getMemberStatus()) &amp;&amp; m.getDeletedAt() == null)
            .orElseThrow(MemberNotFoundException::new);
    if (Role.ADMIN.name().equals(member.getRole())) {
        throw new ModerationConflictException(&quot;운영자의 역할은 바꿀 수 없습니다.&quot;);
    }
    commandMapper.changeRole(memberId, role);
    entityManager.refresh(member);
    return new MemberRoleResponse(member.getId(), member.getNickname(), member.getRole());
}

    Member by = post.getHiddenBy();
    return new PostStatusResponse(post.getId(), post.getStatus(),
            by == null ? null : new AuthorResponse(by.getId(), by.getNickname()), post.getHiddenAt());
