package org.example.board.like;

import org.example.board.auth.LoginUser;
import org.example.board.common.ApiResponse;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class LikeController {

    private final LikeService likeService;

    public LikeController(LikeService likeService) {
        this.likeService = likeService;
    }

    // v1 토글 — 이미 쓰는 화면이 있어 남겨 둔다(새 화면은 v2로)
    @PostMapping("/api/v1/posts/{postId}/like")
    public ApiResponse<LikeResponse> toggle(@LoginUser Long userId, @PathVariable Long postId) {
        return ApiResponse.ok(likeService.toggle(userId, postId));
    }

    // v2 — 의미가 달라져(토글 → 상태 지정) 같은 주소에 덮지 않고 새 버전 경로로 나란히 둔다
    @PutMapping("/api/v2/posts/{postId}/like")
    public ApiResponse<LikeResponse> like(@LoginUser Long userId, @PathVariable Long postId) {
        return ApiResponse.ok(likeService.like(userId, postId));
    }

    @DeleteMapping("/api/v2/posts/{postId}/like")
    public ApiResponse<LikeResponse> unlike(@LoginUser Long userId, @PathVariable Long postId) {
        return ApiResponse.ok(likeService.unlike(userId, postId));
    }
}
