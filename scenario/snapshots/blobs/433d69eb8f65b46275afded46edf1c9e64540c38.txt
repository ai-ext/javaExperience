package org.example.board.post;

import java.time.LocalDateTime;
import java.util.List;

public record PostListItemResponse(
        Long id,
        String title,
        String visibility,
        int viewCount,
        int likeCount,
        AuthorResponse author,
        List<String> tags,
        LocalDateTime createdAt) {

    static PostListItemResponse of(Post post, List<String> tags) {
        return new PostListItemResponse(post.getId(), post.getTitle(), post.getVisibility().name(), post.getViewCount(),
                post.getLikeCount(), AuthorResponse.from(post.getAuthor()), tags, post.getCreatedAt());
    }
}
