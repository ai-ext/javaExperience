package org.example.board.comment;

import java.time.LocalDateTime;

import org.example.board.post.AuthorResponse;

public record CommentCreatedResponse(Long id, Long postId, Long parentId, String content, AuthorResponse author,
                                     LocalDateTime createdAt) {

    static CommentCreatedResponse from(Comment c) {
        return new CommentCreatedResponse(c.getId(), c.getPost().getId(),
                c.getParent() == null ? null : c.getParent().getId(), c.getContent(),
                AuthorResponse.from(c.getAuthor()), c.getCreatedAt());
    }
}
