package org.example.bookmark;

import org.example.auth.AuthenticatedMember;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.jdbc.core.JdbcTemplate;

import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

// 읽기→저장의 경쟁 상태를 실제 커밋 경계에서 검증한다. 테스트 자체에는 @Transactional을 붙이지 않는다.
@SpringBootTest
@AutoConfigureMockMvc
class BookmarkConcurrentSetTest {
    @Autowired BookmarkService bookmarks;
    @Autowired JdbcTemplate jdbc;

    @Test
    void sameMemberSettingSamePostConcurrentlyGetsOneIdAndNoErrors() throws InterruptedException {
        AuthenticatedMember member = new AuthenticatedMember(2L, "김자바", "USER");
        int parallel = 8;
        ExecutorService pool = Executors.newFixedThreadPool(parallel);
        CountDownLatch ready = new CountDownLatch(parallel);
        CountDownLatch start = new CountDownLatch(1);
        CountDownLatch done = new CountDownLatch(parallel);
        List<Long> ids = new CopyOnWriteArrayList<>();
        List<Exception> errors = new CopyOnWriteArrayList<>();
        for (int i = 0; i < parallel; i++) {
            pool.submit(() -> {
                try {
                    ready.countDown();
                    start.await();
                    ids.add(bookmarks.set(16L, member, new BookmarkRequest(null, null)).bookmarkId());
                } catch (Exception e) {
                    errors.add(e);
                } finally {
                    done.countDown();
                }
            });
        }
        boolean allReady = ready.await(10, TimeUnit.SECONDS);
        start.countDown();
        assertTrue(allReady, "모든 PUT이 출발 전에 준비돼야 한다");
        assertTrue(done.await(30, TimeUnit.SECONDS), "모든 PUT이 끝나야 한다");
        pool.shutdown();

        assertTrue(errors.isEmpty(), errors.toString());
        assertEquals(parallel, ids.size());
        assertEquals(1, ids.stream().distinct().count());
        assertEquals(1, jdbc.queryForObject(
                "SELECT COUNT(*) FROM bookmark WHERE member_id = 2 AND post_id = 16", Integer.class));
    }
}
