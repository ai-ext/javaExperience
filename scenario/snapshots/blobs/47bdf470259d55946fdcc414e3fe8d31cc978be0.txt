package org.example.board.post;

import java.time.LocalDateTime;
import java.util.List;

import org.example.board.comment.CommentResponse;

public record PostDetailResponse(
        Long id,
        String title,
        String content,
        Visibility visibility,
        int viewCount,
        int likeCount,
        boolean liked,
        AuthorResponse author,
        List<String> tags,
        List<CommentResponse> comments,
        LocalDateTime createdAt,
        LocalDateTime updatedAt) {

    public static PostDetailResponse of(Post post, int viewCount, boolean liked, List<CommentResponse> comments) {
        return new PostDetailResponse(post.getId(), post.getTitle(), post.getContent(), post.getVisibility(),
                viewCount, post.getLikeCount(), liked, AuthorResponse.from(post.getAuthor()),
                post.getTags().stream().map(Tag::getName).sorted().toList(),
                comments, post.getCreatedAt(), post.getUpdatedAt());
    }
}
