package org.example.comment;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.test.web.servlet.MockMvc;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
class CommentListControllerTest {
    @Autowired MockMvc mvc;

    @Test
    void readsOnlyLiveTopLevelCommentsAndEmptyList() throws Exception {
        mvc.perform(get("/api/posts/11/comments"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$[0].id").value(1))
                .andExpect(jsonPath("$[0].author.nickname").value("박디비"))
                .andExpect(jsonPath("$[0].deleted").value(false))
                .andExpect(jsonPath("$[1].id").value(3))
                .andExpect(jsonPath("$.length()").value(2));
        mvc.perform(get("/api/posts/3/comments"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.length()").value(0));
    }

    @Test
    void hidesUnavailablePost() throws Exception {
        mvc.perform(get("/api/posts/999999/comments")).andExpect(status().isNotFound());
        mvc.perform(get("/api/posts/9/comments")
                        .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isNotFound());
    }
}
