package org.example.admin;

import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

// 매일 새벽 4시(서울) — 수동 실행과 같은 서비스 메서드를 부른다.
// 서버가 여러 대면 모두 4시에 같이 돈다(멱등이라 결과는 같지만 낭비) — 그때는 ShedLock 같은 분산 락으로 한 대만 돌게 한다.
@Component
public class RecountScheduler {

    private final RecountService recountService;

    public RecountScheduler(RecountService recountService) {
        this.recountService = recountService;
    }

    @Scheduled(cron = "0 0 4 * * *", zone = "Asia/Seoul")
    public void recountDaily() {
        recountService.recount();
    }
}
