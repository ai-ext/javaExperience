package org.example.board.post;

import java.time.LocalDateTime;
import java.util.List;

public record PostListItemResponse(
        Long id,
        String title,
        String visibility,
        int viewCount,
        int likeCount,
        AuthorResponse author,
        List<String> tags,
        LocalDateTime createdAt) {

    static PostListItemResponse of(Post post, List<String> tags) {
        return new PostListItemResponse(post.getId(), post.getTitle(), post.getVisibility().name(), post.getViewCount(),
                post.getLikeCount(), AuthorResponse.from(post.getAuthor()), tags, post.getCreatedAt());
    }

    // 10-12: MyBatis가 읽은 한 줄(작성·수정 응답의 listItem) — 목록 항목과 같은 모양
    static PostListItemResponse of(PostListRow row, List<String> tags) {
        return new PostListItemResponse(row.getId(), row.getTitle(), row.getVisibility(), row.getViewCount(),
                row.getLikeCount(), new AuthorResponse(row.getAuthorId(), row.getAuthorNickname()), tags,
                row.getCreatedAt());
    }
}
