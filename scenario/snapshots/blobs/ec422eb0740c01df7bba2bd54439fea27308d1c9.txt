# 테스트 전용 설정 — build.gradle의 test 태스크가 test 프로파일을 켠다.
# compose DB 대신 Testcontainers가 띄운 일회용 MySQL 8.4를 쓴다(Docker가 켜져 있어야 한다).
# 빈 DB에 Flyway가 V1(스키마)을 실제로 실행한 다음, 테스트 전용 마이그레이션 V1_1이 시나리오 시드를 넣는다(9-1).
# 스키마를 만드는 길은 Flyway 하나다 — spring.sql.init은 쓰지 않는다.
spring:
  datasource:
    url: jdbc:tc:mysql:8.4:///community
  flyway:
    locations: classpath:db/migration,classpath:db/testdata
  jpa:
    properties:
      hibernate:
        # 2-4 PostListQueryCountTest가 SQL 문장 수를 센다. 여기(공용 설정)에 두는 이유:
        # 테스트별로 @TestPropertySource를 따로 쓰면 Spring이 별도 ApplicationContext를 만드는데,
        # jdbc:tc:... URL이 같아 Testcontainers는 같은 컨테이너를 재사용한다 → 두 번째 컨텍스트가
        # 스키마 초기화를 다시 돌리다 "Table already exists"로 깨진다. 모든 테스트가 같은 설정을
        # 공유해야 컨텍스트(와 컨테이너)가 하나로 캐시된다.
        generate_statistics: true
