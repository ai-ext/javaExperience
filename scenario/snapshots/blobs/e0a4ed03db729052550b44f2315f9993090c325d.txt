package org.example.board.post;

import java.util.Locale;

import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.Pattern;

// GET /api/v1/posts 의 쿼리 파라미터 묶음. cursor가 오면(빈 값 = 처음) 커서 방식, 아니면 페이지 방식
public record PostListQuery(
        @Pattern(regexp = "latest|popular", message = "정렬은 latest 또는 popular입니다.")
        String sort,
        @Min(value = 0, message = "page는 0 이상입니다.")
        Integer page,
        @Min(value = 1, message = "size는 1 이상 50 이하입니다.")
        @Max(value = 50, message = "size는 1 이상 50 이하입니다.")
        Integer size,
        String cursor,
        String tag,
        String q) {

    public PostListQuery {
        sort = sort == null ? "latest" : sort;
        size = size == null ? 20 : size;
        tag = blankToNull(tag) == null ? null : tag.strip().toLowerCase(Locale.ROOT);
        q = blankToNull(q) == null ? null : q.strip();
    }

    private static String blankToNull(String s) {
        return s == null || s.isBlank() ? null : s;
    }

    public boolean cursorMode() {
        return cursor != null;
    }
}
