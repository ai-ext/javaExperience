-- V3 — 목록용 요약(본문 앞 100자) 컬럼. 이미 행이 있는 테이블이라 세 단계로 나눈다.
-- 처음부터 NOT NULL로 추가하면 기존 행 때문에 실패하거나(엄격 모드) 빈 문자열 기본값이 들어가 버린다.
-- 수백만 행이면 ②의 UPDATE 한 번이 오래 잠그므로, id 범위로 나눠 여러 번 채우는 배치가 낫다.

-- ① NULL 허용으로 추가
ALTER TABLE post ADD COLUMN excerpt VARCHAR(120) NULL AFTER content;

-- ② 기존 글 백필 — 규칙: LEFT(content, 100)
UPDATE post SET excerpt = LEFT(content, 100);

-- ③ 다 채운 뒤 NOT NULL
ALTER TABLE post MODIFY COLUMN excerpt VARCHAR(120) NOT NULL;
