package org.example.comment;

import org.example.auth.AuthenticatedMember;
import org.example.auth.LoginMember;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;

@RestController
@RequestMapping("/api/posts/{postId}/comments")
public class CommentController {
    private final CommentQueryService query;

    public CommentController(CommentQueryService query) {
        this.query = query;
    }

    @GetMapping
    public List<CommentResponse> list(@PathVariable Long postId,
                                      @LoginMember AuthenticatedMember viewer) {
        return query.listTopLevel(postId, viewer);
    }
}
