package org.example.post;

import org.example.auth.AuthenticatedMember;
import org.example.auth.AuthenticationRequiredException;
import org.example.auth.LoginMember;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

// /api/me/... 중 글에 관한 것 — 11-1 내 임시저장 글
@RestController
public class MyPostController {

    private final PostService postService;

    public MyPostController(PostService postService) {
        this.postService = postService;
    }

    @GetMapping("/api/me/drafts")
    public MyDraftsResponse myDrafts(@LoginMember AuthenticatedMember loginMember) {
        // GET은 토큰 없이도 인터셉터를 통과한다 — 내 목록은 누구인지 알아야 하므로 여기서 401
        if (loginMember == null) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }
        return postService.myDrafts(loginMember);
    }
}
