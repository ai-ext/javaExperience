package org.example.marked;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;
import org.springframework.transaction.annotation.Transactional;
import tools.jackson.databind.ObjectMapper;

import static org.junit.jupiter.api.Assertions.assertTrue;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.put;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
@Transactional
class MarkedPostControllerTest {
    @Autowired MockMvc mockMvc;
    @Autowired ObjectMapper objectMapper;

    @Test
    void filtersUnionAndIntersectionBeforePagination() throws Exception {
        long both = createPost("[표시실습] 둘 다");
        long onlyLike = createPost("[표시실습] 좋아요만");
        like(both);
        like(onlyLike);
        mockMvc.perform(put("/api/posts/{id}/bookmark", both)
                .header("Authorization", "Bearer tok-kimjava")
                .contentType(MediaType.APPLICATION_JSON).content("{}"))
                .andExpect(status().isOk());

        mockMvc.perform(get("/api/me/posts?liked=true")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.items[0].postId").value(onlyLike))
                .andExpect(jsonPath("$.items[0].liked").value(true))
                .andExpect(jsonPath("$.items[0].bookmarked").value(false));

        mockMvc.perform(get("/api/me/posts?liked=true&bookmarked=true")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.items[0].postId").value(both))
                .andExpect(jsonPath("$.items[0].bookmarked").value(true));

        mockMvc.perform(get("/api/me/posts?bookmarked=true")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.items[0].postId").value(both));

        var result = mockMvc.perform(get("/api/me/posts?size=1&page=9999")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.items.length()").value(0))
                .andReturn();
        assertTrue(objectMapper.readTree(result.getResponse().getContentAsString())
                .get("totalElements").asLong() >= 2);

        var farPage = mockMvc.perform(get("/api/me/posts?liked=true&page=2147483647&size=100")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.items.length()").value(0))
                .andReturn();
        assertTrue(objectMapper.readTree(farPage.getResponse().getContentAsString())
                .get("totalElements").asLong() >= 2);
    }

    @Test
    void validatesAuthenticationAndPageRange() throws Exception {
        mockMvc.perform(get("/api/me/posts")).andExpect(status().isUnauthorized());
        mockMvc.perform(get("/api/me/posts?size=0")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?page=-1")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?liked=yes")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?bookmarked=1")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?liked=")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?bookmarked=")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?page=")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
        mockMvc.perform(get("/api/me/posts?size=")
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isBadRequest());
    }

    private long createPost(String title) throws Exception {
        var response = mockMvc.perform(post("/api/posts")
                .header("Authorization", "Bearer tok-lee")
                .contentType(MediaType.APPLICATION_JSON)
                .content("{\"boardSlug\":\"dev\",\"title\":\"" + title
                        + "\",\"content\":\"본문\",\"tags\":[],\"visibility\":\"PUBLIC\"}"))
                .andExpect(status().isCreated()).andReturn().getResponse();
        return objectMapper.readTree(response.getContentAsString()).get("id").asLong();
    }

    private void like(long postId) throws Exception {
        mockMvc.perform(put("/api/posts/{id}/like", postId)
                .header("Authorization", "Bearer tok-kimjava"))
                .andExpect(status().isOk());
    }
}
