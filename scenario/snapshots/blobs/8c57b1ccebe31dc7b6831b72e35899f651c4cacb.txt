-- V4 — 알림. 내 글에 좋아요·댓글이 달리면 글 작성자(member_id)에게 쌓인다.
-- 인덱스 (member_id, read_at, created_at, id): "내 안 읽은 알림 최신순"과 안 읽은 개수(COUNT)를
-- 같은 인덱스로 처리한다(같음 조건 member_id·read_at IS NULL → 정렬 created_at, id).
CREATE TABLE notification (
    id         BIGINT      NOT NULL AUTO_INCREMENT,
    member_id  BIGINT      NOT NULL,
    actor_id   BIGINT      NOT NULL,
    type       VARCHAR(20) NOT NULL,
    post_id    BIGINT      NOT NULL,
    comment_id BIGINT      NULL,
    read_at    DATETIME    NULL,
    created_at DATETIME    NOT NULL,
    PRIMARY KEY (id),
    KEY ix_notification_member_unread (member_id, read_at, created_at, id),
    CONSTRAINT fk_notification_member FOREIGN KEY (member_id) REFERENCES member (id),
    CONSTRAINT fk_notification_actor FOREIGN KEY (actor_id) REFERENCES member (id),
    CONSTRAINT fk_notification_post FOREIGN KEY (post_id) REFERENCES post (id),
    CONSTRAINT fk_notification_comment FOREIGN KEY (comment_id) REFERENCES comment (id),
    CONSTRAINT ck_notification_type CHECK (type IN ('LIKE', 'COMMENT'))
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4 COLLATE = utf8mb4_0900_ai_ci COMMENT = '알림 — 좋아요(LIKE)·댓글(COMMENT)';
