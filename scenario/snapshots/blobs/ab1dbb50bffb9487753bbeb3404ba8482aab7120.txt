package org.example.social;

import org.springframework.stereotype.Component;

import java.util.Set;

// 차단 판정 한 곳 — PostAccessPolicy(2-3)와 댓글 트리 조립(5-3)이 같이 쓴다
@Component
public class BlockPolicy {

    private final MemberBlockRepository memberBlockRepository;

    public BlockPolicy(MemberBlockRepository memberBlockRepository) {
        this.memberBlockRepository = memberBlockRepository;
    }

    public boolean isBlockedEitherWay(Long a, Long b) {
        return memberBlockRepository.existsByBlocker_IdAndBlocked_Id(a, b)
                || memberBlockRepository.existsByBlocker_IdAndBlocked_Id(b, a);
    }

    // 여러 건을 한꺼번에 판정할 때 쓴다 — 나와 차단 관계인 회원 id 전부(쿼리 한 번)
    public Set<Long> blockRelatedIds(Long memberId) {
        return Set.copyOf(memberBlockRepository.findBlockRelatedIds(memberId));
    }
}
