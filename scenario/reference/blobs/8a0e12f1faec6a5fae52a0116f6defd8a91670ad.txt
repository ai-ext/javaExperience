package org.example.revision;

import org.junit.jupiter.api.Test;

import java.util.List;

import static org.junit.jupiter.api.Assertions.assertEquals;

class RevisionTagCodecTest {
    @Test
    void 태그_이름의_쉼표와_역슬래시를_보존한다() {
        List<String> tags = List.of("a,b", "a\\b", "java");
        assertEquals(tags, RevisionTagCodec.decode(RevisionTagCodec.encode(tags)));
        assertEquals("java,spring", RevisionTagCodec.encode(List.of("java", "spring")));
        assertEquals(List.of(), RevisionTagCodec.decode(""));
    }
}
