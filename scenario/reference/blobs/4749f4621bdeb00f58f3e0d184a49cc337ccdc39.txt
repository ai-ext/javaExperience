-- V20(11-2) — 글에 붙이는 이미지. 파일 자체는 디스크(app.upload-dir)에, DB에는 "어떤 파일인가"만 둔다.
-- post_id는 업로드 직후엔 NULL(아직 글에 안 붙음)이고, 글을 쓸 때 imageIds로 연결하면 채워진다.
-- stored_name은 서버가 지은 이름(무작위 32자 + 확장자) — 사용자가 보낸 파일 이름(original_name)은 경로에 쓰지 않는다.
CREATE TABLE post_image (
    id            BIGINT       NOT NULL AUTO_INCREMENT,
    uploader_id   BIGINT       NOT NULL,
    post_id       BIGINT       NULL,
    stored_name   VARCHAR(64)  NOT NULL,
    original_name VARCHAR(255) NOT NULL,
    content_type  VARCHAR(50)  NOT NULL,
    size_bytes    INT          NOT NULL,
    created_at    DATETIME     NOT NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uk_post_image_stored_name (stored_name),
    KEY ix_post_image_post (post_id, id),
    CONSTRAINT fk_post_image_uploader FOREIGN KEY (uploader_id) REFERENCES member (id),
    CONSTRAINT fk_post_image_post FOREIGN KEY (post_id) REFERENCES post (id)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4 COLLATE = utf8mb4_0900_ai_ci COMMENT = '글 첨부 이미지 — post_id NULL은 아직 글에 안 붙은 업로드';
