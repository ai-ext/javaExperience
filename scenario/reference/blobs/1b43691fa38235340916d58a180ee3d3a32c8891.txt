-- V21(11-3) — "이 회원이 이 글을 이 날짜에 봤다" 기록. 같은 회원·같은 글·같은 날은 한 행뿐이다(PK).
-- 조회수(post.view_count)는 이 표에 처음 행이 생길 때만 +1 한다. 비로그인 조회는 기록하지 않는다(매번 +1).
CREATE TABLE post_view (
    post_id    BIGINT   NOT NULL,
    member_id  BIGINT   NOT NULL,
    viewed_on  DATE     NOT NULL,
    created_at DATETIME NOT NULL,
    PRIMARY KEY (post_id, member_id, viewed_on),
    KEY ix_post_view_member (member_id),
    CONSTRAINT fk_post_view_post FOREIGN KEY (post_id) REFERENCES post (id),
    CONSTRAINT fk_post_view_member FOREIGN KEY (member_id) REFERENCES member (id)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4 COLLATE = utf8mb4_0900_ai_ci COMMENT = '글 조회 기록 — 회원·글·날짜당 한 행(조회수 중복 방지)';
