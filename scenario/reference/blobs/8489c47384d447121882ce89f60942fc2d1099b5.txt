package org.example.bookmark;

// JSON 노드로 타입을 검사한 뒤 필드별 400 Problem Details로 바꾼다.
public class InvalidBookmarkFieldException extends RuntimeException {
    private final String field;

    public InvalidBookmarkFieldException(String field, String message) {
        super(message);
        this.field = field;
    }

    public String getField() {
        return field;
    }
}
