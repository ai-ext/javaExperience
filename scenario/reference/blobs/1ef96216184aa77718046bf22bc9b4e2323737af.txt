package org.example.social;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class RelationService {

    private final RelationQueryRepository relationQueryRepository;

    public RelationService(RelationQueryRepository relationQueryRepository) {
        this.relationQueryRepository = relationQueryRepository;
    }

    @Transactional(readOnly = true)
    public List<Long> mutuals(Long memberId) {
        return relationQueryRepository.findMutualIds(memberId);
    }

    @Transactional(readOnly = true)
    public List<Long> followBack(Long memberId) {
        return relationQueryRepository.findFollowBackIds(memberId);
    }

    @Transactional(readOnly = true)
    public List<SuggestionResponse> suggestions(Long memberId, int limit) {
        return relationQueryRepository.findSuggestions(memberId, limit).stream()
                .map(r -> new SuggestionResponse(r.getMemberId(), r.getNickname(), r.getMutualCount()))
                .toList();
    }
}
