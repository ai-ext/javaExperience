#LikeService

src/main/java/org/example/board/like/LikeService.java

package org.example.board.like;

import org.example.board.common.ApiException;
import org.example.board.common.ErrorCode;
import org.example.board.post.Post;
import org.example.board.post.PostRepository;
import org.example.board.post.Visibility;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class LikeService {

    private final PostLikeRepository postLikeRepository;
    private final PostRepository postRepository;

    public LikeService(PostLikeRepository postLikeRepository, PostRepository postRepository) {
        this.postLikeRepository = postLikeRepository;
        this.postRepository = postRepository;
    }

    // 토글: 없으면 만들고, 있으면 지운다. like_count도 함께(한 트랜잭션)
    @Transactional
    public LikeResponse toggle(Long userId, Long postId) {
        findVisiblePost(postId, userId);
        PostLikeId id = new PostLikeId(postId, userId);
        boolean liked;
        if (postLikeRepository.existsById(id)) {
            postLikeRepository.deleteById(id);
            if (postRepository.changeLikeCount(postId, userId, Visibility.PUBLIC, -1) == 0)
                throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
            liked = false;
        } else {
            postLikeRepository.save(new PostLike(postId, userId));
            if (postRepository.changeLikeCount(postId, userId, Visibility.PUBLIC, 1) == 0)
                throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
            liked = true;
        }
        return new LikeResponse(liked, postRepository.findLikeCount(postId));
    }

    private Post findVisiblePost(Long postId, Long userId) {
        Post post = postRepository.findByIdAndDeletedAtIsNull(postId)
                .orElseThrow(() -> new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다."));
        if (post.getVisibility() == Visibility.PRIVATE && !post.getAuthor().getId().equals(userId)) {
            throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        }
        return post;
    }
}
코드 설명
@Transactional 좋아요 행과 like_count를 함께 바꾸거나 함께 되돌린다. @Modifying 쿼리는 트랜잭션이 없으면 실행조차 안 된다(망가뜨리기 ①)
findVisiblePost(postId, userId) 글이 없음·삭제·남의 비공개면 404. 10-9의 댓글과 같은 규칙
if (postLikeRepository.existsById(id)) 있으면 → 지우고 −1, 없으면 → 만들고 +1. 이것이 토글
postLikeRepository.save(new PostLike(postId, userId)) 키를 우리가 정해서 넣는 엔티티라, Spring Data는 "새 것인지" 몰라 SELECT로 한 번 확인한 뒤 INSERT한다

package org.example.board.comment;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

// 평평한 댓글 목록(시간순) → 최상위 + replies 트리
public final class CommentTree {

    private CommentTree() {
    }

    public static List<CommentResponse> build(List<CommentRow> flat) {
        Map<Long, List<CommentResponse>> repliesByParent = new LinkedHashMap<>();
        for (CommentRow c : flat) {
            if (c.parentId != null && c.deletedAt == null) {
                repliesByParent.computeIfAbsent(c.parentId, k -> new ArrayList<>()).add(CommentResponse.reply(c));
            }
        }
        List<CommentResponse> result = new ArrayList<>();
        for (CommentRow c : flat) {
            if (c.parentId != null) {
                continue;
            }
            List<CommentResponse> replies = repliesByParent.getOrDefault(c.id, List.of());
            // 지운 댓글에 살아 있는 답글이 없으면 자리도 남기지 않는다
            if (c.deletedAt != null && replies.isEmpty()) {
                continue;
            }
            result.add(CommentResponse.top(c, replies));
        }
        return result;
    }
}

@Service
public class PostQueryService {
    private final PostQueryMapper mapper;

    public PostQueryService(PostQueryMapper mapper) {
        this.mapper = mapper;
    }

    @Transactional(readOnly = true)
    public Object list(PostListQuery query, Long loginUserId) {
        if (query.cursorMode()) {
            return cursorList(query, loginUserId);
        }
        int page = query.page() == null ? 0 : query.page();
        long offset = (long) page * query.size();
        List<PostListRow> rows = mapper.findPage(query, loginUserId, offset);
        long total = mapper.count(query, loginUserId);
        return new PageResponse<>(withTags(rows), page, query.size(), total, offset + rows.size() < total);
    }

    private CursorResponse<PostListItemResponse> cursorList(PostListQuery query, Long loginUserId) {
        // 인기순은 좋아요가 바뀌면 순서가 흔들려 커서가 글을 빠뜨리거나 두 번 준다 — 최신순만 허락한다
        if ("popular".equals(query.sort())) {
            throw ApiException.invalid("cursor", "커서 페이징은 최신순(latest)에서만 쓸 수 있습니다.");
        }
        if (query.page() != null) {
            throw ApiException.invalid("page", "page와 cursor는 함께 쓸 수 없습니다.");
        }
        Long cursorId = parseCursor(query.cursor());
        // size + 1행으로 다음 페이지 존재 여부를 판단한다.
        List<PostListRow> rows = mapper.findAfterCursor(query, loginUserId, cursorId, query.size() + 1);
        boolean hasNext = rows.size() > query.size();
        List<PostListRow> page = hasNext ? rows.subList(0, query.size()) : rows;
        String next = hasNext ? String.valueOf(page.get(page.size() - 1).id) : null;
        return new CursorResponse<>(withTags(page), next);
    }

    private Long parseCursor(String cursor) {
        if (cursor.isBlank()) {
            return null;
        }
        try {
            return Long.valueOf(cursor);
        } catch (NumberFormatException e) {
            throw ApiException.invalid("cursor", "커서 값이 올바르지 않습니다.");
        }
    }

    // 글과 태그를 JOIN한 채 페이징하지 않는다 — 이번 페이지 글들의 태그를 IN 한 번으로 읽는다.
    private List<PostListItemResponse> withTags(List<PostListRow> rows) {
        if (rows.isEmpty()) {
            return List.of();
        }
        Map<Long, List<String>> tagsByPost = mapper.findTagsByPostIds(rows.stream().map(row -> row.id).toList())
                .stream()
                .collect(Collectors.groupingBy(PostTagRow::postId, Collectors.mapping(PostTagRow::name, Collectors.toList())));
        return rows.stream()
                .map(row -> PostListItemResponse.of(row, tagsByPost.getOrDefault(row.id, List.of())))
                .toList();
    }

        Set<Tag> tags = findOrCreateTags(request.tags());
        Post post = new Post(author, request.title(), request.content(),

        postMapper.insertPost(post);
        for (Tag tag : tags) postMapper.linkTag(post.getId(), tag.getId());

    }

    @Transactional
    public PostDetailResponse detail(Long postId, Long loginUserId) {

        postMapper.increaseViewCount(postId);
        boolean liked = loginUserId != null && postMapper.countLike(postId, loginUserId) > 0;
        return response(rows, rows.get(0).viewCount + 1, liked, postId);
    }

    @Transactional
    public PostDetailResponse update(Long userId, Long postId, PostRequest request) {
        findWritable(userId, postId, "작성자만 수정할 수 있습니다.");
        LocalDateTime updatedAt = LocalDateTime.now().truncatedTo(ChronoUnit.SECONDS);
        if (postMapper.updatePost(postId, request.title(), request.content(), request.visibility(), updatedAt) == 0)
            throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        postMapper.unlinkTags(postId);
        for (Tag tag : findOrCreateTags(request.tags())) postMapper.linkTag(postId, tag.getId());
        List<PostDetailRow> rows = postMapper.findDetailRows(postId);
        boolean liked = postMapper.countLike(postId, userId) > 0;
        return response(rows, rows.get(0).viewCount, liked, postId);
    }

    @Transactional
    public void delete(Long userId, Long postId) {
        findWritable(userId, postId, "작성자만 삭제할 수 있습니다.");
        if (postMapper.softDelete(postId, LocalDateTime.now().truncatedTo(ChronoUnit.SECONDS)) == 0)
            throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
    }

    private void findWritable(Long userId, Long postId, String message) {
        List<PostDetailRow> rows = findVisibleRows(postId, userId);
        if (!rows.get(0).authorId.equals(userId)) throw new ApiException(ErrorCode.FORBIDDEN, message);
    }

    private List<PostDetailRow> findVisibleRows(Long postId, Long loginUserId) {
        List<PostDetailRow> rows = postMapper.findDetailRows(postId);
        if (rows.isEmpty()) throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        PostDetailRow first = rows.get(0);
        if ("PRIVATE".equals(first.visibility) && !first.authorId.equals(loginUserId))
            throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        return rows;
    }

    private PostDetailResponse response(List<PostDetailRow> rows, int viewCount, boolean liked, Long postId) {
        PostDetailRow first = rows.get(0);
        List<String> tags = new ArrayList<>();
        for (PostDetailRow row : rows) if (row.tagName != null) tags.add(row.tagName);
        return new PostDetailResponse(first.id, first.title, first.content,
                Visibility.valueOf(first.visibility), viewCount, first.likeCount,
                liked, new AuthorResponse(first.authorId, first.authorNickname), tags,
                CommentTree.build(postMapper.findCommentRows(postId)), first.createdAt, first.updatedAt);
    }

    private Set<Tag> findOrCreateTags(List<String> names) {
        Set<Tag> tags = new LinkedHashSet<>();
        for (String name : names) {
            Tag tag = postMapper.findTagByName(name);
            if (tag == null) {
                tag = new Tag(name);
                postMapper.insertTag(tag);
            }
            tags.add(tag);
        }
        return tags;
    }
