03◇무엇을 만들까설계 포인트열기 ↓
  • N+1이 생기는 자리. 댓글 목록을 findByPostId로 읽고 응답을 만들며 comment.getAuthor().getNickname()을 부르면, 작성자가 LAZY라 댓글마다 SELECT ... FROM users가 나간다. 댓글 6개면 1 + 6번이다. 해결은 "필요한 것을 처음 쿼리에서 같이 가져오기"다. JPQL의 join fetch를 쓴다.
  • 컬렉션 두 개를 한 번에 fetch join하지 않는다. 글 + 태그 + 댓글을 한 JPQL에 모두 join fetch하면 행이 곱으로 불어난다(태그 2 × 댓글 6 = 12행). 컬렉션이 둘 이상이면 Hibernate가 거부하기도 한다(MultipleBagFetchException). 글·작성자·태그 한 쿼리, 댓글·작성자 한 쿼리로 나눈다.
  • 조회수는 "읽고 +1 해서 저장"하지 않는다. 엔티티의 viewCount를 +1 하고 변경 감지에 맡기면, 동시에 두 명이 열 때 둘 다 10을 읽고 11을 쓴다. UPDATE posts SET view_count = view_count + 1을 DB에서 바로 실행한다(@Modifying @Query). 응답에는 읽었던 값 + 1을 싣는다. 두 요청이 동시에 10을 읽으면 각 응답은 11일 수 있지만 DB의 최종 값은 12다. 응답 값은 이 요청의 증가를 반영한 값이며, 동시 요청까지 포함한 최신 합계를 보장하지 않는다.
  • 트리는 메모리에서. 댓글을 시간순 평평한 목록으로 한 번에 읽고, 자바에서 parent_id별로 묶어 트리를 만든다. 부모 댓글마다 답글을 다시 조회하면 그게 또 N+1이다.
  • 남의 비공개는 404. 403("권한 없음")은 "그 글이 있다"를 알려 준다. 접근 규칙은 G2처럼 한곳에 모아 두면 10-8·10-9에서 다시 쓴다.

@Transactional
public PostDetailResponse detail(Long postId, Long loginUserId) {
    Post post = postRepository.findDetailById(postId)
            .orElseThrow(() -> new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다."));
    if (post.getVisibility() == Visibility.PRIVATE && !post.getAuthor().getId().equals(loginUserId)) {
        throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
    }
    postRepository.increaseViewCount(postId);
    boolean liked = loginUserId != null && postRepository.countLike(postId, loginUserId) > 0;
    return PostDetailResponse.of(post, post.getViewCount() + 1, liked,
            CommentTree.build(commentRepository.findAllWithAuthorByPostId(postId)));
}

없는 글·삭제된 글·남의 비공개 글은 조회수를 올리기 전에 404로 끝난다. 응답 조회수는 DB의 원자적 증가 결과와 맞춘다.

    }

    public static List<CommentResponse> build(List<Comment> flat) {
        Map<Long, List<CommentResponse>> repliesByParent = new LinkedHashMap<>();
        for (Comment c : flat) {
            if (c.getParent() != null && !c.isDeleted()) {
                repliesByParent.computeIfAbsent(c.getParent().getId(), k -> new ArrayList<>()).add(CommentResponse.reply(c));
            }
        }
        List<CommentResponse> result = new ArrayList<>();
        for (Comment c : flat) {
            if (c.getParent() != null) {
                continue;
            }
            List<CommentResponse> replies = repliesByParent.getOrDefault(c.getId(), List.of());
            // 지운 댓글에 살아 있는 답글이 없으면 자리도 남기지 않는다
            if (c.isDeleted() && replies.isEmpty()) {
                continue;
            }
            result.add(CommentResponse.top(c, replies));
        }
        return result;

import java.util.LinkedHashSet;

import org.example.board.comment.CommentRepository;
import org.example.board.comment.CommentTree;
import org.example.board.common.ApiException;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

// 게시글 쓰기 — JPA
@Service
public class PostService {

    private final PostRepository postRepository;
    private final TagRepository tagRepository;
    private final UserRepository userRepository;
    private final CommentRepository commentRepository;

    public PostService(PostRepository postRepository, TagRepository tagRepository, UserRepository userRepository,
                       CommentRepository commentRepository) {
        this.postRepository = postRepository;
        this.tagRepository = tagRepository;
        this.userRepository = userRepository;
        this.commentRepository = commentRepository;
    }

    @Transactional
    public PostDetailResponse create(Long userId, PostCreateRequest request) {
        User author = userRepository.findById(userId)
                .orElseThrow(() -> new ApiException(ErrorCode.UNAUTHORIZED));
        Post post = new Post(author, request.title(), request.content(),
                Visibility.valueOf(request.visibility()), findOrCreateTags(request.tags()));
        postRepository.save(post);
        return PostDetailResponse.of(post, post.getViewCount(), false, List.of());
    }

    // 상세: 보이는 글인지 확인 → 조회수 +1 → 좋아요 여부·댓글 트리와 함께
    @Transactional
    public PostDetailResponse detail(Long postId, Long loginUserId) {
        Post post = postRepository.findDetailById(postId)
                .orElseThrow(() -> new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다."));
        boolean isAuthor = post.getAuthor().getId().equals(loginUserId);
        if (post.getVisibility() == Visibility.PRIVATE && !isAuthor) {
            // 남의 비공개 글은 "있지만 못 본다"(403)가 아니라 "없다"(404) — 있다는 사실도 흘리지 않는다
            throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        }
        postRepository.increaseViewCount(postId);
        boolean liked = loginUserId != null && postRepository.countLike(postId, loginUserId) > 0;
        return PostDetailResponse.of(post, post.getViewCount() + 1, liked,
                CommentTree.build(commentRepository.findAllWithAuthorByPostId(postId)));
    }

    // 이름으로 찾고, 없으면 새로 만든다(tags.name은 UNIQUE)
    private Set<Tag> findOrCreateTags(List<String> names) {
        Set<Tag> tags = new LinkedHashSet<>();
        for (String name : names) {
            tags.add(tagRepository.findByName(name).orElseGet(() -> tagRepository.save(new Tag(name))));
        }
        return tags;
    }
}

package org.example.board.post;

import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;

import org.example.board.comment.CommentTree;
import org.example.board.common.ApiException;
import org.example.board.common.ErrorCode;
import org.example.board.user.User;
import org.example.board.user.UserMapper;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class PostService {
    private final PostMapper postMapper;
    private final UserMapper userMapper;

    public PostService(PostMapper postMapper, UserMapper userMapper) {
        this.postMapper = postMapper;
        this.userMapper = userMapper;
    }

    @Transactional
    public PostDetailResponse create(Long userId, PostCreateRequest request) {
        User author = userMapper.findById(userId);
        if (author == null) throw new ApiException(ErrorCode.UNAUTHORIZED);
        Set<Tag> tags = findOrCreateTags(request.tags());
        Post post = new Post(author, request.title(), request.content(),
                Visibility.valueOf(request.visibility()), tags);
        postMapper.insertPost(post);
        for (Tag tag : tags) postMapper.linkTag(post.getId(), tag.getId());
        return PostDetailResponse.of(post, post.getViewCount(), false, List.of());
    }

    @Transactional
    public PostDetailResponse detail(Long postId, Long loginUserId) {
        List<PostDetailRow> rows = postMapper.findDetailRows(postId);
        if (rows.isEmpty()) throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        PostDetailRow first = rows.get(0);
        if ("PRIVATE".equals(first.visibility) && !first.authorId.equals(loginUserId))
            throw new ApiException(ErrorCode.NOT_FOUND, "게시글을 찾을 수 없습니다.");
        postMapper.increaseViewCount(postId);
        boolean liked = loginUserId != null && postMapper.countLike(postId, loginUserId) > 0;
        List<String> tags = new ArrayList<>();
        for (PostDetailRow row : rows) if (row.tagName != null) tags.add(row.tagName);
        return new PostDetailResponse(first.id, first.title, first.content,
                Visibility.valueOf(first.visibility), first.viewCount + 1, first.likeCount,
                liked, new AuthorResponse(first.authorId, first.authorNickname), tags,
                CommentTree.build(postMapper.findCommentRows(postId)), first.createdAt, first.updatedAt);
    }

    private Set<Tag> findOrCreateTags(List<String> names) {
        Set<Tag> tags = new LinkedHashSet<>();
        for (String name : names) {
            Tag tag = postMapper.findTagByName(name);
            if (tag == null) {
                tag = new Tag(name);
                postMapper.insertTag(tag);
            }
            tags.add(tag);
        }
        return tags;
    }
}

힌트 1 — 글·작성자·태그를 한 번에


@Query("select distinct p from Post p join fetch p.author left join fetch p.tags "
        + "where p.id = :id and p.deletedAt is null")

태그가 없는 글도 나와야 하니 태그는 left join fetch다. distinct는 태그 수만큼 같은 글이 겹치는 것을 막는다.






힌트 2 — 댓글을 작성자와 함께


Comment 엔티티(comments)를 만든다. post·author·parent는 모두 @ManyToOne(fetch = LAZY)다. 저장소에서 이렇게 읽는다.


@Query("select c from Comment c join fetch c.author where c.post.id = :postId order by c.createdAt, c.id")
List<Comment> findAllWithAuthorByPostId(Long postId);


c.getParent().getId()는 부모를 따로 조회하지 않는다. 지연 로딩 프록시도 id는 이미 알고 있다.
