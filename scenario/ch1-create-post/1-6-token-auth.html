package org.example.auth;

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.example.member.Member;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.stereotype.Component;
import org.springframework.web.cors.CorsUtils;
import org.springframework.web.servlet.HandlerInterceptor;

import java.time.Clock;
import java.time.LocalDateTime;

// /api/** 요청의 토큰을 한 곳에서 확인한다(컨트롤러마다 반복하지 않는다).
// GET이고 헤더가 없으면 비로그인으로 통과시킨다(2-1에서 필요).
@Component
public class AuthInterceptor implements HandlerInterceptor {

    public static final String LOGIN_MEMBER_ATTRIBUTE = "loginMember";
    private static final String BEARER_PREFIX = "Bearer ";
    private static final String ACTIVE_STATUS = "ACTIVE";

    private final ApiTokenRepository apiTokenRepository;
    private final Clock clock;

    public AuthInterceptor(ApiTokenRepository apiTokenRepository, Clock clock) {
        this.apiTokenRepository = apiTokenRepository;
        this.clock = clock;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        // CORS 사전 요청(OPTIONS)은 토큰 없이 온다 — 막으면 브라우저의 요청이 전부 CORS로 실패한다
        if (CorsUtils.isPreFlightRequest(request)) {
            return true;
        }

        String header = request.getHeader(HttpHeaders.AUTHORIZATION);
        boolean isGet = HttpMethod.GET.matches(request.getMethod());

        if (header == null) {
            if (isGet) {
                return true;
            }
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }
        if (!header.startsWith(BEARER_PREFIX)) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }

        String token = header.substring(BEARER_PREFIX.length()).trim();
        ApiToken apiToken = apiTokenRepository.findByTokenFetchMember(token)
                .orElseThrow(() -> new AuthenticationRequiredException("인증이 필요합니다."));

        LocalDateTime now = LocalDateTime.now(clock);
        boolean expired = !now.isBefore(apiToken.getExpiresAt());
        if (apiToken.getRevokedAt() != null || expired) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }

        Member member = apiToken.getMember();
        if (!ACTIVE_STATUS.equals(member.getMemberStatus())) {
            throw new MemberNotActiveException("정지되었거나 탈퇴한 회원입니다.");
        }

        request.setAttribute(LOGIN_MEMBER_ATTRIBUTE,
                new AuthenticatedMember(member.getId(), member.getNickname(), member.getRole()));
        return true;
    }
}

@Service
public class PostService {

    private final PostRepository postRepository;
    private final BoardRepository boardRepository;
    private final MemberRepository memberRepository;
    private final TagRepository tagRepository;

    public PostService(PostRepository postRepository, BoardRepository boardRepository,
                        MemberRepository memberRepository, TagRepository tagRepository) {
        this.postRepository = postRepository;
        this.boardRepository = boardRepository;
        this.memberRepository = memberRepository;
        this.tagRepository = tagRepository;
    }

    private static final String ADMIN_WRITE_ROLE = "ADMIN";
    private static final String ADMIN_ROLE = "ADMIN";
    private static final String DEFAULT_VISIBILITY = "PUBLIC";

    @Transactional
    public Post create(Long authorId, PostCreateRequest request) {
        Board board = boardRepository.findBySlug(request.boardSlug())
                .orElseThrow(() -> new BoardNotFoundException(request.boardSlug()));
        // 작성자는 요청 본문이 아니라 토큰으로 인증된 회원에서 받는다(authorId 같은 값은 무시한다)
        Member author = memberRepository.getReferenceById(authorId);

        if (ADMIN_WRITE_ROLE.equals(board.getWriteRole()) && !ADMIN_ROLE.equals(author.getRole())) {
            throw new BoardWriteForbiddenException(board.getSlug());
        }

        String visibility = request.visibility() != null ? request.visibility() : DEFAULT_VISIBILITY;
        LocalDateTime now = LocalDateTime.now();
        Post post = new Post(board, author, request.title(), request.content(),
                "PUBLISHED", visibility, now);

        for (String tagName : distinctTagNames(request.tags())) {
            Tag tag = tagRepository.findByName(tagName)
                    .orElseGet(() -> tagRepository.save(new Tag(tagName, now)));
            post.addTag(tag);
        }

        Post saved = postRepository.save(post);
        // 트랜잭션(영속성 컨텍스트) 안에서 지연 로딩 연관을 미리 읽어 둔다.
        // open-in-view: false라서 컨트롤러가 트랜잭션 밖에서 건드리면 LazyInitializationException이 난다.
        saved.getAuthor().getNickname();
        saved.getTags().forEach(Tag::getName);
        return saved;
    }

    private List<String> distinctTagNames(List<String> tags) {
        if (tags == null) {
            return List.of();
        }
        Set<String> seen = new LinkedHashSet<>(tags);
        return List.copyOf(seen);
    }

import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.stereotype.Component;
import org.springframework.web.cors.CorsUtils;
import org.springframework.web.servlet.HandlerInterceptor;

import java.time.Clock;
import java.time.LocalDateTime;

// /api/** 요청의 토큰을 한 곳에서 확인한다(컨트롤러마다 반복하지 않는다).
// GET이고 헤더가 없으면 비로그인으로 통과시킨다(2-1에서 필요).
@Component
public class AuthInterceptor implements HandlerInterceptor {

    public static final String LOGIN_MEMBER_ATTRIBUTE = "loginMember";
    private static final String BEARER_PREFIX = "Bearer ";
    private static final String ACTIVE_STATUS = "ACTIVE";

    private final AuthTokenMapper authTokenMapper;
    private final Clock clock;

    public AuthInterceptor(AuthTokenMapper authTokenMapper, Clock clock) {
        this.authTokenMapper = authTokenMapper;
        this.clock = clock;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        // CORS 사전 요청(OPTIONS)은 토큰 없이 온다 — 막으면 브라우저의 요청이 전부 CORS로 실패한다
        if (CorsUtils.isPreFlightRequest(request)) {
            return true;
        }

        String header = request.getHeader(HttpHeaders.AUTHORIZATION);
        boolean isGet = HttpMethod.GET.matches(request.getMethod());

        if (header == null) {
            if (isGet) {
                return true;
            }
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }
        if (!header.startsWith(BEARER_PREFIX)) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }

        String token = header.substring(BEARER_PREFIX.length()).trim();
        AuthTokenRow tokenRow = authTokenMapper.findByToken(token);
        if (tokenRow == null) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }

        LocalDateTime now = LocalDateTime.now(clock);
        boolean expired = !now.isBefore(tokenRow.expiresAt);
        if (tokenRow.revokedAt != null || expired) {
            throw new AuthenticationRequiredException("인증이 필요합니다.");
        }

        if (!ACTIVE_STATUS.equals(tokenRow.memberStatus)) {
            throw new MemberNotActiveException("정지되었거나 탈퇴한 회원입니다.");
        }

        request.setAttribute(LOGIN_MEMBER_ATTRIBUTE,
                new AuthenticatedMember(tokenRow.memberId, tokenRow.nickname, tokenRow.role));
        return true;
    }
}

import org.example.board.BoardNotFoundException;
import org.example.board.BoardWriteForbiddenException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

// 게시판·작성자 확인부터 글·태그 연결까지 Mapper SQL로 실행한다.
@Service
public class PostService {
    private static final String ADMIN_ROLE = "ADMIN";
    private static final String DEFAULT_VISIBILITY = "PUBLIC";
    private final PostCreationMapper mapper;

    public PostService(PostCreationMapper mapper) {
        this.mapper = mapper;
    }

    @Transactional
    public PostResponse create(Long authorId, PostCreateRequest request) {
        BoardWriteRow board = mapper.findBoardBySlug(request.boardSlug());
        if (board == null) {
            throw new BoardNotFoundException(request.boardSlug());
        }
        // authorId는 요청 JSON이 아니라 AuthInterceptor가 확인한 토큰에서 왔다.
        AuthorWriteRow author = mapper.findAuthorById(authorId);
        if (author == null) {
            throw new IllegalStateException("인증된 작성자를 찾을 수 없습니다.");
        }
        if (ADMIN_ROLE.equals(board.writeRole) && !ADMIN_ROLE.equals(author.role)) {
            throw new BoardWriteForbiddenException(board.slug);
        }

        PostInsert post = new PostInsert();
        post.boardId = board.id;
        post.authorId = author.id;
        post.title = request.title();
        post.content = request.content();
        post.visibility = request.visibility() == null ? DEFAULT_VISIBILITY : request.visibility();
        post.now = LocalDateTime.now();
        mapper.insertPost(post);

        Map<Long, String> storedTags = new LinkedHashMap<>();
        for (String tagName : distinctTagNames(request.tags())) {
            mapper.insertTagIfAbsent(tagName, post.now);
            TagWriteRow tag = mapper.findTagByName(tagName);
            if (!storedTags.containsKey(tag.id)) {
                mapper.linkTag(post.id, tag.id);
                storedTags.put(tag.id, tag.name);
            }
        }
        return new PostResponse(post.id, board.slug, post.title, post.content,
                List.copyOf(storedTags.values()), post.visibility,
                new AuthorResponse(author.id, author.nickname), post.now);
    }

    private List<String> distinctTagNames(List<String> tags) {
        if (tags == null) {
            return List.of();
        }
        Set<String> seen = new LinkedHashSet<>(tags);
        return List.copyOf(seen);
    }
}

힌트 1 — 어디서 확인할까


HandlerInterceptor의 preHandle은 컨트롤러 메서드의 매개변수(@RequestBody)를 만들기 전에 실행된다. 그래서 C9(깨진 본문 + 토큰 없음 → 401)를 맞추기 쉽다.


확인한 회원은 request.setAttribute(...)로 넘긴다. /api/**에 적용하되 0장의 공개 실습 API /api/start/**는 excludePathPatterns로 뺀다(WebMvcConfigurer.addInterceptors).


/hello도 /api/** 밖이라 대상이 아니다. 2-1부터는 로그인 없이 보는 GET이 생기므로, "GET이고 Authorization 헤더가 없으면 통과"를 지금 넣어 두거나 2-1에서 고친다(헤더가 있는데 틀리면 GET이라도 401).

@LoginMember 같은 애노테이션을 만들고 HandlerMethodArgumentResolver로 그 매개변수에 회원을 넣어 주면 컨트롤러가 create(@LoginMember LoginMember me, @Valid @RequestBody PostCreateRequest req)처럼 깔끔해진다.
